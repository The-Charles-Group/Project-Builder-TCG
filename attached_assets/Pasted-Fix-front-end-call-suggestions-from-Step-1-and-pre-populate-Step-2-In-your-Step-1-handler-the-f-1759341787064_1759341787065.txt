In your Step 1 handler (the function that runs when the user clicks “Analyze with AI”), after you process the RFP text / file, add:

// Step 1: after extracting text or getting the file
async function analyzeWithAI() {
  const fileEl = document.querySelector('#rfpFile');
  const textEl = document.querySelector('#rfpText');
  const rfpText = (textEl?.value || '').trim();

  let data;
  if (fileEl?.files?.length) {
    const form = new FormData();
    form.append('file', fileEl.files[0]);
    data = await fetch('/api/suggest_by_file', { method: 'POST', body: form }).then(r=>r.json()); // suggested[]
  } else {
    data = await fetch('/api/suggest_by_text', {
      method: 'POST',
      headers: {'Content-Type':'application/json'},
      body: JSON.stringify({ rfp_text: rfpText })
    }).then(r=>r.json()); // suggested[]
  }

  // Persist for Step 2
  window.APP = window.APP || {};
  window.APP.rfpText = rfpText;
  window.APP.aiSuggestions = data?.suggested || [];

  // Move to Step 2 and render
  showStep(2);
  renderAISuggestions(window.APP.aiSuggestions);
}

Then wire it:

document.querySelector('#retainersToggle').addEventListener('change', onToggleRetainers);

async function onToggleRetainers(e) {
  const enabled = e.target.checked;
  window.APP = window.APP || {};
  if (!enabled) {
    window.APP.retainers = [];          // none
    renderRetainerPanel([]);            // hide/clear green box
    return;
  }

  const payload = {
    rfp_text: window.APP.rfpText || '',
    deliverable_codes: Array.from(window.APP.selectedCodes || [])
  };
  const rec = await fetch('/api/retainer_detect', {
    method: 'POST',
    headers: {'Content-Type':'application/json'},
    body: JSON.stringify(payload)
  }).then(r=>r.json()); // {retainers:[{deliverable_code, suggested_months,...}]}

  // Persist and render
  window.APP.retainers = (rec.retainers || []).map(r => ({
    deliverable_code: r.deliverable_code,
    months: r.suggested_months || 12
  }));
  renderRetainerPanel(window.APP.retainers);  // show the green inputs
}
