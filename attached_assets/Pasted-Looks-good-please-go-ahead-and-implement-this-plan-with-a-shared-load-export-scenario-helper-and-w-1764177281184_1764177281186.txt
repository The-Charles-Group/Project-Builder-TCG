1. Shared helper: load_export_scenario

Please add a helper that every export endpoint will use:

def load_export_scenario(session_id: Optional[str], payload: XMLExportPayload) -> Dict[str, Any]:
    """
    Priority order:
    0) Working scenario in SCENARIO_STORE (session_id)
    1) payload.scenario with items
    2) payload.scenario without items + selected_deliverables (build items)
    3) selected_deliverables only (build full scenario)
    """

    scenario = None

    # 0) Preferred: working scenario from SCENARIO_STORE
    if session_id:
        state = get_session_state(session_id)
        scenario = get_working_scenario(session_id)
        if scenario:
            print(f"[XML EXPORT] Using working scenario from SCENARIO_STORE for session {session_id}")
            scenario = _recompute_totals(scenario)

    # 1) payload.scenario already has items
    if scenario is None and payload.scenario and payload.scenario.get("items"):
        scenario = _inflate_components_if_missing(payload.scenario)
        print("[XML EXPORT] Using scenario from payload.scenario (with items)")

    # 2) payload.scenario without items, but we have selected_deliverables
    if scenario is None and payload.scenario and payload.selected_deliverables:
        # reuse existing Case 2 logic
        # (build items into payload.scenario from selected_deliverables / DB)
        scenario = build_scenario_from_payload_and_codes(payload)
        print("[XML EXPORT] Built scenario from payload.scenario + selected_deliverables")

    # 3) Only selected_deliverables
    if scenario is None and payload.selected_deliverables:
        # reuse existing Case 3 logic
        scenario = build_scenario_from_codes(payload.selected_deliverables, payload)
        print("[XML EXPORT] Built scenario purely from selected_deliverables")

    if scenario is None:
        raise HTTPException(400, "No scenario or deliverables provided for XML export")

    scenario = _inflate_components_if_missing(scenario)

    # Debug: verify hours before WBS build
    items = scenario.get("items", [])
    if items:
        first = items[0]
        print("[XML CHECK] First item:",
              first.get("deliverable_code"),
              "Planned_Hours=", first.get("Planned_Hours"),
              "hours=", first.get("hours"),
              "total_hours=", first.get("total_hours"))

    return scenario
