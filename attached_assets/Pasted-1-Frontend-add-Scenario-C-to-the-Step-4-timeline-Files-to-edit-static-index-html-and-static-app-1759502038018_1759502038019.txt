1B. JS — state, wiring, rendering, and reorder (Step 4)

Add/merge the following into static/app.js.

Assumes you already populate A/B from /api/build and C from /api/build_scenario_c. We just persist them into a single state.scenarios and render.

// ---------- Global UI state ----------
const state = {
  projectName: '',
  projectStartISO: '',            // Keep the ISO string the user picked in Step 3
  scenarios: { A: null, B: null, C: null }
};

// Utility formatters
const fmtMoney = (n) => (n == null ? '0' : Number(n).toLocaleString());
const el = (id) => document.getElementById(id);

// Call this right after Step 3 loads /api/last_upload_name and when user types
function setProjectName(name) {
  state.projectName = name || '';
}

// Call this when user changes the Start Date in Step 3
function setProjectStart(iso) {
  state.projectStartISO = iso || ''; // e.g., "2025-10-06T09:00:00"
}

// After /api/build -> persist and render Step 4 for A & B
function storeAB(payload) {
  state.scenarios.A = payload.scenarios?.A || null;
  state.scenarios.B = payload.scenarios?.B || null;
  // Budget banners for A & B
  renderBudget('A');
  renderBudget('B');
  // Timelines become available
  renderTimeline('A');
}

// After /api/build_scenario_c -> persist and render C tab
function storeC(payloadC) {
  state.scenarios.C = payloadC?.C || null;
  renderBudget('C');
  enableTab('C', !!state.scenarios.C?.items?.length);
  if (document.querySelector('#tabC').classList.contains('is-active')) {
    renderTimeline('C');
  }
}

// --- Tabs wiring
function wireTimelineTabs() {
  const tabs = document.querySelectorAll('#timelineTabs .tab');
  tabs.forEach(btn => {
    btn.addEventListener('click', () => {
      const letter = btn.dataset.letter;
      tabs.forEach(t => t.classList.toggle('is-active', t === btn));
      // toggle timelines, dates, and save buttons
      ['A','B','C'].forEach(x => {
        document.getElementById(`timeline${x}`).classList.toggle('is-hidden', x !== letter);
        document.getElementById(`dates${x}`).classList.toggle('is-hidden', x !== letter);
        document.getElementById(`saveOrder${x}`).classList.toggle('is-hidden', x !== letter);
      });
      renderTimeline(letter);
    });
  });
}

function enableTab(letter, enabled) {
  const tab = el(`tab${letter}`);
  if (!tab) return;
  tab.disabled = !enabled;
  tab.title = enabled ? '' : 'Build this scenario first';
}

// --- Budget banner renderer (works for A, B, C)
function renderBudget(letter) {
  const sc = state.scenarios[letter];
  const box = el(`budget${letter}`);
  if (!box || !sc) return;

  const info = sc.budget_info;
  if (!info || !info.client_budget_usd) {
    box.hidden = true;
    return;
  }

  const delta = (info.total_price || 0) - (info.client_budget_usd || 0);
  const pct = (info.client_budget_usd > 0) ? Math.abs(delta) / info.client_budget_usd * 100 : 0;
  box.hidden = false;
  box.classList.toggle('over', delta > 0);
  box.classList.toggle('under', delta <= 0);
  box.textContent = delta > 0
    ? `Budget Analysis: Over by $${fmtMoney(delta)} (${pct.toFixed(1)}%)`
    : `Budget Analysis: Under by $${fmtMoney(-delta)} (${pct.toFixed(1)}%)`;
}

// --- List renderer (drag & drop) for any scenario letter
function renderTimeline(letter) {
  const sc = state.scenarios[letter];
  const wrap = el(`timeline${letter}`);
  const dateLbl = el(`dates${letter}`);
  if (!wrap || !sc) return;

  wrap.innerHTML = '';
  (sc.items || []).forEach((it, idx) => {
    const card = document.createElement('div');
    card.className = 'timeline-item';
    card.draggable = true;
    card.dataset.code = it.deliverable_code;
    const first = it.schedule && it.schedule[0]?.start_date;
    const last  = it.schedule && it.schedule[it.schedule.length - 1]?.end_date;
    const durDays = (it.schedule || []).reduce((a, b) => a + Number(b.duration_days || 0), 0);

    card.innerHTML = `
      <div class="drag">⋮⋮</div>
      <div class="meta">
        <div class="title">${it.deliverable || it.deliverable_code}</div>
        <div class="subtext">${first || '—'} → ${last || '—'} · ${durDays || 0} days · ${Math.round(it.total_hours || 0)}h · $${fmtMoney(it.price || 0)}</div>
      </div>
    `;
    wrap.appendChild(card);
  });

  // simple drag & drop
  let dragging = null;
  wrap.querySelectorAll('.timeline-item').forEach(item => {
    item.addEventListener('dragstart', e => {
      dragging = item;
      item.classList.add('dragging');
    });
    item.addEventListener('dragend', e => {
      item.classList.remove('dragging');
      dragging = null;
    });
    item.addEventListener('dragover', e => {
      e.preventDefault();
      const target = e.currentTarget;
      if (!dragging || target === dragging) return;
      const rect = target.getBoundingClientRect();
      const before = (e.clientY - rect.top) < rect.height / 2;
      wrap.insertBefore(dragging, before ? target : target.nextSibling);
    });
  });

  // scenario dates (from server)
  const s = sc.timeline?.start;
  const e = sc.timeline?.end;
  if (s && e) {
    const d1 = new Date(s), d2 = new Date(e);
    const days = Math.max(1, Math.round((d2 - d1) / (1000*60*60*24)) + 1);
    dateLbl.textContent = `Scenario ${letter}: ${s} → ${e} · ~${days} days`;
  } else {
    dateLbl.textContent = '';
  }

  // enable Save button only if 2+ items
  el(`saveOrder${letter}`).disabled = (sc.items || []).length < 2;

  // Budget banner refresh
  renderBudget(letter);
}

// --- Save order -> POST /api/reorder_timeline (works for A/B/C)
async function saveOrder(letter) {
  const sc = state.scenarios[letter];
  if (!sc) return;

  const container = el(`timeline${letter}`);
  const codes = Array.from(container.querySelectorAll('.timeline-item'))
    .map(x => x.dataset.code);

  // included_map should accompany the reorder; if missing, backend derives safely, but we pass it anyway
  const included_map = {};
  (sc.items || []).forEach(it => {
    included_map[it.deliverable_code] = it.included_task_groups || [];
  });

  const payload = {
    scenario_letter: letter,
    deliverable_codes: codes,
    included_map,
    project_start: (state.projectStartISO || '').split('T')[0] || null,
    use_slack: sc.use_slack ?? true,
    slack_after_internal: sc.slack_after_internal ?? 1,
    slack_after_client: sc.slack_after_client ?? 2,
    slack_global_pct: sc.slack_global_pct ?? 0.0,
    complexity: sc.complexity || 'Advanced',
    tier: sc.tier || 'T2_MediumVolume'
  };

  const res = await fetch('/api/reorder_timeline', {
    method: 'POST',
    headers: {'Content-Type': 'application/json'},
    body: JSON.stringify(payload)
  });
  if (!res.ok) {
    const err = await res.text();
    alert(`Save order failed: ${err}`);
    return;
  }
  const data = await res.json();
  // IMPORTANT: keep the full scenario returned by the server to avoid drift
  state.scenarios[letter] = data.scenario;
  renderTimeline(letter);
}

// Wire up save buttons and tabs once (run at Step 4 mount)
function wireStep4() {
  wireTimelineTabs();
  ['A','B','C'].forEach(letter => {
    const btn = el(`saveOrder${letter}`);
    if (btn) btn.addEventListener('click', () => saveOrder(letter));
  });

  // Exports (A+B+C)
  el('exportXlsxABC').addEventListener('click', exportXlsxABC);
  el('exportXmlABC').addEventListener('click', exportXmlABC);
}

// Combined exports (A+B+C) — uses your existing endpoints
async function exportXlsxABC() {
  const body = {
    project_name: state.projectName || '',
    scenario_a: state.scenarios.A || {},
    scenario_b: state.scenarios.B || {},
    scenario_c: state.scenarios.C || {}
  };
  const res = await fetch('/api/export_workbook_abc', {
    method: 'POST', headers: {'Content-Type': 'application/json'},
    body: JSON.stringify(body)
  });
  const blob = await res.blob();
  downloadBlob(blob, fileNameFromResponse(res) || 'Scenarios-A-B-C.xlsx');
}

async function exportXmlABC() {
  const body = {
    project_name: state.projectName || '',
    scenario_a: state.scenarios.A || {},
    scenario_b: state.scenarios.B || {},
    scenario_c: state.scenarios.C || {},
    // the XML route accepts start-date settings inside the exporter; we keep defaults
  };
  const res = await fetch('/api/export_workbook_xml_abc', {
    method: 'POST', headers: {'Content-Type': 'application/json'},
    body: JSON.stringify(body)
  });
  const blob = await res.blob();
  downloadBlob(blob, fileNameFromResponse(res) || 'Scenarios-A-B-C.zip');
}

// Helpers to download with the server-suggested filename
function fileNameFromResponse(res) {
  const cd = res.headers.get('Content-Disposition') || '';
  const m = cd.match(/filename="?([^"]+)"?/i);
  return m ? m[1] : null;
}
function downloadBlob(blob, filename) {
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url; a.download = filename || 'download';
  document.body.appendChild(a); a.click(); a.remove();
  URL.revokeObjectURL(url);
}

4) “Project name” → Workfront project title

No change needed server‑side: the WBS builder inserts a root row (WBS=1) whose Task_Name is the Project name, which becomes the Workfront project title once imported. If you still see “Project from Scenario A,” double‑check you’re sending project_name with the export request; the builder uses that string.

5) Why hours/prices sometimes became 0 after “Build Scenarios → Build Timeline”

That happened if the frontend discarded the server’s full scenario and kept only the array of items (or re‑rendered with stale items). In the code above we always replace state.scenarios[letter] with the scenario returned by /api/reorder_timeline to keep hours/pricing/schedules in sync with the backend store. The reorder endpoint also recalculates scenario timeline.start/end for you.
