Add a small store so Deliverables → Components → L3 remain in sync:

// static/js/step2-store.js
export const store = {
  selectedDeliverables: new Set(),               // Set<deliverableCode>
  // Map<deliverableCode, Set<componentName>>
  selectedComponents: new Map(),
  // Map<deliverableCode, Map<componentName, Set<l3Task>>>
  selectedL3: new Map(),

  ensure(d) {
    if (!this.selectedComponents.has(d)) this.selectedComponents.set(d, new Set());
    if (!this.selectedL3.has(d)) this.selectedL3.set(d, new Map());
  },

  toggleDeliverable(d, checked) {
    if (checked) this.selectedDeliverables.add(d);
    else {
      this.selectedDeliverables.delete(d);
      this.selectedComponents.delete(d);
      this.selectedL3.delete(d);
    }
  },

  toggleComponent(d, c, checked) {
    this.ensure(d);
    const set = this.selectedComponents.get(d);
    if (checked) set.add(c); else {
      set.delete(c);
      // also drop all L3 under this component
      const m = this.selectedL3.get(d);
      m.delete(c);
    }
  },

  toggleL3(d, c, t, checked) {
    this.ensure(d);
    const m = this.selectedL3.get(d);
    if (!m.has(c)) m.set(c, new Set());
    const set = m.get(c);
    if (checked) set.add(t); else set.delete(t);
  }
};

# main.py – place after /api/step2/l3  (FastAPI)
@app.post("/api/step2/l3/bulk")
def step2_l3_bulk(payload: dict):
    """
    payload: {"deliverable": "deck_strategy", "components": ["brief","art_direction",...]}
    returns: {"brief": ["deck_build","internal_review",...], "art_direction": [...]}
    """
    if not DB.loaded: DB.load()
    dcode = str(payload.get("deliverable","")).strip()
    comps = [str(x).strip() for x in payload.get("components",[])]
    df = DB.all_rows.copy()
    out = {}
    for comp in comps:
        rows = df[(df["Deliverable_Code"]==dcode) & (df["Component"]==comp)]
        col = "Task_Label" if "Task_Label" in rows.columns else ("task_group" if "task_group" in rows.columns else None)
        tasks = []
        if col:
            tasks = sorted({t for t in rows[col].fillna("").astype(str).str.strip().tolist() if t and t!="nan"})
        out[comp] = tasks
    return out

Frontend timeline math (Step‑4 view): count business days only, 9:00–17:00, skipping Sat/Sun and app holidays. End date must be the next business day 17:00, never a weekend.

function addBusinessDays(startDate, businessDays, holidays = new Set()) {
  // startDate = Date at 09:00
  const d = new Date(startDate);
  let added = 0;
  while (added < businessDays) {
    d.setDate(d.getDate() + 1);
    const dow = d.getDay(); // 0=Sun ... 6=Sat
    const ymd = d.toISOString().slice(0,10);
    if (dow!==0 && dow!==6 && !holidays.has(ymd)) added++;
  }
  // set to 17:00 local
  d.setHours(17,0,0,0);
  return d;
}

async function refreshL3(deliverableCode) {
  // gather selected components under this deliverable
  const comps = [...(store.selectedComponents.get(deliverableCode)||[])];
  if (comps.length===0) { renderL3([]); return; }

  // Option A: BULK endpoint (preferred)
  const res = await fetch('/api/step2/l3/bulk', {
    method:'POST', headers:{'Content-Type':'application/json'},
    body: JSON.stringify({ deliverable: deliverableCode, components: comps })
  });
  const byComp = await res.json(); // {comp: [tasks...]}

  // Merge for UI (union), but keep comp→tasks map for chips & persistence
  const rows = [];
  for (const c of comps) {
    for (const t of (byComp[c]||[])) rows.push({component:c, task:t});
  }
  // Deduplicate by {component, task} (guard against dup rows)
  const key = new Set(); const merged = [];
  for (const r of rows) {
    const k = r.component+'::'+r.task;
    if (!key.has(k)) { key.add(k); merged.push(r); }
  }
  renderL3(merged);
}
