5. Sync to backend + recompute totals
Task D – syncScenarioToBackend(scenario) helper

Architect already has some version of this, but this is the shape we want:

async function syncScenarioToBackend(scenario) {
  const sessionId = window.SESSION_ID;

  const payload = {
    session_id: sessionId,
    scenario,                 // full scenario object
    // include version/hash fields if you’re using them:
    client_version: scenario.client_version || 0,
  };

  const res = await fetch('/api/scenario/sync', {
    method: 'POST',
    headers: {'Content-Type': 'application/json'},
    body: JSON.stringify(payload),
  });

  const data = await res.json();
  if (!data.success) {
    throw new Error(data.error || 'Scenario sync failed');
  }

  // Server returns authoritative scenario & totals after merge
  const mergedScenario = data.scenario || scenario;
  window.SCENARIO_A = mergedScenario;
  return mergedScenario;
}
