@app.post("/api/reorder_timeline")
def api_reorder_timeline(p: ReorderPayload):
    if not DB.loaded: DB.load()
    letter = (p.scenario_letter or "A").upper()

    # Canonicalize: accept names or codes safely
    order_codes = [_canonical_deliverable_code(x) for x in p.deliverable_codes]
    included_map = { _canonical_deliverable_code(k): v for k, v in (p.included_map or {}).items() }

    items = []
    cursor_date = None
    for code in order_codes:
        # Included TGs: client map → else derive from DB
        included = included_map.get(code)
        if not included:
            sub = DB.all_rows[DB.all_rows["Deliverable_Code"].astype(str) == str(code)]
            included = sorted(set(sub["task_group"].dropna().astype(str).tolist()))

        start_for_this = p.project_start if cursor_date is None else str(cursor_date)
        sched = DB.build_schedule(
            deliverable_code=code,
            included_task_groups=included,
            complexity=p.complexity, tier=p.tier,
            use_slack=p.use_slack,
            slack_after_internal=p.slack_after_internal,
            slack_after_client=p.slack_after_client,
            slack_global_pct=p.slack_global_pct,
            project_start=start_for_this,
            scenario_letter=letter
        )

        if sched:
            d_start = sched[0]["start_date"]
            d_end   = sched[-1]["end_date"]
            d_days  = sum(int(x["duration_days"]) for x in sched)
            y, m, d = map(int, d_end.split("-"))
            cursor_date = datetime.date(y, m, d) + datetime.timedelta(days=1)
        else:
            d_start = d_end = start_for_this
            d_days = 0

        items.append({
            "deliverable_code": code,
            "schedule": sched,
            "start": d_start,
            "end": d_end,
            "duration_days": d_days
        })

    # Persist to scenario so exports match
    scens = _current_scenarios()
    scen = scens.get(letter)
    if scen and "items" in scen:
        old = {it["deliverable_code"]: it for it in scen["items"]}
        scen["items"] = [old.get(c, {"deliverable_code": c}) for c in order_codes]
        new_by_code = {x["deliverable_code"]: x for x in items}
        for it in scen["items"]:
            x = new_by_code.get(it["deliverable_code"])
            if x:
                it.update({
                    "schedule": x["schedule"],
                    "start": x["start"],
                    "end": x["end"],
                    "duration_days": x["duration_days"]
                })
        if p.project_start:
            scen["project_start"] = p.project_start
        scens[letter] = scen
        return {"items": scen["items"]}

    return {"items": items}


Why this matters: this is the same canonicalizing/persisting shape already used in your docs and avoids the “names vs codes” mismatch that previously left Step‑4 empty.

// --- Utility: POST JSON
async function postJSON(url, body) {
  const res = await fetch(url, {
    method: "POST",
    headers: {"Content-Type":"application/json"},
    body: JSON.stringify(body || {})
  });
  if (!res.ok) throw new Error(`${url} -> ${res.status}`);
  return await res.json();
}

// --- Utility: download a FileResponse
async function downloadPOST(url, body) {
  const res = await fetch(url, {
    method: "POST",
    headers: {"Content-Type":"application/json"},
    body: JSON.stringify(body || {})
  });
  if (!res.ok) throw new Error(`${url} -> ${res.status}`);
  const blob = await res.blob();
  const a = document.createElement("a");
  const cd = res.headers.get("Content-Disposition");
  const filename = (cd && /filename="?([^"]+)"?/.exec(cd)?.[1]) || "download";
  a.href = URL.createObjectURL(blob);
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  a.remove();
  URL.revokeObjectURL(a.href);
}

window.computeTimeline = async function computeTimeline(letter) {
  const scen = window.APP.scenarios[letter];
  if (!scen || !Array.isArray(scen.items) || scen.items.length === 0) {
    window.renderTimeline(letter, []); // blank
    return;
  }
  const codes = (scen.user_order && scen.user_order.length ? scen.user_order : scen.ai_order) || scen.items.map(i=>i.deliverable_code);

  // Authoritative packer produces sequential dates
  const packed = await postJSON("/api/timeline/pack_v2", {
    scenario_letter: letter,
    deliverable_codes: codes,
    project_start: scen.project_start || null,
    complexity: scen.items[0]?.complexity || "Advanced",
    tier: scen.items[0]?.tier || "T2_MediumVolume",
    use_slack: scen.use_slack ?? true,
    slack_after_internal: scen.slack_after_internal ?? 1,
    slack_after_client: scen.slack_after_client ?? 2,
    slack_global_pct: scen.slack_global_pct ?? 0.0
  });

  // Persisted items come back via scenario store; just render what server returns
  const items = packed.items || [];
  window.renderTimeline(letter, items);
};

// This is the function your page was trying to call
window.renderTimeline = function renderTimeline(letter, items) {
  // Minimal renderer: expects a <tbody id="timeline-body-A"> and "...-B"
  const tbody = document.getElementById(`timeline-body-${letter}`);
  if (!tbody) return;
  tbody.innerHTML = "";

  for (const it of (items || [])) {
    const tr = document.createElement("tr");
    const name = window.APP.codeToName[it.deliverable_code] || it.deliverable_code;
    tr.innerHTML = `
      <td>${name}</td>
      <td>${it.start || ""}</td>
      <td>${it.end || ""}</td>
      <td style="text-align:right">${it.duration_days ?? ""}</td>
    `;
    tbody.appendChild(tr);
  }
};
