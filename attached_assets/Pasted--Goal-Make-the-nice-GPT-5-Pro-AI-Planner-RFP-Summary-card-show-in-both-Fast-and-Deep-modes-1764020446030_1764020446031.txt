2. Create a single renderer for the RFP Summary card

Refactor whatever code currently renders the 🧠 card into a single function:

function renderRfpSummaryCard(summary, options = {}) {
  const {
    plannerLabel = "GPT-5 Pro AI Planner",   // Deep default
    complexity = "High"                      // or from summary/meta if you have it
  } = options;

  const container = document.getElementById("rfp-summary-card");
  if (!container || !summary) return;

  const deliverableList = (summary.deliverables || [])
    .map(d => `<li>${d.label}: ${d.short_desc}</li>`)
    .join("");

  container.innerHTML = `
    <div class="card">
      <div class="card-header">
        <span class="planner-badge">🧠 ${plannerLabel}</span>
        <span class="card-meta">Evidence-backed • Reasoning-powered</span>
      </div>
      <h3>📋 RFP Summary</h3>
      <p>${summary.summary_text}</p>
      <h4>Goals:</h4>
      <ul>${deliverableList}</ul>
      <div class="card-footer">
        <span><strong>Channels:</strong> <span id="rfp-channels">—</span></span>
        <span><strong>Markets:</strong> <span id="rfp-markets">—</span></span>
        <span><strong>Complexity:</strong> ${complexity}</span>
      </div>
    </div>
  `;
}


Keep using whatever CSS you already have for that screenshot; just centralize the HTML in one place.
