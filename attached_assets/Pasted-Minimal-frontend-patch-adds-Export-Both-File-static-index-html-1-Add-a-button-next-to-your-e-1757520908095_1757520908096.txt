Minimal frontend patch (adds “Export Both”)

File: static/index.html

1) Add a button next to your existing Export A/B controls (Step 2 area):

<div class="row" style="margin-top:10px;">
  <label>Export format:
    <select id="exportFormat">
      <option value="csv">CSV (.csv)</option>
      <option value="xlsx">Excel (.xlsx)</option>
    </select>
  </label>
  <label><input id="exportTimestamp" type="checkbox" /> Add timestamp</label>
  <button id="btnExportA">Export Scenario A</button>
  <button id="btnExportB">Export Scenario B</button>
  <button id="btnExportBoth">Export Both (A & B)</button> <!-- NEW -->
</div>


2) Use these scripts (drop into your existing <script> block).
They reuse the same logic you already have for single‑scenario export; the “both” button just runs A then B back‑to‑back with the correct filenames.

<script>
  function fallbackName(letter, fmt) {
    const projectName = (document.getElementById('projectName')?.value || 'Proposal').trim();
    const ts = document.getElementById('exportTimestamp')?.checked
      ? (' - ' + new Date().toISOString().replace(/[-:T]/g,'').slice(0,12))
      : '';
    return `${projectName} - Scenario ${letter} - Workfront Export${ts}.${fmt}`;
  }

  function parseFilenameFromContentDisposition(cd) {
    if (!cd) return null;
    const star = cd.match(/filename\*\s*=\s*([^;]+)/i);
    if (star) {
      const v = star[1].trim();
      const parts = v.split("''");
      const raw = parts.length === 2 ? parts[1] : v;
      try { return decodeURIComponent(raw.replace(/^"(.*)"$/, '$1')); } catch { return null; }
    }
    const simple = cd.match(/filename\s*=\s*"([^"]+)"|filename\s*=\s*([^;]+)/i);
    if (simple) return (simple[1] || simple[2]).trim().replace(/^"(.*)"$/, '$1');
    return null;
  }

  async function exportOne(letter, fmt, projectName, addTs) {
    if (!latestScenarios || !latestScenarios[letter]) return;

    const res = await fetch('/api/export', {
      method: 'POST',
      headers: {'Content-Type':'application/json'},
      body: JSON.stringify({
        scenario: latestScenarios[letter],
        project_name: projectName,
        file_format: fmt,
        scenario_label: `Scenario ${letter}`,
        add_timestamp: addTs
      })
    });

    const blob = await res.blob();
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;

    const cd = res.headers.get('content-disposition');
    a.download = parseFilenameFromContentDisposition(cd) || fallbackName(letter, fmt);

    a.click();
    URL.revokeObjectURL(url);
  }

  async function exportScenario(letter) {
    const fmt = document.getElementById('exportFormat').value || 'csv';
    const projectName = document.getElementById('projectName').value || '';
    const addTs = document.getElementById('exportTimestamp').checked;
    await exportOne(letter, fmt, projectName, addTs);
  }

  async function exportBoth() {
    const fmt = document.getElementById('exportFormat').value || 'csv';
    const projectName = document.getElementById('projectName').value || '';
    const addTs = document.getElementById('exportTimestamp').checked;

    // Export A then B as two separate files
    await exportOne('A', fmt, projectName, addTs);
    // tiny gap helps some browsers show two download toasts separately
    await new Promise(r => setTimeout(r, 200));
    await exportOne('B', fmt, projectName, addTs);
  }

  document.getElementById('btnExportA').onclick = () => exportScenario('A');
  document.getElementById('btnExportB').onclick = () => exportScenario('B');
  document.getElementById('btnExportBoth').onclick = exportBoth;
</script>


This assumes you already have the Project name field (auto‑prefilled from the uploaded filename) and the format selector (CSV/XLSX) from the previous patch. If not, I can re‑post those snippets.
