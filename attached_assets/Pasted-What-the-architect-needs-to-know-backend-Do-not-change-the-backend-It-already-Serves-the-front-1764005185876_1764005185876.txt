function closeDrawer() {
  const drawer = document.getElementById('details-drawer');
  const backdrop = document.getElementById('drawer-backdrop');
  drawer.classList.remove('open');
  backdrop.hidden = true;                // <- critical
  drawer.setAttribute('aria-hidden','true');
}

3) Wire into existing flow

At the end of updatePricingSummary(scenario), build the exec-view payload from the same objects already used for the legacy panel, then call renderStep2Exec(execData).

function updatePricingSummary(scenario) {
  // ... existing totals/date computation

  const price = scenario?.totals?.price;         // server computes totals
  const startDate = scenario?.timeline?.start_date || scenario?.timeline?.startDate || scenario?.dates?.start;
  const endDate   = scenario?.timeline?.end_date   || scenario?.timeline?.endDate   || scenario?.dates?.end;
  const durationText = (startDate && endDate)
    ? Math.ceil((new Date(endDate)-new Date(startDate)) / (1000*60*60*24*7)) + ' weeks'
    : (scenario?.timeline?.weeks ? scenario.timeline.weeks + ' weeks' : '—');

  const goal     = scenario?.meta?.goal || window.RFP?.goal || 'Fast analysis';
  const channels = scenario?.meta?.channels || window.RFP?.channels || ['Digital'];
  const markets  = scenario?.meta?.markets  || window.RFP?.markets  || ['US'];

  const scopeText = scenario?.summary?.bulletsText || window.Step2?.summaryText || '';

  const execData = {
    priceFormatted: price?.toLocaleString?.() ? `$${price.toLocaleString()}` : price,
    price, durationText, startDate, endDate, goal, channels, markets, scopeText,
    risks: scenario?.risks || []
  };

  window.step2Data = execData;          // optional: reuse on re-renders
  window.renderStep2Exec?.(execData);
}
