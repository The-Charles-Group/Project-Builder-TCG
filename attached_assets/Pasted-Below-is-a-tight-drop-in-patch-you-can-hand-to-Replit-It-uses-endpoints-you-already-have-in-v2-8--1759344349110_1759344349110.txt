Below is a tight, drop‑in patch you can hand to Replit. It uses endpoints you already have in v2.8:

/api/summarize and /api/summarize_by_file return an RfpSummary object with summary_text, word_count, and deliverables (labels + short_desc/tasks).
(The backend builds this with ai_summarize_rfp_text(…).)

/api/reconcile takes summary_deliverables + the user’s current DB selections and returns Add/Delete/Unchanged suggestions (we’ll render those on the right, with accept/remove controls).

What will change (UX)

Step 2 right column becomes two stacked cards:

AI Suggestions (same as now, but with quick Accept/Remove for each line and a small “Selected ✓” badge when already in Your Selection).

AI Summary (collapsible) – shows the 500‑word prose summary with a word counter and Copy button.

Removing an AI‑suggested deliverable via the ❌ in Your Selection still lets the user add it back from either the Deliverables list (left) or the AI Suggestions list (right). Nothing else changes.

JS patch (works with your current vanilla JS wiring)

Drop this into static/app.js (or wherever Step 1/Step 2 logic lives). It assumes you already have selectedCodes state and the Deliverables (select) + Your Selection renderers in place.

// ---------- Shared state ----------
window.APB = window.APB || {};
// Keep summary across steps and refreshes
const SUMMARY_KEY = 'apb:rfpSummary';

// ---------- Step 1: Analyze with AI -> fetch summary, persist, then show Step 2 ----------
async function analyzeWithAI() {
  const fileInput = document.querySelector('#rfpUpload');
  const textArea  = document.querySelector('#rfpText');

  let summary;
  if (fileInput && fileInput.files && fileInput.files[0]) {
    const fd = new FormData();
    fd.append('file', fileInput.files[0]);
    const res = await fetch('/api/summarize_by_file', { method: 'POST', body: fd });
    summary = await res.json(); // { summary_text, deliverables: [{label, short_desc, tasks}], word_count }
  } else {
    const rfp_text = (textArea?.value || '').trim();
    if (!rfp_text) { alert('Paste RFP text or upload a file first.'); return; }
    const res = await fetch('/api/summarize', {
      method: 'POST', headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ rfp_text })
    });
    summary = await res.json();
  }

  window.APB.summary = summary;
  sessionStorage.setItem(SUMMARY_KEY, JSON.stringify(summary));

  // show Step 2 (your existing showStep logic)
  showStep(2);

  // render summary & suggestions on Step 2
  initAISummaryAndSuggestions();
}

// ---------- Step 2: render AI Summary + Suggestions ----------
function initAISummaryAndSuggestions() {
  // hydrate from memory or session
  const sum = window.APB.summary || JSON.parse(sessionStorage.getItem(SUMMARY_KEY) || 'null');
  if (!sum) {
    // No analysis yet – keep UI available but empty
    renderAISummary(null);
    renderAISuggestions([]);
    return;
  }
  renderAISummary(sum);
  // Build suggestions from AI deliverable labels vs current selection
  const labels = (sum.deliverables || []).map(d => d.label).filter(Boolean);
  const selCodes = Array.from(window.APB.selectedCodes || []);
  reconcileAndRender(labels, selCodes);
}

function renderAISummary(sum) {
  const body  = document.querySelector('#s2-ai-summary-body');
  const count = document.querySelector('#s2-ai-summary-count');
  const copy  = document.querySelector('#s2-ai-summary-copy');
  const tog   = document.querySelector('#s2-ai-summary-toggle');

  if (!body) return;
  if (!sum) {
    body.textContent = 'Run analysis to see summary.';
    count.textContent = '';
    copy.onclick = null;
    tog.onclick  = null;
    return;
  }

  body.textContent = sum.summary_text || '';
  count.textContent = sum.word_count ? `${sum.word_count}/500 words` : '';
  copy.onclick = () => navigator.clipboard.writeText(sum.summary_text || '');
  let hidden = false;
  tog.onclick = () => {
    hidden = !hidden;
    body.style.display = hidden ? 'none' : 'block';
    tog.textContent = hidden ? 'Show' : 'Hide';
  };
}

async function reconcileAndRender(aiLabels, selectedCodes) {
  try {
    const res = await fetch('/api/reconcile', {
      method: 'POST', headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        summary_deliverables: aiLabels,                   // from AI summary
        db_selected_deliverable_codes: selectedCodes,     // left-panel selection
        rfp_text: ''                                      // optional
      })
    });
    const data = await res.json(); // { add, delete, unchanged, db_used_codes, db_used_labels }
    renderAISuggestions(data.add || [], data.delete || [], data.unchanged || []);
  } catch (e) {
    console.error(e);
    renderAISuggestions([]);
  }
}

// Right panel list with Accept/Remove buttons and Selected badge
function renderAISuggestions(add = [], del = [], unchanged = []) {
  const root = document.querySelector('#s2-ai-suggest');
  if (!root) return;

  const mkRow = (sug, type) => {
    const row = document.createElement('div');
    row.className = 'ai-sug-row';
    row.style = 'display:flex;justify-content:space-between;align-items:center;padding:8px 0;border-bottom:1px solid rgba(255,255,255,.06);';

    const left = document.createElement('div');
    left.innerHTML = `<strong>${sug.label}</strong> <span style="opacity:.65;">${type === 'add' ? '• suggested' : type === 'del' ? '• not in AI' : '• unchanged'}</span>`;

    const right = document.createElement('div');

    // already selected?
    const isSel = (window.APB.selectedCodes || new Set()).has(sug.code);
    if (isSel) {
      const b = document.createElement('span');
      b.className = 'badge';
      b.textContent = 'Selected ✓';
      right.appendChild(b);

      const rm = document.createElement('button');
      rm.className = 'btn btn-xs btn-danger'; rm.textContent = 'Remove';
      rm.onclick = () => onRemove(sug.code);
      right.appendChild(rm);
    } else {
      const addBtn = document.createElement('button');
      addBtn.className = 'btn btn-xs btn-primary'; addBtn.textContent = 'Add';
      addBtn.onclick = () => onAdd(sug.code);
      right.appendChild(addBtn);
    }

    row.appendChild(left); row.appendChild(right);
    return row;
  };

  root.innerHTML = '';
  if (!add.length && !del.length && !unchanged.length) {
    root.innerHTML = '<div style="opacity:.7;">Run analysis to see suggestions.</div>';
    return;
  }

  if (add.length) {
    const h = document.createElement('div'); h.className = 'subhead'; h.textContent = 'AI‑Suggested Deliverables';
    root.appendChild(h);
    add.forEach(s => root.appendChild(mkRow(s, 'add')));
  }
  if (del.length) {
    const h = document.createElement('div'); h.className = 'subhead'; h.textContent = 'Consider Removing';
    root.appendChild(h);
    del.forEach(s => root.appendChild(mkRow(s, 'del')));
  }
  if (unchanged.length) {
    const h = document.createElement('div'); h.className = 'subhead'; h.textContent = 'Unchanged';
    root.appendChild(h);
    unchanged.forEach(lbl => {
      root.appendChild(mkRow({ code: '', label: lbl }, 'unchanged'));
    });
  }
}

// Your existing handlers (kept intact)
function onAdd(code) {
  const set = window.APB.selectedCodes || new Set();
  if (!set.has(code)) set.add(code);
  window.APB.selectedCodes = set;
  renderYourSelection();     // your existing function
  // refresh right panel badges without recomputing server-side
  initAISummaryAndSuggestions();
}
function onRemove(code) {
  const set = window.APB.selectedCodes || new Set();
  set.delete(code);
  window.APB.selectedCodes = set;
  renderYourSelection();
  initAISummaryAndSuggestions();
}

// Call once when Step 2 loads
if (document.readyState !== 'loading') initAISummaryAndSuggestions();
else document.addEventListener('DOMContentLoaded', initAISummaryAndSuggestions);


Why this works

We reuse your existing backend for summary and suggestions; no schema changes. The summary endpoints already return a summary_text and a list of AI deliverables; the reconcile endpoint already knows how to propose Add/Delete/Unchanged vs the user’s DB selections.

We persist the summary in sessionStorage so users can reload without losing the right panel.

Accept/Remove actions simply call your existing onAdd/onRemove (left panel remains the source of truth).
