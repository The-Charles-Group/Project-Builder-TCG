Shared apply function

Use one function with a mode flag:

type SmartMode = "ai" | "tfidf";

const applySmartSelect = (mode: SmartMode) => {
  const raw =
    mode === "ai" ? aiSmartThreshold : tfidfSmartThreshold;

  // clamp to 0–100 and coerce to number
  const threshold = Math.max(0, Math.min(100, Number(raw) || 0));

  setDeliverables(prev =>
    prev.map(d => {
      const aiPercent =
        typeof d.match_percent === "number"
          ? d.match_percent
          : (d.confidence ?? 0);

      // backend returns tfidf_similarity as 0–1, convert to %
      const tfidfPercent =
        typeof d.tfidf_similarity === "number"
          ? d.tfidf_similarity * 100
          : (d.tfidf_similarity_pct ?? 0);

      const meets =
        mode === "ai"
          ? aiPercent >= threshold
          : tfidfPercent >= threshold;

      return {
        ...d,
        selected: meets,
        // If you also want to flag WHY it was selected:
        smartSelectionReason: mode,
      };
    })
  );
};


If you’re also selecting components/tasks under each deliverable, extend the map to walk d.components / d.tasks and apply the same meets flag down the tree.
