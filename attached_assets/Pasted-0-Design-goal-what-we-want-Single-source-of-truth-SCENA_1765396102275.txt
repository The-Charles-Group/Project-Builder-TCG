1. Frontend – Step 3 Initial Load
Task F1 – Add initPricingStep() on Step 3

Goal: When Step 3 loads, auto‑load the working scenario and build both panels; don’t require clicking “Build Scenario”.

In the Step 3 JS file (where buildScenariosAB, updatePricing, etc. live) add:

async function initPricingStep() {
  const sessionId = window.SESSION_ID; // however you already store this

  // 1) Ask backend for the working scenario (no reset)
  const resp = await fetch('/api/scenarios', {
    method: 'POST',
    headers: {'Content-Type': 'application/json'},
    body: JSON.stringify({ session_id: sessionId, reset: false })
  });
  const data = await resp.json();
  if (!data.ok || !data.scenarios || !data.scenarios.A) {
    console.warn('initPricingStep: no scenario available');
    return;
  }

  // 2) Cache scenario A in window
  window.SCENARIO_A = data.scenarios.A;

  // 3) Build pricing panels using backend‑authoritative breakdown
  await rebuildPricingFromBackend(window.SCENARIO_A);
}


Then in your Step 3 HTML page JS bootstrap (where you currently set up listeners), call:

document.addEventListener('DOMContentLoaded', () => {
  initPricingStep().catch(console.error);
});

function onPricingCellChange(wbsId, field, value) {

  updatePricingGridState(wbsId, field, value); // existing or new helper

  // 2) Update the in-memory scenario
  const scen = window.SCENARIO_A;
  if (scen && scen.items) {
    const item = scen.items.find(i => i.WBS_ID === wbsId);
    if (item) {
      if (field === 'hours') item.Planned_Hours = Number(value) || 0;

      if (field === 'type') item.Cadence = value; // or your cadence field

  }

  // 3) Mark dirty
  markPricingDirty();
}

Task F5 – Implement syncScenarioToBackend and reuse everywhere

Shape it to match ScenarioSyncPayload on the backend (you can see that in main.py):

async function syncScenarioToBackend(scenario) {
  const sessionId = window.SESSION_ID;

  const payload = {
    session_id: sessionId,
    client_version: (window.SCENARIO_CLIENT_VERSION || 1),
    last_server_version: (window.SCENARIO_SERVER_VERSION || 0),
    scenario,
    selections: window.SCENARIO_SELECTIONS || {},
    timestamp: new Date().toISOString(),
    checksum: null
  };

  const resp = await fetch('/api/scenario/sync', {
    method: 'POST',
    headers: {'Content-Type': 'application/json'},
    body: JSON.stringify(payload)
  });
  const data = await resp.json();

  window.SCENARIO_SERVER_VERSION = data.serverVersion;
  window.SCENARIO_CLIENT_VERSION = (window.SCENARIO_CLIENT_VERSION || 1) + 1;

  // make it easy for callers
  return {
    scenario: (data.serverScenario || payload.scenario),
    serverState: data
  };
}

Task B4 – Add a tiny debug endpoint for sanity checks (optional but recommended)

This will help you or the architect verify Brand Identity Development hours at each stage.

from fastapi import HTTPException

@app.get("/api/debug/deliverable_hours/{session_id}/{deliverable_code}")
def debug_deliverable_hours(session_id: str, deliverable_code: str):
    scenario = get_working_scenario(session_id)
    if not scenario or not scenario.get("items"):
        raise HTTPException(404, "Scenario not found")

    items = scenario["items"]
    rows = [i for i in items if i.get("Deliverable_Code") == deliverable_code]
    hours = sum(i.get("Planned_Hours", 0) for i in rows)

    return {
        "deliverable_code": deliverable_code,
        "rows": len(rows),
        "scenario_hours_sum": hours,
        "rows_detail": [
            {
                "WBS_ID": r.get("WBS_ID"),
                "Component": r.get("Component"),
                "Planned_Hours": r.get("Planned_Hours")
            } for r in rows
        ]
    }


If you hit this endpoint and see 307, then export and it’s 300 somewhere, you know the issue is further down (e.g. in build_wbs_dataframe_from_scenario or MS Project XML conversion). That gives the architect a precise place to look instead of guessing.
