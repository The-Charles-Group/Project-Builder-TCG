def save_synced_scenario_to_store(session_id: str):
    """Copy scenario from SCENARIO_SYNC_STATE into SCENARIO_STORE and recompute totals."""
    from copy import deepcopy
    sync_state = SCENARIO_SYNC_STATE.get(session_id)
    if not sync_state or not sync_state.get("scenario"):

    scen = deepcopy(sync_state["scenario"])
    scen = _recompute_totals(scen)

def apply_cadence_to_item(item: dict, blended_rate: float | None = None, rate_band: str | None = None):
    cadence = (item.get("billing_cadence") or "one_time").lower()
    units = int(item.get("cadence_units") or 1)
    cadence_price = float(item.get("cadence_price") or 0.0)

    # How many months does one cadence period represent?
    months_per_cadence = {
        "one time": 0,
        "one_time": 0,
        "monthly": 1,
        "quarterly": 3,
        "semi-annual": 6,
        "semiannual": 6,
        "annual": 12,
    }.get(cadence, 0)

    total_months = units * months_per_cadence

    total_price = cadence_price * max(units, 1)
    item["total_price"] = round(total_price, 2)

    # Decide if this is a retainer
    is_retainer = cadence != "one_time" and total_months > 0
    item["is_retainer"] = is_retainer
    item["retainer"] = {"months": total_months}
    item["retainer_months"] = total_months

    # Use existing rate to convert price -> hours
    rate = float(item.get("Rate_USD") or blended_rate or 0)  # fallback to blended rate if needed
    if rate > 0:
        total_hours = total_price / rate
    else:
        total_hours = float(item.get("Planned_Hours") or 0)

    item["Planned_Hours"] = round(total_hours, 2)

    if is_retainer and total_months > 0:
        item["monthly_hours"] = round(total_hours / total_months, 2)
        item["monthly_price"] = round(total_price / total_months, 2)
    else:
        item["monthly_hours"] = 0
        item["monthly_price"] = 0

from datetime import date, timedelta

def month_bounds(start_date: date, offset: int):
    # offset = 0 for first month, 1 for month+1, etc.
    year = start_date.year + (start_date.month - 1 + offset) // 12
    month = (start_date.month - 1 + offset) % 12 + 1
    first = date(year, month, 1)
    if month == 12:
        last = date(year + 1, 1, 1) - timedelta(days=1)
    else:
        last = date(year, month + 1, 1) - timedelta(days=1)
    return first, last

5.2. Respect existing PM changes while adding hours

Brad also said:

If a PM already tweaked the Gantt, adding more hours in Step 3 should add extra time onto the end of that deliverable instead of nuking their changes.

How to approximate that with your current pipeline:

When you save the Gantt (via /api/timeline/save or similar), keep per-deliverable IDs and their start/end dates in SCENARIO_STORE[session_id]["scenario"]["items"] or a separate timeline key.

When Step 3 changes price/cadence:

Compute new total hours for each deliverable (from apply_cadence_to_item).

Compare to old total (that was stored pre-save).

If delta_hours > 0:

For one-time deliverables:

Increase Planned_Hours on the last set of child tasks or append an “Extra hours” task whose Start_Offset_Days = last child’s end date, so it stacks at the end.

For monthly retainers:

If months increased: append more monthly child tasks using the per-month logic above.

If monthly_hours increased: bump Planned_Hours on each monthly task proportionally or add a small “extra hours” task at the end of each month.

Write the updated WBS back to SCENARIO_STORE so export uses the merged, PM-respecting schedule.

That’s the “ideal” behaviour Brad described; you can start simpler (rebuild the retainer’s monthly tasks from scratch but preserve deliverable start date) and then refine if PMs complain.
