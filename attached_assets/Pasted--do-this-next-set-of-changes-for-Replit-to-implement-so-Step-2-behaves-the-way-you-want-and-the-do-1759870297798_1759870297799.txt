  setState(clone(state));

function addBusinessDays(start: Date, days: number, holidays: Set<string>) {
  const d = new Date(start);
  let added = 0;
  while (added < days) {
    d.setDate(d.getDate() + 1);
    const dow = d.getDay(); // 0=Sun ... 6=Sat
    const isWeekend = dow === 0 || dow === 6;
    const isHoliday = holidays.has(d.toISOString().slice(0,10));
    if (!isWeekend && !isHoliday) added++;
  }
  return d;
}
function businessDaysInclusive(start: Date, end: Date, holidays: Set<string>) {
  let d = new Date(start), count = 0;
  while (d <= end) {
    const dow = d.getDay();
    if (dow !== 0 && dow !== 6 && !holidays.has(d.toISOString().slice(0,10))) count++;
    d.setDate(d.getDate()+1);
  }
  return count;
}
