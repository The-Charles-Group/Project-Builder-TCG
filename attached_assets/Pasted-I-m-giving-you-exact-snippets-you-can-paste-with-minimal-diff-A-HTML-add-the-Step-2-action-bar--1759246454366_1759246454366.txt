C) Build payload strictly from Step 2 state

This guarantees Step 4 shows only what the user selected (no silent AI/template add‑ins).

function buildPayloadForApi() {
  // selected deliverables from "Your Selection"
  const selected_deliverable_codes = Array.from(S2.selectedCodes);

  // map of chosen components (omit a code -> backend assumes "all components")
  const selected_components_map = {};
  for (const code of selected_deliverable_codes) {
    const pick = S2.componentsByDeliv[code];
    if (!pick || pick === 'ALL') continue;
    selected_components_map[code] = pick instanceof Set ? Array.from(pick) : pick; // supports list or {name:hours}
  }

  return {
    selected_deliverable_codes,
    selected_components_map,
    scenario_a: S2.scenarioA,            // whatever you store (e.g., {mode:"template", scenario_key:"MED_LOW"})
    scenario_b: S2.scenarioB,
    pricing_mode: S2.pricingMode,
    blended_rate: S2.blendedRate,
    rate_band: S2.rateBand,
    use_slack: S2.useSlack,
    slack_after_internal: S2.slackAfterInternal,
    slack_after_client: S2.slackAfterClient,
    slack_global_pct: S2.slackGlobalPct,
    project_start: S2.projectStart,
    retainers: S2.retainers
  };
}

Proceed → Pricing / Build
async function proceedToPricing() {
  const payload = buildPayloadForApi();

  if (!payload.selected_deliverable_codes.length) {
    alert('Please select at least one deliverable.');
    return;
  }

  // Sanity check in the console (helps diagnose if extra codes ever creep in)
  console.log('[BUILD] selected_deliverable_codes:', payload.selected_deliverable_codes);

D) “Get AI Suggestions” (from Step 2)

If the user has already run Analyze with AI in Step 1, you likely store an App.aiSummary (or raw RFP text/file). Use that to ask the server for adds/deletes against what’s currently selected:

async function getAISuggestions() {
  if (!App || !App.aiSummary) {
    alert('Enter RFP text or upload a file, then click “Analyze with AI” in Step 1 first.');
    return;
  }

  const payload = {
    selected_deliverable_codes: Array.from(S2.selectedCodes),
    ai_summary: App.aiSummary
  };

  const res = await fetch('/api/reconcile', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(payload)
  });
  if (!res.ok) {
    alert('Could not get AI suggestions. Please try again.');
    return;
  }
  const data = await res.json();

  // Render into the right panel (ADD / DELETE buckets with checkboxes & apply buttons)
  renderAISuggestions(data);
}


Notes

If your backend takes rfp_text or a file handle instead of ai_summary, adjust the payload accordingly.

In the suggestions panel, keep Apply buttons that explicitly add/remove items by editing S2.selectedCodes (don’t silently merge them into pricing).

How to use: 1) Select deliverables on the left → 2) In “Your Selection”, click Components… for each deliverable and adjust (all are selected by default) → 3) Click Apply selection → 4) Proceed to Pricing (or Get AI Suggestions first).

If you want the “Apply selection” button to confirm how many deliverables/components will be built, you can make the label dynamic:

function updateApplyLabel() {
  const codes = Array.from(S2.selectedCodes);
  let comps = 0;
  for (const c of codes) {
    const pick = S2.componentsByDeliv[c];
    comps += (!pick || pick === 'ALL') ? -1 : pick.size; // -1 means "all" (don’t count individually)
  }
  const btn = document.getElementById('s2-apply');
  if (btn) btn.textContent = 'Apply selection';
}


(You can replace the label with something like: “Apply selection (3 deliverables)” if you prefer.)
