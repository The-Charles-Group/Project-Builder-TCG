<script>
(function(){
  const state = {
    options: { deliverables: [], scenario_templates: [], bundles: [], rate_bands: [] },
    selected: new Set(),
  };

  const els = {
    list: document.getElementById('deliverableList'),
    search: document.getElementById('deliverableSearch'),
    selectAll: document.getElementById('selectAllBtn'),
    clearAll: document.getElementById('clearAllBtn'),
    suggestBtn: document.getElementById('suggestBtn'),
    rfp: document.getElementById('rfpText'),
    pricingMode: document.getElementById('pricingMode'),
    blendedRate: document.getElementById('blendedRate'),
    rateBand: document.getElementById('rateBand'),
    useSlack: document.getElementById('useSlack'),
    slackI: document.getElementById('slackI'),
    slackC: document.getElementById('slackC'),
    slackPct: document.getElementById('slackPct'),
    projectStart: document.getElementById('projectStart'),
    buildBtn: document.getElementById('buildBtn'),
    step1: document.getElementById('step1'),
    step2: document.getElementById('step2'),
    scenariosWrap: document.getElementById('scenariosWrap'),
    scModeRadios: document.querySelectorAll('input[name="scMode"]'),
    templateBlock: document.getElementById('templateBlock'),
    bundleBlock: document.getElementById('bundleBlock'),
    scenarioASelect: document.getElementById('scenarioASelect'),
    scenarioBSelect: document.getElementById('scenarioBSelect'),
    bundleASelect: document.getElementById('bundleASelect'),
    bundleBSelect: document.getElementById('bundleBSelect')
  };

  function applyScModeVisibility() {
    const mode = document.querySelector('input[name="scMode"]:checked').value;
    els.templateBlock.style.display = mode === 'template' ? 'block' : 'none';
    els.bundleBlock.style.display = mode === 'bundle' ? 'block' : 'none';
  }
  [...els.scModeRadios].forEach(r => r.addEventListener('change', applyScModeVisibility));

  function renderDeliverables(filter='') {
    const q = (filter || '').toLowerCase().trim();
    const rows = state.options.deliverables.filter(d =>
      !q || d.Deliverable.toLowerCase().includes(q) || d.Category.toLowerCase().includes(q) || d.Deliverable_Code.toLowerCase().includes(q)
    );
    els.list.innerHTML = rows.map(d => `
      <label style="display:flex; align-items:center; gap:8px; padding:6px 4px;">
        <input type="checkbox" data-code="${d.Deliverable_Code}" ${state.selected.has(d.Deliverable_Code) ? 'checked' : ''}/>
        <span>${d.Deliverable}</span>
        <span style="margin-left:auto; font-size:12px; opacity:0.8;">${d.Category}</span>
      </label>
    `).join('') || '<div style="opacity:.7;">No deliverables</div>';
    // wire events
    els.list.querySelectorAll('input[type="checkbox"]').forEach(cb => {
      cb.addEventListener('change', e => {
        const code = e.target.getAttribute('data-code');
        if (e.target.checked) state.selected.add(code); else state.selected.delete(code);
      });
    });
  }

  async function loadOptions() {
    const res = await fetch('/api/options');           // v2.8 returns deliverables, templates, bundles, bands
    const data = await res.json();
    state.options = data;
    // deliverables
    renderDeliverables('');
    // templates
    els.scenarioASelect.innerHTML = data.scenario_templates.map(t => `<option value="${t.Scenario_Key}">${t.Scenario_Key}</option>`).join('');
    els.scenarioBSelect.innerHTML = els.scenarioASelect.innerHTML;
    // bundles
    els.bundleASelect.innerHTML = data.bundles.map(b => `<option value="${b}">${b}</option>`).join('');
    els.bundleBSelect.innerHTML = els.bundleASelect.innerHTML;
    // rate bands
    els.rateBand.innerHTML = data.rate_bands.map(b => `<option value="${b}">${b}</option>`).join('');
    applyScModeVisibility();
  }

  async function suggestFromRfp() {
    const text = els.rfp.value || '';
    if (!text.trim()) return;
    const res = await fetch('/api/suggest_by_text', {
      method:'POST', headers:{'Content-Type':'application/json'},
      body: JSON.stringify({ rfp_text: text })
    });
    const data = await res.json(); // { suggested: [{deliverable_code,...}] }
    for (const s of (data.suggested || [])) state.selected.add(s.deliverable_code);
    renderDeliverables(els.search.value);
  }

  function selectedCodes() { return Array.from(state.selected); }

  function scenarioSpecAB() {
    const mode = document.querySelector('input[name="scMode"]:checked').value;
    if (mode === 'bundle') {
      return {
        scenario_a: { mode:'bundle', bundle: els.bundleASelect.value },
        scenario_b: { mode:'bundle', bundle: els.bundleBSelect.value }
      };
    }
    return {
      scenario_a: { mode:'template', scenario_key: els.scenarioASelect.value },
      scenario_b: { mode:'template', scenario_key: els.scenarioBSelect.value }
    };
  }

  async function buildScenarios() {
    const codes = selectedCodes();
    if (codes.length === 0) { alert('Pick at least one deliverable.'); return; }

    const specs = scenarioSpecAB();
    const payload = {
      selected_deliverable_codes: codes,
      scenario_a: specs.scenario_a,
      scenario_b: specs.scenario_b,
      pricing_mode: els.pricingMode.value,
      blended_rate: els.pricingMode.value === 'Flat_Blended' ? Number(els.blendedRate.value) : undefined,
      rate_band: els.rateBand.value,
      use_slack: els.useSlack.checked,
      slack_after_internal: Number(els.slackI.value || 0),
      slack_after_client: Number(els.slackC.value || 0),
      slack_global_pct: Number(els.slackPct.value || 0),
      project_start: els.projectStart.value || null
    };

    const res = await fetch('/api/build', {
      method:'POST', headers:{'Content-Type':'application/json'},
      body: JSON.stringify(payload)
    });
    const data = await res.json(); // { A: {...}, B: {...} }
    renderStep2(data);
  }

  function renderStep2(scens) {
    els.step1.style.display = 'none';
    els.step2.style.display = 'block';
    els.scenariosWrap.innerHTML = ['A','B'].map(letter => {
      const s = scens[letter] || {items:[], totals:{hours:0, price:0}};
      const rows = (s.items || []).map(it => `
        <tr>
          <td>${it.deliverable}</td>
          <td>${it.category || ''}</td>
          <td style="text-align:right;">${Math.round(it.total_hours)}</td>
          <td style="text-align:right;">$${Number(it.price).toLocaleString()}</td>
        </tr>
      `).join('');
      return `
        <div style="margin-bottom:24px;">
          <h3>Scenario ${letter}</h3>
          <table style="width:100%; border-collapse:collapse;">
            <thead>
              <tr><th style="text-align:left;">Deliverable</th><th>Category</th><th style="text-align:right;">Hours</th><th style="text-align:right;">Price</th></tr>
            </thead>
            <tbody>${rows || '<tr><td colspan="4" style="opacity:.7;">No items.</td></tr>'}</tbody>
            <tfoot>
              <tr><td colspan="2" style="text-align:right; font-weight:600;">Totals</td>
                  <td style="text-align:right; font-weight:600;">${(s.totals && s.totals.hours) || 0}</td>
                  <td style="text-align:right; font-weight:600;">$${(s.totals && s.totals.price || 0).toLocaleString()}</td></tr>
            </tfoot>
          </table>
        </div>`;
    }).join('');
  }

  // events
  els.search.addEventListener('input', () => renderDeliverables(els.search.value));
  els.selectAll.addEventListener('click', () => { state.options.deliverables.forEach(d => state.selected.add(d.Deliverable_Code)); renderDeliverables(els.search.value); });
  els.clearAll.addEventListener('click', () => { state.selected.clear(); renderDeliverables(els.search.value); });
  els.suggestBtn.addEventListener('click', suggestFromRfp);
  els.buildBtn.addEventListener('click', buildScenarios);

  loadOptions(); // on load
})();
</script>


Why this works (and matches v2.8):

The picker is filled by GET /api/options → deliverables + scenario_templates + bundles (the exact keys you return in 2.8). 

v2.8 prod

“Suggest deliverables” posts the RFP text to POST /api/suggest_by_text and auto‑checks any deliverable_codes it returns. (Retainer hints are present but optional.) 

v2.8 prod

Clicking Build sends only the checked selected_deliverable_codes to POST /api/build; the server iterates those codes to assemble Scenarios A/B and totals.

Payload shape the UI sends (for reference)
{
  "selected_deliverable_codes": ["CONTENT_PLAN","MEETINGS", "PRODUCTION"],
  "scenario_a": {"mode":"template","scenario_key":"MED_LOW"},
  "scenario_b": {"mode":"template","scenario_key":"MED_HIGH"},
  "pricing_mode": "Flat_Blended",
  "blended_rate": 195,
  "rate_band": "Standard_US",
  "use_slack": true,
  "slack_after_internal": 1,
  "slack_after_client": 2,
  "slack_global_pct": 0.05,
  "project_start": "2025-10-01"
}


This is exactly what v2.8 expects—selected_deliverable_codes controls which items are built; template vs bundle is selected by scenario_{a,b}.mode.
