async function openComponentsModal(code) {
  const complexity = state.complexity;   // e.g. "Advanced"
  const tier       = state.tier;         // e.g. "T2_MediumVolume"

  const res = await fetch(`/api/components_for?deliverable_code=${encodeURIComponent(code)}&complexity=${encodeURIComponent(complexity)}&tier=${encodeURIComponent(tier)}`);
  const data = await res.json(); // { items: [{name, hours}, ...] }
  // Default to selected unless user previously customized:
  const selectedSet = new Set((state.componentsByCode[code]?.map(x=>x.name)) || data.items.map(x=>x.name));
  renderComponentsModal(code, data.items, selectedSet);
  showModal('#componentsModal');
}

Fix 2 — Build the payload for /api/build correctly

Apply selection must send both the selected codes and the component selections. Use the v2.8 contract (works with “all” or custom hours):

async function buildScenariosFromStep2() {
  const selectedCodes = Array.from(state.selectedCodes);   // e.g. ["DEL-0029", ...]
  if (!selectedCodes.length) return alert('Select at least one deliverable');

  // Compose selected_components_map with safe defaults:
  const compMap = {};
  for (const code of selectedCodes) {
    if (!state.componentsByCode[code] || !state.componentsByCode[code].length) {
      // User never opened the modal → include all components
      compMap[code] = "__ALL__";
    } else {
      // Old format list is fine; backend accepts it
      compMap[code] = state.componentsByCode[code].map(c => c.name);
    }
  }

  const payload = {
    selected_deliverable_codes: selectedCodes,
    selected_components_map:    compMap,                          // <-- KEY
    scenario_a: { mode: "template", complexity: state.complexity, tier: state.tier },
    scenario_b: { mode: "template", complexity: state.complexity, tier: state.tier },
    pricing_mode: state.pricingMode,          // "Flat_Blended" | "Per_Resource"
    blended_rate: state.blendedRate || null,
    rate_band: state.rateBand || "Standard_US",
    use_slack: state.useSlack,
    slack_after_internal: state.slackInternalDays || 1,
    slack_after_client: state.slackClientDays || 2,
    slack_global_pct: state.slackGlobalPct || 0.05,
    project_start: state.projectStart || null,
    retainers: (state.retainers || []).map(r => ({ deliverable_code: r.code, months: r.months }))
  };

  const res = await fetch('/api/build', { method: 'POST', headers: {'Content-Type':'application/json'}, body: JSON.stringify(payload) });
  const data = await res.json();
  // Store for steps 3–5 (server also stores in-session)
  state.buildResponse = data;
  goToStep(3);
}


The backend expects selected_components_map and also honors the "__ALL__" sentinel you send when the user didn’t open the modal, so your hours/prices and XML stay fully expanded.

function populateSelectOnce(sel, values) {
  const seen = new Set();
  sel.innerHTML = '';
  values.forEach(v => { if (!seen.has(v)) { sel.append(new Option(v, v)); seen.add(v); } });
}
