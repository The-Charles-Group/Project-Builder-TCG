2) Vanilla JS module for Step 2 (state + rendering)

Paste the following once in static/app.js (or the JS file that drives Step 2).
If you already have Step 2 logic there, add this as a module and wire the two small hooks noted at the bottom (“Hook A/B”).

// ===== Step 2 State & Rendering (Vanilla JS) =====

const Step2 = (() => {
  // ---------- State ----------
  const LS_KEY = "apb.step2.selection.v1";

  const state = {
    selectedDeliverables: new Set(),             // Set<deliverableCode>
    selectedComponents: {},                      // { [deliverableCode]: Set<component> }
    selectedL3: {},                               // { [deliverableCode]: { [component]: Set<taskLabel> } }
    caches: {                                    // response caches
      components: {},                            // { [deliverableCode]: string[] }
      l3: {}                                     // { [deliverableCode]: { [component]: string[] } }
    },
    previewDeliverable: null,                    // deliverableCode being previewed (not selected)
    includeAnchors: false                        // XML toggle
  };

  // ---------- Persistence ----------
  function persist() {
    const obj = {
      selectedDeliverables: [...state.selectedDeliverables],
      selectedComponents: Object.fromEntries(
        Object.entries(state.selectedComponents).map(([d, set]) => [d, [...set]])
      ),
      selectedL3: Object.fromEntries(
        Object.entries(state.selectedL3).map(([d, compMap]) => [
          d,
          Object.fromEntries(Object.entries(compMap).map(([c, set]) => [c, [...set]]))
        ])
      ),
      includeAnchors: state.includeAnchors
    };
    localStorage.setItem(LS_KEY, JSON.stringify(obj));
  }

  function rehydrate() {
    try {
      const raw = localStorage.getItem(LS_KEY);
      if (!raw) return;
      const obj = JSON.parse(raw);
      state.selectedDeliverables = new Set(obj.selectedDeliverables || []);
      state.selectedComponents = {};
      for (const [d, arr] of Object.entries(obj.selectedComponents || {})) {
        state.selectedComponents[d] = new Set(arr || []);
      }
      state.selectedL3 = {};
      for (const [d, cobj] of Object.entries(obj.selectedL3 || {})) {
        state.selectedL3[d] = {};
        for (const [c, arr] of Object.entries(cobj || {})) {
          state.selectedL3[d][c] = new Set(arr || []);
        }
      }
      state.includeAnchors = !!obj.includeAnchors;
      const chk = document.getElementById("toggle-anchors");
      if (chk) chk.checked = state.includeAnchors;
    } catch {}
  }

  // ---------- API wrappers ----------
  async function fetchComponents(deliverableCode) {
    if (state.caches.components[deliverableCode]) {
      return state.caches.components[deliverableCode];
    }
    const payload = { deliverables: [deliverableCode] };
    const res = await fetch("/api/step2/components", {
      method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify(payload)
    });
    const data = await res.json();
    // API returns: { [deliverableCode]: ["component", ...] }
    const comps = (data && data[deliverableCode]) || [];
    state.caches.components[deliverableCode] = comps;
    return comps;
  }

  async function fetchL3(deliverableCode, components) {
    // Only fetch what we don't have cached
    const missing = (components || []).filter(c => !(state.caches.l3[deliverableCode] && state.caches.l3[deliverableCode][c]));
    if (missing.length) {
      const res = await fetch("/api/step2/l3/bulk", {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({ deliverable: deliverableCode, components: missing })
      });
      const data = await res.json(); // { [component]: [taskLabel, ...] }
      state.caches.l3[deliverableCode] = state.caches.l3[deliverableCode] || {};
      for (const [c, l3] of Object.entries(data || {})) {
        state.caches.l3[deliverableCode][c] = l3 || [];
      }
    }
    // return a normalized map for all requested components
    const out = {};
    for (const c of components) {
      out[c] = (state.caches.l3[deliverableCode] && state.caches.l3[deliverableCode][c]) || [];
    }
    return out;
  }

  // ---------- Selection helpers ----------
  function ensureMaps(d) {
    state.selectedComponents[d] = state.selectedComponents[d] || new Set();
    state.selectedL3[d] = state.selectedL3[d] || {};
  }

  async function toggleDeliverable(d, checked) {
    ensureMaps(d);
    if (checked) {
      state.selectedDeliverables.add(d);
      // default-select no components until user checks them
    } else {
      state.selectedDeliverables.delete(d);
      delete state.selectedComponents[d];
      delete state.selectedL3[d];
    }
    persist(); recount(); renderSummary();
  }

  async function toggleComponent(d, c, checked) {
    ensureMaps(d);
    if (checked) {
      state.selectedComponents[d].add(c);
      // load L3 and default select all for that component if first time
      const l3Map = await fetchL3(d, [c]);
      const all = l3Map[c] || [];
      state.selectedL3[d][c] = state.selectedL3[d][c] || new Set(all);
    } else {
      state.selectedComponents[d].delete(c);
      delete state.selectedL3[d][c];
      if (state.selectedComponents[d].size === 0) {
        // keep deliverable but with no components selected (allowed)
      }
    }
    persist(); recount(); renderSummary();
  }

  function toggleL3(d, c, taskLabel, checked) {
    ensureMaps(d);
    state.selectedL3[d][c] = state.selectedL3[d][c] || new Set();
    if (checked) state.selectedL3[d][c].add(taskLabel);
    else state.selectedL3[d][c].delete(taskLabel);
    persist(); recount(); // summary already on-screen
  }

  // ---------- Preview ----------
  async function previewDeliverable(d) {
    state.previewDeliverable = d;
    const comps = await fetchComponents(d);
    // Render the components pane reflecting current selection
    renderComponentsPane(d, comps);
  }

  // ---------- Counts ----------
  function recount() {
    const dCount = state.selectedDeliverables.size;
    let cCount = 0, l3Count = 0;
    for (const d of Object.keys(state.selectedComponents)) {
      cCount += (state.selectedComponents[d] ? state.selectedComponents[d].size : 0);
    }
    for (const [d, compMap] of Object.entries(state.selectedL3)) {
      for (const set of Object.values(compMap)) l3Count += (set ? set.size : 0);
    }
    const dEl = document.getElementById("count-deliverables");
    const cEl = document.getElementById("count-components");
    const l3El = document.getElementById("count-l3");
    if (dEl) dEl.textContent = dCount;
    if (cEl) cEl.textContent = cCount;
    if (l3El) l3El.textContent = l3Count;
  }

  // ---------- Rendering: Components pane ----------
  function renderComponentsPane(d, components) {
    // Find your actual pane container:
    const pane = document.querySelector("#components-pane") || document.querySelector('[data-pane="components"]');
    if (!pane) return;
    const sel = state.selectedComponents[d] || new Set();
    pane.innerHTML = `
      <div class="panel-head row-between">
        <h3>Components</h3>
        <div class="muted">${d || ""}</div>
      </div>
      <div class="list">
        ${components.map(c => {
          const id = `cmp_${d}_${c}`.replace(/\W+/g,'_');
          const checked = sel.has(c) ? 'checked' : '';
          return `
            <label class="row">
              <input data-kind="component" data-d="${d}" data-c="${c}" type="checkbox" ${checked}>
              <span>${c}</span>
            </label>`;
        }).join("")}
      </div>
    `;
  }

  // ---------- Rendering: Summary (nested checkboxes with scroll) ----------
  function renderSummary() {
    const root = document.getElementById("summary-list");
    if (!root) return;
    const blocks = [];
    for (const d of state.selectedDeliverables) {
      const comps = [...(state.selectedComponents[d] || new Set())].sort();
      if (!comps.length) continue;
      blocks.push(`
        <div class="summary-group" data-d="${d}">
          <h4>${d}</h4>
          ${comps.map(c => {
            const allL3 = (state.caches.l3[d] && state.caches.l3[d][c]) || [];
            const sel = (state.selectedL3[d] && state.selectedL3[d][c]) ? [...state.selectedL3[d][c]] : [];
            return `
              <div class="summary-comp" data-d="${d}" data-c="${c}">
                <div class="bar">
                  <h5>${c}</h5>
                  <div class="summary-actions">
                    <button type="button" data-act="all">All</button>
                    <button type="button" data-act="none">None</button>
                  </div>
                </div>
                <div class="summary-l3">
                  ${allL3.map(t => {
                    const id = `l3_${d}_${c}_${t}`.replace(/\W+/g,'_');
                    const checked = sel.includes(t) ? 'checked' : '';
                    return `
                      <label title="${t}">
                        <input data-kind="l3" data-d="${d}" data-c="${c}" data-t="${encodeURIComponent(t)}" type="checkbox" ${checked}>
                        <span>${t}</span>
                      </label>`;
                  }).join("")}
                </div>
              </div>`;
          }).join("")}
        </div>
      `);
    }
    root.innerHTML = blocks.join("");
  }

  // ---------- Events (delegated) ----------
  function bindEvents() {
    // Clicking the deliverable row should PREVIEW only.
    const deliverablesPane = document.querySelector("#deliverables-pane") || document.querySelector('[data-pane="deliverables"]');
    if (deliverablesPane) {
      deliverablesPane.addEventListener("click", async (e) => {
        const row = e.target.closest("[data-deliverable]");
        if (!row) return;
        // If the click was on the checkbox, allow the default handler (Hook A) to run.
        if (e.target.closest('input[type="checkbox"]')) return;
        const d = row.getAttribute("data-deliverable");
        if (d) previewDeliverable(d); // PREVIEW ONLY, no mutation
      });

      // Hook A: deliverable checkbox changes selection (not preview)
      deliverablesPane.addEventListener("change", (e) => {
        const cb = e.target.closest('input[type="checkbox"][data-deliverable]');
        if (!cb) return;
        const d = cb.getAttribute("data-deliverable");
        toggleDeliverable(d, cb.checked);
        if (cb.checked) previewDeliverable(d); // nice UX
      });
    }

    // Components pane -> toggle component selection
    const componentsPane = document.querySelector("#components-pane") || document.querySelector('[data-pane="components"]');
    if (componentsPane) {
      componentsPane.addEventListener("change", async (e) => {
        const cb = e.target.closest('input[type="checkbox"][data-kind="component"]');
        if (!cb) return;
        const d = cb.getAttribute("data-d");
        const c = cb.getAttribute("data-c");
        await toggleComponent(d, c, cb.checked);
        // When component is selected, make sure we have its L3 for the summary UI
        if (cb.checked) { await fetchL3(d, [c]); renderSummary(); }
      });
    }

    // Summary pane: All/None + L3 checkbox
    const summary = document.getElementById("summary-list");
    if (summary) {
      summary.addEventListener("click", (e) => {
        const btn = e.target.closest('button[data-act]');
        if (!btn) return;
        const compBlock = btn.closest(".summary-comp");
        if (!compBlock) return;
        const d = compBlock.getAttribute("data-d");
        const c = compBlock.getAttribute("data-c");
        ensureMaps(d);
        state.selectedL3[d][c] = state.selectedL3[d][c] || new Set();
        const all = (state.caches.l3[d] && state.caches.l3[d][c]) || [];
        if (btn.dataset.act === "all") {
          state.selectedL3[d][c] = new Set(all);
        } else {
          state.selectedL3[d][c].clear();
        }
        persist(); recount(); renderSummary();
      });

      summary.addEventListener("change", (e) => {
        const cb = e.target.closest('input[type="checkbox"][data-kind="l3"]');
        if (!cb) return;
        const d = cb.getAttribute("data-d");
        const c = cb.getAttribute("data-c");
        const t = decodeURIComponent(cb.getAttribute("data-t") || "");
        toggleL3(d, c, t, cb.checked);
      });
    }

    // Anchors toggle
    const anchorsChk = document.getElementById("toggle-anchors");
    if (anchorsChk) {
      anchorsChk.addEventListener("change", () => {
        state.includeAnchors = anchorsChk.checked;
        persist();
      });
    }
  }

  // ---------- Public init ----------
  function init() {
    rehydrate();
    recount();
    renderSummary();
    bindEvents();
  }

  // expose API if other code needs it
  return { init, previewDeliverable };
})();

// Call once after DOM is ready:
document.addEventListener("DOMContentLoaded", () => Step2.init());

3) Backend: make Start/End anchors optional

Your converter already supports add_deliverable_milestones (default False).
When True, it injects Start — <Deliverable> and End — <Deliverable> rows before rebuilding WBS/dependencies.

Patch: thread a boolean from the export payload to the converter.

Add a field to your export request model(s) (where you define ExportXMLPayload / similar):

# + in your Pydantic payload model for /api/export_xml and /api/export_workbook_xml
add_deliverable_milestones: bool = False


Pass it through in the routes that call convert_excel_to_mspdi:

stats = convert_excel_to_mspdi(
    input_xlsx=temp_xlsx,
    output_xml=output_xml,
    sheet_name=payload.sheet_name,
    start_date_mode=payload.start_date_mode,
    fixed_start_iso=project_start_iso,
    hours_per_day=payload.hours_per_day,
    merge_identical_children=False,
    project_name=payload.project_name,
    pricing_mode=scenario.get("pricing_mode", "Flat_Blended"),
    rate_band=scenario.get("rate_band", "Standard_US"),
    blended_rate=scenario.get("blended_rate"),
    add_deliverable_milestones=bool(payload.add_deliverable_milestones)  # <-- new
)


…and in the A/B workbook export: pass the same flag into both A and B calls to convert_excel_to_mspdi. (Those endpoints currently call the converter without that flag. )

Now, when you toggle “Include Start/End anchors in XML” in Step 2, your UI can send add_deliverable_milestones: true|false to the export endpoint and the XML will include or omit the Start — DEL… / End — DEL… rows accordingly.
