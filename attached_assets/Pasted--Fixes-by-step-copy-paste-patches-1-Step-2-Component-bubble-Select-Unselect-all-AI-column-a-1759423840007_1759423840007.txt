// Load AI suggestions (right column)
async function loadAiColumn() {
  const aiDiv = document.getElementById('aiSuggestions');
  aiDiv.innerHTML = state.suggestedAI.map(s => `• ${s.label}`).join('<br>');
  document.getElementById('aiSummary').value =
    state.summaryDeliverables.map(s => `• ${s}`).join('\n');
}

// Refresh AI suggestions based on latest RFP input
document.getElementById('refreshAiBtn').onclick = async () => {
  const r = await fetch('/api/suggest_by_text', {
    method: 'POST',
    headers: {'Content-Type':'application/json'},
    body: JSON.stringify({ rfp_text: state.rfpText || "" })
  });
  const data = await r.json();
  state.suggestedAI = (data.suggested || []).map(d => ({
    code: d.deliverable_code, label: d.deliverable
  }));
  loadAiColumn();
};

// Open components bubble when a deliverable is chosen
async function openComponentsBubble(deliv) {
  const dlg = document.getElementById('componentsDialog');
  document.getElementById('compTitle').textContent = `Components – ${deliv.deliverable}`;
  const box = document.getElementById('componentsContainer');
  box.innerHTML = 'Loading…';
  // ask backend for components; fallback handled server-side
  const r = await fetch(`/api/components_for?deliverable_code=${encodeURIComponent(deliv.code)}`);
  const data = await r.json();
  const comps = data.items || [];
  // build checkboxes
  box.innerHTML = comps.map(c =>
    `<label><input class="compChk" type="checkbox" data-name="${c.name}" checked> ${c.name} <small>(${Math.round(c.hours)} h)</small></label>`
  ).join('<br>');
  // default is ALL selected
  state.selectedComponentsMap[deliv.code] = "__ALL__";

  document.getElementById('selectAllComps').onclick = () => {
    for (const chk of box.querySelectorAll('.compChk')) chk.checked = true;
    state.selectedComponentsMap[deliv.code] = "__ALL__";
  };
  document.getElementById('unselectAllComps').onclick = () => {
    for (const chk of box.querySelectorAll('.compChk')) chk.checked = false;
    state.selectedComponentsMap[deliv.code] = {}; // none selected
  };
  box.addEventListener('change', () => {
    const checked = [...box.querySelectorAll('.compChk')].filter(c => c.checked).map(c => c.dataset.name);
    state.selectedComponentsMap[deliv.code] = checked.length === comps.length
      ? "__ALL__"
      : Object.fromEntries(checked.map(n => [n, null])); // no custom hours; null=use defaults
  });

  dlg.showModal();
}

function populateSelect(selectEl, items) {
  const unique = [...new Set(items)];

    opt.value = opt.textContent = v;

}

d) 500 when ticking “retainers”

If your UI sometimes sends malformed months (e.g., "", null, or "6" as a string), the current comprehension can raise, returning a 500. Add a safe parser and reuse it in all three places you compute ret_map.

main.py – add helper near other helpers:

def _safe_retainer_map(retainers) -> dict[str,int]:
    out = {}
    for r in (retainers or []):
        code = (getattr(r, "deliverable_code", "") or (r.get("deliverable_code") if isinstance(r, dict) else "")).strip()
        raw = getattr(r, "months", None) if not isinstance(r, dict) else r.get("months")
        try:
            m = int(raw) if raw is not None else 0
        except Exception:
            m = 0
        m = max(1, min(12, m)) if m > 0 else 0
        if code:
            out[code] = m
    return out

3) Step 5 – XML must include components, tasks, subtasks
What happened & fix

Your current export pipeline will include components/tasks only if they’re present in the build’s internal structure or resolvable from the DB. In your XML I only see flat deliverables, which means no components were produced for the selected deliverables during build. I added a server‑side fallback so export always creates components/tasks even when the DB function returns nothing.

main.py — in build_wbs_with_pricing where components are added, insert a fallback right after:

comps = DB.components_for_deliverable(dcode, tg_order)
# --- NEW robust fallback if DB returns no components ---
if not comps:
    sub = DB.all_rows[DB.all_rows["Deliverable_Code"].astype(str)==str(dcode)]
    if not sub.empty:
        # Use distinct Component values; if blank, synthesize one
        comps = sorted({str(x) for x in sub["Component"].dropna().astype(str) if str(x).strip() and str(x).strip() != "nan"})
    if not comps:
        comps = ["Work Package"]  # last-resort single component
