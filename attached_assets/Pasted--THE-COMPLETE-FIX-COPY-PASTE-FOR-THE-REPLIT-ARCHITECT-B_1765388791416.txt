async function initPricingStep() {
  const sessionId = window.SESSION_ID;
  if (!sessionId) return;

  // 1. Load working scenario (NO RESET!)
  const res = await fetch('/api/scenarios', {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ session_id: sessionId, reset: false })
  });

  const data = await res.json();
  if (!data.ok || !data.scenarios || !data.scenarios.A) {
    console.error("No scenario found. Cannot load Step 3.");
    return;
  }

  // 2. Store working scenario globally
  window.SCENARIO_A = data.scenarios.A;

  // 3. Populate BOTH panels + summary cards
  await rebuildPricingFromBackend(window.SCENARIO_A);
}

document.addEventListener("DOMContentLoaded", () => {
  initPricingStep();
});

function collectScenarioFromUi() {
  const scenario = JSON.parse(JSON.stringify(window.SCENARIO_A));

  // Collect top-level deliverable hours from Project Scenario table
  document.querySelectorAll('[data-deliverable-row]').forEach(row => {
    const dcode = row.dataset.deliverableCode;
    const hours = parseFloat(row.querySelector('.deliverable-hours-input').value);
    const rate  = parseFloat(row.querySelector('.deliverable-rate-input').value);

    const item = scenario.items.find(i => String(i.Deliverable_Code) === String(dcode));
    if (item) {
      item.Planned_Hours = hours;
      item.Rate_USD = rate;
      item.Price_USD = hours * rate;
    }
  });

  // Collect component hours from Pricing Details table
  document.querySelectorAll('[data-component-row]').forEach(row => {
    const dcode = row.dataset.deliverableCode;
    const comp  = row.dataset.component;
    const hours = parseFloat(row.querySelector('.component-hours-input').value);
    const rate  = parseFloat(row.querySelector('.component-rate-input').value);

    const item = scenario.items.find(i =>
      String(i.Deliverable_Code) === String(dcode) &&
      String(i.Component) === String(comp)
    );
    if (item) {
      item.Planned_Hours = hours;
      item.Rate_USD = rate;
      item.Price_USD = hours * rate;
    }
  });

  return scenario;
}


📌 This ensures deliverable = 308 hours is always reflected in the scenario.

C. Sync before any pricing rebuild
async function syncScenarioToBackend(scenario) {
  const sessionId = window.SESSION_ID;

  const res = await fetch('/api/scenario/sync', {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      session_id: sessionId,
      scenario: scenario
    })
  });

  const data = await res.json();
  if (!data.success) throw new Error("Scenario sync failed");

  // Backend returns authoritative scenario
  window.SCENARIO_A = data.scenario;
  return data.scenario;
}

D. Rebuild pricing afterwards (one source of truth)
async function rebuildPricingFromBackend(scenario) {
  const sessionId = window.SESSION_ID;

  // 1. Sync first
  await syncScenarioToBackend(scenario);

  // 2. Ask backend for authoritative breakdown
  const res = await fetch("/api/pricing/rebuild_breakdown", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ session_id: sessionId })
  });

  const data = await res.json();

  // 3. Render BOTH panels from backend response
  renderProjectScenario(data.breakdown);
  renderPricingDetails(data.breakdown);
  renderSummaryCards(data.summary);

  return data;
}
