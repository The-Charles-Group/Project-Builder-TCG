class ReorderPayload(BaseModel):
    scenario_letter: str
    deliverable_codes: list[str]                        # new order from UI
    included_map: dict[str, list[str]] | None = None   # {code: [task_groups]}
    use_slack: bool = True
    slack_after_internal: int = 1
    slack_after_client: int = 2
    slack_global_pct: float = 0.0
    project_start: str | None = None                   # "YYYY-MM-DD"
    complexity: str = "Advanced"
    tier: str = "T2_MediumVolume"

@app.post("/api/reorder_timeline")
def api_reorder_timeline(p: ReorderPayload):
    if not DB.loaded:
        DB.load()

    letter = (p.scenario_letter or "A").upper()
    items: list[dict] = []
    cursor_date = None  # sequential chain: each deliverable starts after the previous ends

    for code in p.deliverable_codes:
        # included task groups (from UI) else derive all that exist for that deliverable
        included = (p.included_map or {}).get(code)
        if not included:
            sub = DB.all_rows[DB.all_rows["Deliverable_Code"].astype(str) == str(code)]
            included = sorted(set(sub["task_group"].dropna().astype(str).tolist()))

        # first deliverable uses project_start; others start the day after prior end
        start = p.project_start if cursor_date is None else str(cursor_date)

        sched = DB.build_schedule(
            deliverable_code=code,
            included_task_groups=included,
            complexity=p.complexity, tier=p.tier,
            use_slack=p.use_slack,
            slack_after_internal=p.slack_after_internal,
            slack_after_client=p.slack_after_client,
            slack_global_pct=p.slack_global_pct,
            project_start=start,
            scenario_letter=letter
        )

        items.append({"deliverable_code": code, "schedule": sched})

        if sched:
            last_end = sched[-1]["end_date"]
            y, m, d = map(int, last_end.split("-"))
            cursor_date = datetime.date(y, m, d) + datetime.timedelta(days=1)

    return {"items": items}

// ——— helpers ———
function getScenario(letter){ return (window.appState?.scenarios||{})[letter] || null; }
function setText(id, t){ const el=document.getElementById(id); if(el) el.textContent=t; }
function setHTML(id, h){ const el=document.getElementById(id); if(el) el.innerHTML=h; }

// ——— render the timeline for a scenario ———
function renderTimeline(letter) {
  const scen = getScenario(letter);
  const body = document.getElementById('tl-body');
  document.querySelectorAll('[data-timeline-sel]').forEach(b =>
    b.classList.toggle('active', b.dataset.timelineSel === letter)
  );
  if (!scen || !Array.isArray(scen.items) || !scen.items.length) {
    setHTML('tl-body','<tr><td colspan="4"><em>Build scenarios in Step 3 to see a timeline.</em></td></tr>');
    return;
  }
  const rows = scen.items.map(d => {
    const sch = d.schedule||[];
    const start = sch[0]?.start_date ?? '';
    const end   = sch[sch.length-1]?.end_date ?? '';
    const days  = sch.reduce((n,s)=>n+(s.duration_days||0),0);
    return `<tr class="tl-row" draggable="true" data-dcode="${d.deliverable_code}">
      <td>${d.deliverable}</td><td>${start}</td><td>${end}</td><td>${days}</td>
    </tr>`;
  }).join('');
  setHTML('tl-body', rows);
  enableTimelineDnD(letter);
}

// ——— drag & drop wiring ———
function enableTimelineDnD(letter){
  const body = document.getElementById('tl-body');
  let dragEl = null;

  body.querySelectorAll('.tl-row').forEach(row=>{
    row.addEventListener('dragstart', e => { dragEl=row; row.classList.add('dragging'); });
    row.addEventListener('dragend',   e => { row.classList.remove('dragging'); dragEl=null; });
    row.addEventListener('dragover',  e => {
      e.preventDefault();
      const after = getAfter(body, e.clientY);
      if(!after) body.appendChild(row); else if(after!==row) body.insertBefore(row, after);
    });
  });

  function getAfter(container, y){
    const els=[...container.querySelectorAll('.tl-row:not(.dragging)')];
    return els.reduce((closest, child)=>{
      const box=child.getBoundingClientRect();
      const offset = y - (box.top + box.height/2);
      return (offset<0 && offset>closest.offset) ? {offset, element:child} : closest;
    }, {offset: -Infinity}).element;
  }

  // add/refresh Save button
  let btn = document.getElementById('tl-save');
  if(!btn){
    btn = document.createElement('button');
    btn.id='tl-save'; btn.textContent='Save Order';
    document.getElementById('timeline-controls').appendChild(btn);
  }
  btn.onclick = () => saveTimelineOrder(letter);
}

// ——— persist new order to backend and merge schedules back ———
async function saveTimelineOrder(letter){
  const body = document.getElementById('tl-body');
  const codes = [...body.querySelectorAll('.tl-row')].map(tr=>tr.dataset.dcode);
  const scen = getScenario(letter) || {};
  const includedMap = {};
  (scen.items||[]).forEach(it => includedMap[it.deliverable_code] = it.included_task_groups || []);

  const payload = {
    scenario_letter: letter,
    deliverable_codes: codes,
    included_map: includedMap,
    use_slack: !!scen.use_slack,
    slack_after_internal: scen.slack_after_internal ?? 1,
    slack_after_client:  scen.slack_after_client ?? 2,
    slack_global_pct:    scen.slack_global_pct ?? 0,
    project_start:       scen.project_start || null,
    complexity:          (scen.items?.[0]?.complexity || 'Advanced'),
    tier:                (scen.items?.[0]?.tier || 'T2_MediumVolume')
  };

  const res = await fetch('/api/reorder_timeline', {
    method: 'POST', headers: {'Content-Type':'application/json'},
    body: JSON.stringify(payload)
  }).then(r => r.json());

  if(res?.items?.length){
    const byCode = new Map(res.items.map(i=>[i.deliverable_code, i.schedule]));
    scen.items = (scen.items||[]).map(it => byCode.has(it.deliverable_code) ? {...it, schedule: byCode.get(it.deliverable_code)} : it);
    (window.appState.scenarios||{})[letter] = scen;
    renderTimeline(letter);
  } else {
    alert('Reorder did not return schedules. Check backend /api/reorder_timeline.');
  }
}
