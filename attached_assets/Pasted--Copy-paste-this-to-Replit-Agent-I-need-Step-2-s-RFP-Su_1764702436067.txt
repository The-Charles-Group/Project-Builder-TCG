Introduce a helper that uses summary.summary_bullets when available, and falls back to summary.summary_text split on \n if not.

If we’re using plain JS:

function renderRfpSummary(summary) {
  const bodyEl = document.getElementById("rfp-summary-body");
  if (!bodyEl) return;

  const bullets =
    (summary.summary_bullets && summary.summary_bullets.length)
      ? summary.summary_bullets.map(b => ({
          label: b.label || "",
          short_desc: b.short_desc || ""
        }))
      : (summary.summary_text || "")
          .split(/\n+/)
          .map(line => line.trim())
          .filter(Boolean)
          .map(line => {
            line = line.replace(/^•\s*/, "");
            const [label, ...rest] = line.split(":");
            return {
              label: (label || "").trim(),
              short_desc: rest.join(":").trim()
            };
          });

  if (!bullets.length) {
    bodyEl.innerHTML = '<p class="text-muted">No RFP summary available.</p>';
    return;
  }

  const itemsHtml = bullets.map(b => {
    const labelHtml = b.label ? `<strong>${b.label}</strong>` : "";
    const descHtml  = b.short_desc ? ` — ${b.short_desc}` : "";
    return `<li>${labelHtml}${descHtml}</li>`;
  }).join("");

  bodyEl.innerHTML = `<ul class="rfp-summary-list">${itemsHtml}</ul>`;
}
