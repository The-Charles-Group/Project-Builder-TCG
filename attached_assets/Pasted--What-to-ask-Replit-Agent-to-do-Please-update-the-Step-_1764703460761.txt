function renderRfpSummary(summary) {
  const bodyEl = document.getElementById("rfp-summary-body");
  if (!bodyEl) return;

  const raw = (summary.summary_text || "").trim();
  if (!raw) {
    bodyEl.innerHTML = '<p class="text-muted">No RFP summary available.</p>';
    return;
  }

  // Split on newlines, clean up, and strip the leading bullet dot
  const lines = raw
    .split(/\n+/)
    .map(line => line.trim())
    .filter(Boolean)
    .map(line => line.replace(/^•\s*/, ""));

  const itemsHtml = lines
    .map(line => {
      // Optional: bold the label before the colon, like “Brand Strategy”
      const [label, ...rest] = line.split(":");
      const labelHtml = label ? `<strong>${label.trim()}</strong>` : "";
      const desc = rest.join(":").trim();
      const descHtml = desc ? ` — ${desc}` : "";
      return `<li>${labelHtml}${descHtml}</li>`;
    })
    .join("");

  bodyEl.innerHTML = `<ul class="rfp-summary-list">${itemsHtml}</ul>`;
}
