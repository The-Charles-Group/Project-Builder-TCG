Minimal store (React Context + useReducer)

Create /src/state/step2Context.tsx:

import React, { createContext, useContext, useMemo, useReducer, useEffect } from 'react';

export type DeliverableCode = string;
export type ComponentName   = string;
export type L3Key           = string; // Task_Label or task_group fallback

type L3Buckets = Record<ComponentName, L3Key[]>;
type L3Sets    = Record<ComponentName, Set<L3Key>>;

type State = {
  selectedDeliverables: Set<DeliverableCode>;
  selectedComponents:   Record<DeliverableCode, Set<ComponentName>>;
  l3Options:            Record<DeliverableCode, L3Buckets>; // from API (cache)
  l3Selected:           Record<DeliverableCode, L3Sets>;     // user-checked L3
  l3Hidden:             Record<DeliverableCode, L3Sets>;     // user-removed L3
  active: { deliverable?: DeliverableCode; component?: ComponentName }; // preview only
};

type Action =
  | { type: 'SET_ACTIVE'; d?: DeliverableCode; c?: ComponentName }
  | { type: 'TOGGLE_DELIVERABLE'; d: DeliverableCode; checked: boolean }
  | { type: 'TOGGLE_COMPONENT'; d: DeliverableCode; c: ComponentName; checked: boolean; restorePrior?: boolean }
  | { type: 'SEED_L3'; d: DeliverableCode; c: ComponentName; list: L3Key[] }
  | { type: 'TOGGLE_L3'; d: DeliverableCode; c: ComponentName; k: L3Key; checked: boolean }
  | { type: 'SELECT_ALL_L3'; d: DeliverableCode; c: ComponentName }
  | { type: 'CLEAR_ALL_L3'; d: DeliverableCode; c: ComponentName }
  | { type: 'RESET_L3_DEFAULTS'; d: DeliverableCode; c: ComponentName }
  | { type: 'HYDRATE'; snapshot: Serialized };

type Serialized = {
  selectedDeliverables: string[];
  selectedComponents:   Record<string,string[]>;
  l3Options:            Record<string,Record<string,string[]>>;
  l3Selected:           Record<string,Record<string,string[]>>;
  l3Hidden:             Record<string,Record<string,string[]>>;
  active:               { deliverable?: string; component?: string };
};

const STORAGE_KEY = 'step2-state-v281';

const toArr = (s?: Set<string>) => Array.from(s ?? new Set<string>());
const toSet = (a?: string[]) => new Set(a ?? []);
const ensure = <T,>(obj: Record<string, T>, key: string, init: T): T => (obj[key] ??= init);

const initialState: State = {
  selectedDeliverables: new Set(),
  selectedComponents:   {},
  l3Options:            {},
  l3Selected:           {},
  l3Hidden:             {},
  active:               {},
};

function serialize(state: State): Serialized {
  const mapSetToArr = (m: Record<string, Set<string>>) =>
    Object.fromEntries(Object.entries(m).map(([k, v]) => [k, toArr(v)]));

  const nestedSetToArr = (m: Record<string, Record<string, Set<string>>>) =>
    Object.fromEntries(Object.entries(m).map(([d, inner]) => [
      d, Object.fromEntries(Object.entries(inner).map(([c, set]) => [c, toArr(set)]))
    ]));

  const l3OptionsFlat: Serialized['l3Options'] = {};
  for (const [d, b] of Object.entries(state.l3Options)) l3OptionsFlat[d] = { ...b };

  return {
    selectedDeliverables: toArr(state.selectedDeliverables),
    selectedComponents:   Object.fromEntries(Object.entries(state.selectedComponents).map(([d, s]) => [d, toArr(s)])),
    l3Options:            l3OptionsFlat,
    l3Selected:           nestedSetToArr(state.l3Selected),
    l3Hidden:             nestedSetToArr(state.l3Hidden),
    active:               state.active,
  };
}

function deserialize(s: Serialized): State {
  const selectedComponents: State['selectedComponents'] = {};
  for (const [d, arr] of Object.entries(s.selectedComponents || {})) {
    selectedComponents[d] = toSet(arr);
  }

  const l3Selected: State['l3Selected'] = {};
  for (const [d, inner] of Object.entries(s.l3Selected || {})) {
    const obj: Record<string, Set<string>> = {};
    for (const [c, arr] of Object.entries(inner || {})) obj[c] = toSet(arr);
    l3Selected[d] = obj;
  }

  const l3Hidden: State['l3Hidden'] = {};
  for (const [d, inner] of Object.entries(s.l3Hidden || {})) {
    const obj: Record<string, Set<string>> = {};
    for (const [c, arr] of Object.entries(inner || {})) obj[c] = toSet(arr);
    l3Hidden[d] = obj;
  }

  return {
    selectedDeliverables: toSet(s.selectedDeliverables),
    selectedComponents,
    l3Options: s.l3Options || {},
    l3Selected,
    l3Hidden,
    active: s.active || {},
  };
}

function reducer(state: State, action: Action): State {
  switch (action.type) {
    case 'HYDRATE':
      return deserialize(action.snapshot);

    case 'SET_ACTIVE':
      return { ...state, active: { deliverable: action.d, component: action.c } };

    case 'TOGGLE_DELIVERABLE': {
      const s = new Set(state.selectedDeliverables);
      if (action.checked) s.add(action.d); else s.delete(action.d);
      if (!action.checked) {
        // Drop state for this deliverable
        const { [action.d]: _, ...restComps } = state.selectedComponents;
        const { [action.d]: __, ...restSel }  = state.l3Selected;
        const { [action.d]: ___, ...restHid } = state.l3Hidden;
        const { [action.d]: ____, ...restOpt }= state.l3Options;
        return { ...state, selectedDeliverables: s, selectedComponents: restComps, l3Selected: restSel, l3Hidden: restHid, l3Options: restOpt };
      }
      return { ...state, selectedDeliverables: s };
    }

    case 'TOGGLE_COMPONENT': {
      const comps = ensure({ ...state.selectedComponents }, action.d, new Set<ComponentName>(state.selectedComponents[action.d] ?? []));
      const sel   = ensure({ ...state.l3Selected }, action.d, { ...(state.l3Selected[action.d] ?? {}) });
      const hid   = ensure({ ...state.l3Hidden },   action.d, { ...(state.l3Hidden[action.d]   ?? {}) });

      if (action.checked) {
        comps.add(action.c);
        // seed defaults unless asked to restore prior
        if (!sel[action.c] || !action.restorePrior) {
          const defaults = state.l3Options[action.d]?.[action.c] ?? [];
          sel[action.c] = new Set(defaults);
          hid[action.c] = new Set();
        }
      } else {
        comps.delete(action.c);
        // keep sel/hid so “restore prior” can work when re-adding
      }
      return { ...state, selectedComponents: { ...state.selectedComponents, [action.d]: comps }, l3Selected: sel, l3Hidden: hid };
    }

    case 'SEED_L3': {
      const opt = ensure({ ...state.l3Options }, action.d, { ...(state.l3Options[action.d] ?? {}) });
      opt[action.c] = [...new Set(action.list)];
      // if component is selected and no prior selection, adopt defaults
      if (state.selectedComponents[action.d]?.has(action.c)) {
        const sel = ensure({ ...state.l3Selected }, action.d, { ...(state.l3Selected[action.d] ?? {}) });
        if (!sel[action.c]) sel[action.c] = new Set(opt[action.c]);
        return { ...state, l3Options: opt, l3Selected: sel };
      }
      return { ...state, l3Options: opt };
    }

    case 'TOGGLE_L3': {
      const sel = ensure(ensure({ ...state.l3Selected }, action.d, { ...(state.l3Selected[action.d] ?? {}) }), action.c, new Set(state.l3Selected[action.d]?.[action.c] ?? []));
      const hid = ensure(ensure({ ...state.l3Hidden },   action.d, { ...(state.l3Hidden[action.d]   ?? {}) }), action.c, new Set(state.l3Hidden[action.d]?.[action.c] ?? []));
      if (action.checked) { sel.add(action.k); hid.delete(action.k); } else { sel.delete(action.k); hid.add(action.k); }
      return { ...state, l3Selected: { ...state.l3Selected }, l3Hidden: { ...state.l3Hidden } };
    }

    case 'SELECT_ALL_L3': {
      const all = new Set(state.l3Options[action.d]?.[action.c] ?? []);
      const sel = ensure(ensure({ ...state.l3Selected }, action.d, { ...(state.l3Selected[action.d] ?? {}) }), action.c, new Set());
      const hid = ensure(ensure({ ...state.l3Hidden },   action.d, { ...(state.l3Hidden[action.d]   ?? {}) }), action.c, new Set());
      sel.clear(); hid.clear();
      state.l3Selected[action.d][action.c] = all;
      return { ...state, l3Selected: { ...state.l3Selected }, l3Hidden: { ...state.l3Hidden } };
    }

    case 'CLEAR_ALL_L3': {
      const sel = ensure(ensure({ ...state.l3Selected }, action.d, { ...(state.l3Selected[action.d] ?? {}) }), action.c, new Set());
      const hid = ensure(ensure({ ...state.l3Hidden },   action.d, { ...(state.l3Hidden[action.d]   ?? {}) }), action.c, new Set());
      sel.clear();
      const all = state.l3Options[action.d]?.[action.c] ?? [];
      state.l3Hidden[action.d][action.c] = new Set(all);
      return { ...state, l3Selected: { ...state.l3Selected }, l3Hidden: { ...state.l3Hidden } };
    }

    case 'RESET_L3_DEFAULTS': {
      const defaults = state.l3Options[action.d]?.[action.c] ?? [];
      const sel = ensure(ensure({ ...state.l3Selected }, action.d, { ...(state.l3Selected[action.d] ?? {}) }), action.c, new Set());
      const hid = ensure(ensure({ ...state.l3Hidden },   action.d, { ...(state.l3Hidden[action.d]   ?? {}) }), action.c, new Set());
      sel.clear(); hid.clear();
      state.l3Selected[action.d][action.c] = new Set(defaults);
      return { ...state, l3Selected: { ...state.l3Selected }, l3Hidden: { ...state.l3Hidden } };
    }

    default:
      return state;
  }
}

const Ctx = createContext<{ state: State; dispatch: React.Dispatch<Action> } | null>(null);

export function Step2Provider({ children }: { children: React.ReactNode }) {
  const [state, dispatch] = useReducer(reducer, initialState);

  // hydrate once
  useEffect(() => {
    try {
      const raw = localStorage.getItem(STORAGE_KEY);
      if (raw) dispatch({ type: 'HYDRATE', snapshot: JSON.parse(raw) as Serialized });
    } catch {}
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  // persist on every change
  useEffect(() => {
    try { localStorage.setItem(STORAGE_KEY, JSON.stringify(serialize(state))); } catch {}
  }, [state]);

  const value = useMemo(() => ({ state, dispatch }), [state]);
  return <Ctx.Provider value={value}>{children}</Ctx.Provider>;
}

export function useStep2() {
  const ctx = useContext(Ctx);
  if (!ctx) throw new Error('useStep2 must be used within Step2Provider');
  // derived counts from the same source that renders the UI
  const counts = useMemo(() => {
    const deliverables = ctx.state.selectedDeliverables.size;
    const components = Object.values(ctx.state.selectedComponents).reduce((n, s) => n + (s?.size ?? 0), 0);
    const l3 = Object.values(ctx.state.l3Selected).reduce(
      (sum, compMap) => sum + Object.values(compMap ?? {}).reduce((m, set) => m + (set?.size ?? 0), 0), 0
    );
    return { deliverables, components, l3 };
  }, [ctx.state.selectedDeliverables, ctx.state.selectedComponents, ctx.state.l3Selected]);
  return { ...ctx, counts };
}

How to use it in Step‑2

Wrap your page:

// Step2Page.tsx
import { Step2Provider } from '@/state/step2Context';

export default function Step2Page() {
  return (
    <Step2Provider>
      {/* left: Deliverables, middle: Components, right: Summary */}
    </Step2Provider>
  );
}


Deliverables list (fixes the “row click selects” bug):

import { useStep2 } from '@/state/step2Context';

function DeliverablesPane({ items }: { items: { code: string; label: string }[] }) {
  const { state, dispatch } = useStep2();
  return (
    <div>
      {items.map(d => {
        const checked = state.selectedDeliverables.has(d.code);
        return (
          <div key={d.code} onClick={() => dispatch({ type: 'SET_ACTIVE', d: d.code })}>
            <input
              type="checkbox"
              checked={checked}
              onClick={(e) => e.stopPropagation()}
              onChange={(e) => dispatch({ type: 'TOGGLE_DELIVERABLE', d: d.code, checked: e.currentTarget.checked })}
            />
            {d.label}
          </div>
        );
      })}
    </div>
  );
}

Summary pane (scrollable + checkboxes + per‑component controls):

import { useStep2 } from '@/state/step2Context';

export function SummaryPane() {
  const { state, dispatch, counts } = useStep2();

  return (
    <aside>
      <div>Deliverables: {counts.deliverables} | Components: {counts.components} | L3 Subtasks: {counts.l3}</div>
      <div style={{ maxHeight: 420, overflowY: 'auto' }}>
        {Array.from(state.selectedDeliverables).map(d => (
          <section key={d}>
            <h4>{d}</h4>
            {Array.from(state.selectedComponents[d] ?? new Set()).map(c => {
              const all = state.l3Options[d]?.[c] ?? [];
              const sel = state.l3Selected[d]?.[c] ?? new Set<string>();
              return (
                <div key={c} style={{ border: '1px solid var(--hairline)', borderRadius: 8, marginBottom: 10 }}>
                  <div style={{ display: 'flex', justifyContent: 'space-between', padding: 8 }}>
                    <strong>{c}</strong>
                    <span>
                      <button onClick={() => dispatch({ type: 'SELECT_ALL_L3', d, c })}>All</button>
                      <button onClick={() => dispatch({ type: 'CLEAR_ALL_L3', d, c })}>Clear</button>
                      <button onClick={() => dispatch({ type: 'RESET_L3_DEFAULTS', d, c })}>Reset</button>
                    </span>
                  </div>
                  <div style={{ maxHeight: 260, overflowY: 'auto', padding: '4px 10px' }}>
                    {all.map(k => (
                      <label key={k} style={{ display: 'flex', gap: 8, padding: '4px 0' }}>
                        <input
                          type="checkbox"
                          checked={sel.has(k)}
                          onChange={(e) => dispatch({ type: 'TOGGLE_L3', d, c, k, checked: e.currentTarget.checked })}
                        />
                        <span>{k}</span>
                      </label>
                    ))}
                  </div>
                </div>
              );
            })}
          </section>
        ))}
      </div>
    </aside>
  );
}
