// Ensure Stage‑2 containers exist so we never write into null nodes
function ensureStage2Dom() {
  const targets = {
    'left-list':   '#stage2-left',
    'left-total':  '#stage2-left',
    'add-form':    '#stage2-middle',
    'del-form':    '#stage2-middle',
    'summary':     '#stage2-right',
    'wordcount':   '#stage2-right'
  };
  for (const [id, parentSel] of Object.entries(targets)) {
    if (!$(id)) {
      const parent = document.querySelector(parentSel) || document.body;
      const tag = (id === 'summary') ? 'pre' : (id.endsWith('form') ? 'form' : 'div');
      const el = document.createElement(tag);
      el.id = id;
      parent.appendChild(el);
    }
  }
}
document.addEventListener('DOMContentLoaded', ensureStage2Dom);


If your Stage‑2 columns use different container selectors, just change #stage2-left, #stage2-middle, #stage2-right to your actual wrappers. This prevents the null write that triggered the alert.

2) Wire Analyze in the correct order (no null writes)

Call: summarize_by_file → suggest_by_file → build (Average) → reconcile.
This seeds Step 2 left panel and stores the selected codes so Step 3 works.

async function analyzeFile(file) {
  ensureStage2Dom();

  // 1) AI Summary (right panel, GPT‑5 only)
  const fd = new FormData(); fd.append('file', file);
  const sum = await fetch('/api/summarize_by_file', { method:'POST', body: fd }).then(r => r.json());

  setText('wordcount', `${sum.word_count}/500 words`);
  setText('summary',  sum.summary_text);

  // 2) DB suggestions (left) strictly from rules
  const sugg = await fetch('/api/suggest_by_file', { method:'POST', body: fd }).then(r => r.json());
  const selectedCodes = (sugg || []).map(s => s.deliverable_code);
  window.appState = window.appState || {};
  appState.selectedCodes = selectedCodes;

  // 3) Build Average scenario now to populate left panel with pricing
  const scenarios = await fetch('/api/build', {
    method: 'POST',
    headers: {'Content-Type':'application/json'},
    body: JSON.stringify({
      selected_deliverable_codes: selectedCodes,
      pricing_mode: 'Flat_Blended',
      blended_rate: 195,
      rate_band: 'Standard_US',
      scenario_a: {mode:'template', complexity:'Advanced', tier:'T2_MediumVolume'},
      scenario_b: {mode:'template', complexity:'Advanced', tier:'T2_MediumVolume'}
    })
  }).then(r => r.json());

  // Render left panel (Average = Scenario A)
  renderLeftPanel(scenarios.A.items || [], scenarios.A.totals || {price_total:0});

  // 4) Reconcile (middle panel) using AI labels vs selected codes
  const aiLabels = (sum.deliverables || []).map(d => d.label);
  const recon = await fetch('/api/reconcile', {
    method: 'POST',
    headers: {'Content-Type':'application/json'},
    body: JSON.stringify({
      summary_deliverables: aiLabels,
      db_selected_deliverable_codes: selectedCodes    // pass codes to keep panels in sync
    })
  }).then(r => r.json());

  renderReconcile(recon);
}


Renderers (safe, no null writes):

function renderLeftPanel(items, totals) {
  const rows = (items || []).map(i =>
    `<div class="row"><span>${i.deliverable}</span><span>$${(i.price_total ?? 0).toLocaleString()}</span></div>`
  );
  setHTML('left-list', rows.length ? rows.join('') : '<em>No deliverables selected</em>');
  setText('left-total', `Total: $${(totals?.price_total ?? 0).toLocaleString()}`);
}

function renderReconcile(data) {
  const addRows = (data.add || []).map(s =>
    `<label><input type="checkbox" class="add" data-code="${s.code}" checked> ${s.label} – ${s.reason}</label>`
  );
  const delRows = (data.delete || []).map(s =>
    `<label class="danger"><input type="checkbox" class="del" data-code="${s.code}" checked> ${s.label} – ${s.reason}</label>`
  );
  setHTML('add-form', addRows.length ? addRows.join('<br>') : '<em>No suggestions to add</em>');
  setHTML('del-form', delRows.length ? delRows.join('<br>') : '<em>No deletions suggested</em>');
}


Apply Selection (keeps Step 3 happy):

function applySelection() {
  const curr = new Set(appState.selectedCodes || []);
  document.querySelectorAll('#add-form input.add:checked').forEach(cb => curr.add(cb.dataset.code));
  document.querySelectorAll('#del-form input.del:checked').forEach(cb => curr.delete(cb.dataset.code));
  appState.selectedCodes = [...curr];
  // Optionally, re‑build Average to refresh left panel total after changes
}
