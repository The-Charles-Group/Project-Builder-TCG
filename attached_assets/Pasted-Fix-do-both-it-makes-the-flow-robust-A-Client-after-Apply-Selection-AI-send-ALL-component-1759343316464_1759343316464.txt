def _inflate_components_if_missing(scen: dict) -> dict:
    for item in scen.get("items", []):
        dcode = str(item.get("deliverable_code") or item.get("Deliverable_Code") or "").strip()
        if not dcode:
            continue
        # If no component info, fetch all task groups/components from DB
        included = item.get("included_task_groups") or []
        if not included:
            # Derive all task groups for this deliverable (DB knows how) and keep order by driver
            included = DB.task_groups_for_deliverable(dcode)
            item["included_task_groups"] = included

        # If components missing, the WBS builder will still create them using included/tg order.
        # (It reads included groups, sorts them, and then enumerates components and tasks.)
    return scen
