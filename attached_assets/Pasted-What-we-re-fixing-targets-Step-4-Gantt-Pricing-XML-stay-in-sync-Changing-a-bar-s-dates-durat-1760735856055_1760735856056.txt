# main.py (near other globals)
SCENARIO_STORE: dict[str, dict] = {}  # session_id -> scenario dict

def _recompute_totals(scn: dict) -> dict:
    for row in scn["items"]:
        row["Price_USD"] = round(float(row.get("Rate_USD", 0)) * float(row.get("Planned_Hours", 0)), 2)
    scn["totals"] = {
        "hours": sum(r.get("Planned_Hours", 0) for r in scn["items"]),
        "price": round(sum(r.get("Price_USD", 0) for r in scn["items"]), 2)
    }
    return scn
