async function buildAB({useRetainers}) {
  const selectedCodes = Array.from(window.apb?.selectedCodes || []); // comes from Step 2 “Apply selection”
  if (!selectedCodes.length) { alert("Pick deliverables in Step 2 first."); return; }

  // collect retainer months (if the toggle is on)
  const retainers = useRetainers
    ? Array.from(document.querySelectorAll('[data-retain-code]')).map(inp => ({
        deliverable_code: inp.dataset.retainCode,
        months: Number(inp.value || 0)
      }))
    : [];

  const payload = {
    selected_deliverable_codes: selectedCodes,       // ⬅️ NEVER omit this
    scenario_a: {mode: "template", scenario_key: window.apb?.scenarioAKey || "MED_LOW"},
    scenario_b: {mode: "template", scenario_key: window.apb?.scenarioBKey || "MED_HIGH"},
    pricing_mode: document.getElementById('pricingMode').value,
    blended_rate: Number(document.getElementById('blendedRate').value || 0),
    rate_band: document.getElementById('rateBand').value,
    use_slack: true,
    slack_after_internal: 1,
    slack_after_client: 2,
    slack_global_pct: 0.05,
    project_start: document.getElementById('projectStart').value || null,
    // NEW: keep explicit component choices from Step 2 (or '__ALL__' sentinel)
    included_task_groups_map: window.apb?.includedMap || "__ALL__",
    // NEW: retainers
    retainers
  };

  const r = await fetch("/api/build", {method:"POST", headers:{'Content-Type':'application/json'}, body: JSON.stringify(payload)});
  const data = await r.json();
  window.apb.scenarios = data.scenarios;     // replace (do not merge)
  renderPricing(data.scenarios);
}


Why this matters: the server does not fall back to any previous selection. If you don’t send the codes again, it builds an empty scenario and you get zeros. (The server totals you saw are literally the sum over per_deliv, which is empty if selected_deliverable_codes is empty. )

Root cause 2 — We’re appending instead of replacing after a rebuild

When you click Build Scenarios A & B after toggling retainers, the UI is concatenating the new items with the old tables (hence the duplicate “Meetings / Reporting / Post‑Production”).

function renderPricing(scenarios){
  const A = scenarios.A, B = scenarios.B;
  // 🔧 clear before render to avoid duplication
  document.getElementById('pricingA').innerHTML = "";
  document.getElementById('pricingB').innerHTML = "";
  drawScenarioPanel('pricingA', A);
  drawScenarioPanel('pricingB', B);
}

function hydratePricingOptions(opts){
  const pm = document.getElementById('pricingMode');
  const rb = document.getElementById('rateBand');

  pm.replaceChildren();   // ⬅️ prevents duplicates
  rb.replaceChildren();

  for (const v of (opts.pricing_modes || ["Flat_Blended","Per_Resource"])) {
    pm.append(new Option(v.replace("_"," "), v));
  }
  for (const v of (opts.rate_bands || ["Standard_US"])) {
    rb.append(new Option(v, v));
  }
}
