# --- Write XML with a guaranteed MSPDI-friendly prolog ---
import re

xml_string = tostring(project, encoding='unicode')

try:
    dom = minidom.parseString(xml_string)
    pretty_xml = dom.toprettyxml(indent="  ", encoding=None)
    # Remove blank lines that toprettyxml adds
    payload = "\n".join([ln for ln in pretty_xml.split("\n") if ln.strip()])
except Exception:
    payload = xml_string

# Guarantee: exact header, uppercase UTF-8, no leading whitespace or BOM
header = '<?xml version="1.0" encoding="UTF-8"?>'
p = payload.lstrip()
if p.startswith("<?xml"):
    p = re.sub(r'^<\?xml[^>]*\?>', header, p, count=1)
else:
    p = header + "\n" + p

with open(output_xml, "w", encoding="utf-8", newline="\n") as f:
    f.write(p)
