1.4 JS: ensure we have a sequential schedule for the active scenario

The server packs deliverables sequentially (start = day after previous end) when you call /api/reorder_timeline — include scenario_letter: 'A'|'B'|'C'. It was already coded to accept C.

Add this helper (call it from the handlers above and after any build completes):

async function ensureTimeline(letter) {
  const scen = state.scenarios[letter];
  const table = document.getElementById('timelineTable');
  table.innerHTML = '';

  if (!scen || !Array.isArray(scen.items) || !scen.items.length) {
    table.innerHTML = `<tr><td colspan="4">Build Scenario ${letter} in Step 3 to see a timeline.</td></tr>`;
    return;
  }

  // Current order preference: user_order > ai_order > as-built
  const want = (scen.user_order && scen.user_order.length)
    ? scen.user_order
    : (scen.ai_order && scen.ai_order.length)
      ? scen.ai_order
      : scen.items.map(it => String(it.deliverable_code));

  const includedMap = {};
  scen.items.forEach(it => { includedMap[String(it.deliverable_code)] = it.included_task_groups || []; });

  const payload = {
    scenario_letter: letter,
    deliverable_codes: want,
    included_map: includedMap,
    project_start: scen.project_start,
    complexity: scen.complexity || 'Advanced',
    tier: scen.tier || 'T2_MediumVolume',
    use_slack: !!scen.use_slack,
    slack_after_internal: Number(scen.slack_after_internal ?? 1),
    slack_after_client: Number(scen.slack_after_client ?? 2),
    slack_global_pct: Number(scen.slack_global_pct ?? 0)
  };

  const res = await fetch('/api/reorder_timeline', {
    method: 'POST', headers: {'Content-Type': 'application/json'},
    body: JSON.stringify(payload)
  });
  if (!res.ok) {
    const err = await res.text();
    console.error(err);
    table.innerHTML = `<tr><td colspan="4">Unable to build timeline for Scenario ${letter}.</td></tr>`;
    return;
  }
  const data = await res.json();

  // Persist packed scenario (server returns items + full scenario)
  if (data.scenario) state.scenarios[letter] = data.scenario;

  // Render rows
  const items = data.items || scen.items;
  const rows = items.map(d => {
    const sched = d.schedule || [];
    const start = sched.length ? sched[0].start_date : '';
    const end   = sched.length ? sched[sched.length - 1].end_date : '';
    const duration = sched.reduce((s,r) => s + Number(r.duration_days||0), 0);
    return `<tr>
      <td>${d.deliverable}</td>
      <td>${start}</td>
      <td>${end}</td>
      <td style="text-align:right">${duration}</td>
    </tr>`;
  }).join('');
  table.innerHTML = `
    <thead>
      <tr><th>DELIVERABLE</th><th>START</th><th>END</th><th>DURATION (DAYS)</th></tr>
    </thead>
    <tbody>${rows}</tbody>
  `;
}
