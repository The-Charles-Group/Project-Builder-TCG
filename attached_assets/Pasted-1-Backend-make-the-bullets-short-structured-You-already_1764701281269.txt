def _truncate_words(s: str, max_words: int = 30) -> str:
    """Trim text to at most max_words words."""
    words = re.findall(r"\b\w+\b", s or "")
    if len(words) <= max_words:
        return (s or "").strip()

2.1 Render as a <ul> / <li> list

Where you render the RFP summary card in Step 2:

Give the body area an ID, e.g. <div id="rfp-summary-body"></div>.

Replace the plain text assignment with something like:

function renderSummaryCard(summary) {
  const body = document.getElementById("rfp-summary-body");
  body.innerHTML = "";

  const bullets = (summary.summary_text || "")
    .split("\n")
    .map(line => line.trim())
    .filter(Boolean);

  const ul = document.createElement("ul");
  ul.className = "rfp-summary-bullets";

  bullets.forEach(line => {
    const li = document.createElement("li");
    // strip the leading "• " if present
    li.textContent = line.replace(/^•\s*/, "");
    ul.appendChild(li);
  });

  body.appendChild(ul);

  // Key Deliverables list, channels/markets/complexity pills
  // keep using summary.deliverables, summary.channels, summary.markets, summary.complexity
}
