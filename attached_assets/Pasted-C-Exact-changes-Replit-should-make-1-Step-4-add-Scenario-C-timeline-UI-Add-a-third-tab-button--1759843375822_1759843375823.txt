Example (front‑end pseudo‑code):

async function saveOrder(letter) {
  const body = {
    letter, // "A" | "B" | "C"
    project_start: state.projectStartISO,
    deliverable_codes: state.timeline[letter].map(x => x.deliverable_code),
    included_map: state.includedMap,         // TGs by deliverable (if any)
    complexity: state.complexity,
    tier: state.volumeTier,
    use_slack: state.useSlack,
    slack_after_internal: state.slackInt,
    slack_after_client: state.slackClient,
    slack_global_pct: state.slackPct
  };
  const res = await fetch('/api/reorder_timeline', {method:'POST', body:JSON.stringify(body), headers:{'Content-Type':'application/json'}});
  const data = await res.json();
  // IMPORTANT: replace your in‑memory scenario with what the server returns:
  state.scenarios[letter] = data.scenario;
  state.timeline[letter] = data.items;
}
