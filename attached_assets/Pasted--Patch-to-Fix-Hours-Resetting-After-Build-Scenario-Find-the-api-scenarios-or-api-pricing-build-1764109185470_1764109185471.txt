@app.post("/api/scenarios")
def build_scenario():
    """
    Builds or refreshes a pricing scenario.
    - If reset=true -> rebuilds from Step 2 baseline.
    - Otherwise -> uses the working copy stored in SCENARIO_STORE to preserve Step 3 edits.
    """
    session_id = request.args.get("session_id")
    reset_flag = request.args.get("reset", "false").lower() == "true"

    state = get_session_state(session_id)
    baseline = state.get("baseline", {})
    scenario = state.get("scenario", {})

    # --- CASE 1: RESET (explicit rebuild from baseline) -------------------
    if reset_flag:
        new_scenario = deepcopy(baseline)
        _recompute_totals(new_scenario)
        state["scenario"] = new_scenario
        save_session_state(session_id, state)
        return jsonify({
            "scenario": new_scenario,
            "totals": state.get("totals", {}),
            "metadata": state.get("metadata", {})
        })

    # --- CASE 2: NORMAL BUILD (preserve Step 3 edits) ---------------------
    # If a working copy exists, reuse it — don't rebuild from Step 2
    if scenario:
        scenario = _recompute_totals(scenario)
        state["scenario"] = scenario
        save_session_state(session_id, state)
        return jsonify({
            "scenario": scenario,
            "totals": state.get("totals", {}),
            "metadata": state.get("metadata", {})
        })

    # --- CASE 3: FIRST BUILD (no working copy yet) ------------------------
    if not baseline:
        # fall back to Step 2 if baseline missing
        baseline = build_scenario_from_step2_baseline(session_id)
        state["baseline"] = deepcopy(baseline)

    scenario = deepcopy(baseline)
    _recompute_totals(scenario)
    state["scenario"] = scenario
    save_session_state(session_id, state)
    return jsonify({
        "scenario": scenario,
        "totals": state.get("totals", {}),
        "metadata": state.get("metadata", {})
    })
