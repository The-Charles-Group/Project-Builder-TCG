C) Front‑end download routine (to fix “nothing happens”)

When you click Export Scenario A, Export Scenario B, or Export A & B (one file), the browser must consume the response as a blob and apply the server’s filename from the Content‑Disposition header (CORS already exposes this).

Use this fetch pattern:

async function downloadFromEndpoint(url, payload) {
  const res = await fetch(url, {
    method: 'POST',
    headers: {'Content-Type': 'application/json'},
    body: JSON.stringify(payload),
  });
  if (!res.ok) {
    const text = await res.text();
    alert(`Export failed: ${text}`);
    return;
    }
  const blob = await res.blob();
  // Prefer server-suggested name, else fall back
  const disp = res.headers.get('Content-Disposition') || '';
  const match = /filename\*=UTF-8''([^;]+)|filename="?([^"]+)"?/.exec(disp);
  const suggested = decodeURIComponent((match?.[1] || match?.[2] || 'Workfront_Export.xlsx'));
  const urlObj = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = urlObj;
  a.download = suggested;
  document.body.appendChild(a);
  a.click();
  a.remove();
  URL.revokeObjectURL(urlObj);
}
