@app.post("/api/export_workbook_xml_abc")
def api_export_workbook_xml_abc(payload: ExportWorkbookXMLABCPayload):
    if not DB.loaded:
        DB.load()

    project = (payload.project_name or _upload_title_default() or f"Proposal {datetime.date.today().isoformat()}").strip()

    # Guards
    _assert_has_items(payload.scenario_a or {}, "Scenario A XML export")
    _assert_has_items(payload.scenario_b or {}, "Scenario B XML export")
    _assert_has_items(payload.scenario_c or {}, "Scenario C XML export")

    # Inflate
    scen_a = _inflate_components_if_missing(payload.scenario_a or {})
    scen_b = _inflate_components_if_missing(payload.scenario_b or {})
    scen_c = _inflate_components_if_missing(payload.scenario_c or {})

    # Build WBS
    dfA = build_wbs_dataframe_from_scenario(scen_a, project)
    dfB = build_wbs_dataframe_from_scenario(scen_b, project)
    dfC = build_wbs_dataframe_from_scenario(scen_c, project)
    dfA = _ensure_v3_ae_columns(dfA); dfB = _ensure_v3_ae_columns(dfB); dfC = _ensure_v3_ae_columns(dfC)

    base = _export_basename(project, "Scenarios A, B & C XML")
    tmp = []

    try:
        # A
        tmp_xa, out_xa = f"{base}_A_temp.xlsx", f"{base}_Scenario_A.xml"
        tmp += [tmp_xa, out_xa]
        with pd.ExcelWriter(tmp_xa, engine="openpyxl") as xw:
            dfA.to_excel(xw, sheet_name="Scenario A", index=False); _apply_number_formats(xw.sheets["Scenario A"], dfA)
        stats_a = convert_excel_to_mspdi(input_xlsx=tmp_xa, output_xml=out_xa, sheet_name="Scenario A",
                                         merge_identical_children=False, project_name=project)

        # B
        tmp_xb, out_xb = f"{base}_B_temp.xlsx", f"{base}_Scenario_B.xml"
        tmp += [tmp_xb, out_xb]
        with pd.ExcelWriter(tmp_xb, engine="openpyxl") as xw:
            dfB.to_excel(xw, sheet_name="Scenario B", index=False); _apply_number_formats(xw.sheets["Scenario B"], dfB)
        stats_b = convert_excel_to_mspdi(input_xlsx=tmp_xb, output_xml=out_xb, sheet_name="Scenario B",
                                         merge_identical_children=False, project_name=project)

        # C
        tmp_xc, out_xc = f"{base}_C_temp.xlsx", f"{base}_Scenario_C.xml"
        tmp += [tmp_xc, out_xc]
        with pd.ExcelWriter(tmp_xc, engine="openpyxl") as xw:
            dfC.to_excel(xw, sheet_name="Scenario C", index=False); _apply_number_formats(xw.sheets["Scenario C"], dfC)
        stats_c = convert_excel_to_mspdi(input_xlsx=tmp_xc, output_xml=out_xc, sheet_name="Scenario C",
                                         merge_identical_children=False, project_name=project)

        import zipfile, json, os
        zip_path = f"{base}.zip"
        with zipfile.ZipFile(zip_path, 'w') as z:
            z.write(out_xa, "Scenario_A.xml")
            z.write(out_xb, "Scenario_B.xml")
            z.write(out_xc, "Scenario_C.xml")
            z.writestr("export_stats.json", json.dumps({"A": stats_a, "B": stats_b, "C": stats_c}, indent=2))
        return FileResponse(zip_path, filename=os.path.basename(zip_path), media_type="application/zip")
    finally:
        for f in tmp:
            if os.path.exists(f): os.remove(f)
