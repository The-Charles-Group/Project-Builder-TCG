function renderTimeline(letter) {
  const scen = appState.scenarios?.[letter];
  const box = document.getElementById('timeline');
  if (!scen || !scen.items?.length) {
    box.innerHTML = '<em>Build scenarios in Step 3 to see a timeline.</em>';
    return;
  }
  // Simple, readable timeline: one row per deliverable with start–end
  const rows = scen.items.map(d => {
    const sched = d.schedule || [];
    const start = sched.length ? sched[0].start_date : '';
    const end   = sched.length ? sched[sched.length - 1].end_date : '';
    const days  = sched.reduce((n, s) => n + (s.duration_days || 0), 0);
    return `<tr>
      <td>${d.deliverable}</td>
      <td>${start}</td>
      <td>${end}</td>
      <td>${days}</td>
    </tr>`;
  }).join('');

  box.innerHTML = `
    <table class="table-compact">
      <thead><tr><th>Deliverable</th><th>Start</th><th>End</th><th>Duration (days)</th></tr></thead>
      <tbody>${rows}</tbody>
    </table>
    <p class="hint">Drag‑drop reordering will call <code>/api/reorder_timeline</code> in Stage 4.</p>
  `;
}


That uses the schedule the backend builds from Timeline_Params, scaling, slack, etc., which is already in your code.

3) Keep export logic only in Step 5

Wire the existing export endpoints to the Step 5 buttons:

async function downloadBlobFromPost(url, payload) {
  const res = await fetch(url, {method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify(payload)});
  if (!res.ok) { alert(`Export failed: ${res.status} — ${await res.text()}`); return; }
  const blob = await res.blob();
  const cd = res.headers.get('Content-Disposition') || '';
  const m = /filename="?([^"]+)"?/i.exec(cd);
  const filename = m ? m[1] : 'Workfront_Export.xlsx';
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob); a.download = filename; a.click();
  URL.revokeObjectURL(a.href);
}

function getExportPrefs() {
  return {
    file_format: (document.getElementById('export-format')?.value || 'xlsx'),
    add_timestamp: !!document.getElementById('export-add-ts')?.checked,
    project_name: document.getElementById('project-name')?.value || 'Proposal'
  };
}

async function exportScenario(letter) {
  const scen = appState.scenarios?.[letter];
  if (!scen || !scen.items?.length) { alert('Build scenarios first in Step 3.'); return; }
  const prefs = getExportPrefs();
  await downloadBlobFromPost('/api/export', {
    scenario: scen, project_name: prefs.project_name,
    file_format: prefs.file_format, scenario_label: `Scenario ${letter}`,
    add_timestamp: prefs.add_timestamp
  });
}
async function exportWorkbookAB() {
  const prefs = getExportPrefs();
  await downloadBlobFromPost('/api/export_workbook', {
    scenario_a: appState.scenarios?.A || {items:[]},
    scenario_b: appState.scenarios?.B || {items:[]},
    project_name: prefs.project_name,
    sheet_name_a: 'Scenario A', sheet_name_b: 'Scenario B',
    add_timestamp: prefs.add_timestamp
  });
}
