2) Replace /api/export with robust, named CSV/XLSX export

Builds a friendly filename from project_name, scenario label, and optional timestamp.

Writes CSV or XLSX (requires openpyxl for XLSX).

Sets both filename and filename* in Content-Disposition for maximum browser compatibility.

# requirements.txt: add openpyxl
# pip install openpyxl

import urllib.parse
import datetime
import re
import pandas as pd
from fastapi import HTTPException
from fastapi.responses import FileResponse
from pydantic import BaseModel

# --- helpers ---
def _safe_filename(s: str) -> str:
    s = (s or "Proposal").strip()
    s = re.sub(r'[\\/:*?"<>|]+', "-", s)   # sanitize
    s = re.sub(r"\s+", " ", s).strip()
    return s

class ExportPayload(BaseModel):
    scenario: dict
    project_name: str | None = None
    file_format: str | None = "csv"         # "csv" | "xlsx"
    scenario_label: str | None = None       # e.g., "Scenario A"
    add_timestamp: bool | None = False

@app.post("/api/export")
def api_export(payload: ExportPayload):
    if not DB.loaded:
        DB.load()

    scenario = payload.scenario or {}
    project_name = _safe_filename(payload.project_name or f"Proposal {datetime.date.today().isoformat()}")
    scen_label  = _safe_filename(payload.scenario_label or "")
    fmt = (payload.file_format or "csv").lower()
    if fmt not in ("csv", "xlsx"):
        raise HTTPException(400, "file_format must be 'csv' or 'xlsx'.")

    # Build rows (same structure as current export)
    rows = []
    for i, d in enumerate(scenario.get("items", []), start=1):
        wbs_id = f"1.{i}"
        rows.append({
            "Project_Name": project_name,
            "WBS_ID": wbs_id, "Parent_WBS_ID": "1",
            "Task_Name": d["deliverable"], "Deliverable": d["deliverable"],
            "Component": "", "Task": "", "Role": "", "Seniority": "",
            "Planned_Hours": d["total_hours"], "Start_Offset_Days": 0,
            "Duration_Days": sum(x["duration_days"] for x in d["schedule"]),
            "Dependencies": "", "Assignee_External_ID": "",
            "Notes": f"{d['complexity']} / {d['tier']}"
        })
        for j, t in enumerate(d["schedule"], start=1):
            rows.append({
                "Project_Name": project_name,
                "WBS_ID": f"{wbs_id}.{j}", "Parent_WBS_ID": wbs_id,
                "Task_Name": t["task_group"], "Deliverable": d["deliverable"],
                "Component": "", "Task": t["task_group"], "Role": "", "Seniority": "",
                "Planned_Hours": "", "Start_Offset_Days": "", "Duration_Days": t["duration_days"],
                "Dependencies": "", "Assignee_External_ID": "", "Notes": ""
            })

    df = pd.DataFrame(rows)

    # Friendly base name
    parts = [p for p in [project_name, scen_label, "Workfront Export"] if p]
    base = " - ".join(parts)
    if payload.add_timestamp:
        base += " - " + datetime.datetime.now().strftime("%Y%m%d-%H%M")

    if fmt == "csv":
        out_name = f"{base}.csv"
        df.to_csv(out_name, index=False)
        resp = FileResponse(out_name, filename=out_name, media_type="text/csv")
    else:
        out_name = f"{base}.xlsx"
        try:
            df.to_excel(out_name, index=False, engine="openpyxl")
        except Exception as ex:
            raise HTTPException(400, "XLSX export requires 'openpyxl'. Add it to requirements.txt.") from ex
        resp = FileResponse(
            out_name,
            filename=out_name,
            media_type="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
        )

    # Add RFC 5987 filename* for better cross‑browser support
    quoted = urllib.parse.quote(out_name)
    resp.headers["Content-Disposition"] = f'attachment; filename="{out_name}"; filename*=UTF-8\'\'{quoted}'
    return resp


In your current v2, /api/export hard‑codes workfront_export.csv, so if the frontend can’t read or overrides the header, you’ll keep seeing generic names. The patch above replaces that behavior.

Frontend patch (always produce a friendly filename)

Why: Even with perfect headers, fetch(...).blob() + object URLs do not automatically use the server filename. We must set a.download ourselves. If the header isn’t readable, we’ll compute the name from the UI (project name, scenario, timestamp).

Replace your export handler with this:

<script>
  function fallbackName(letter, fmt) {
    const projectName = (document.getElementById('projectName')?.value || 'Proposal').trim();
    const ts = document.getElementById('exportTimestamp')?.checked
      ? (' - ' + new Date().toISOString().replace(/[-:T]/g,'').slice(0,12))
      : '';
    return `${projectName} - Scenario ${letter} - Workfront Export${ts}.${fmt}`;
  }

  function parseFilenameFromContentDisposition(cd) {
    if (!cd) return null;
    // filename*=UTF-8''<urlencoded>
    const star = cd.match(/filename\*\s*=\s*([^;]+)/i);
    if (star) {
      const v = star[1].trim();
      const parts = v.split("''");
      const raw = parts.length === 2 ? parts[1] : v;
      try { return decodeURIComponent(raw.replace(/^"(.*)"$/, '$1')); } catch { return null; }
    }
    // filename="<quoted>" or filename=bare
    const simple = cd.match(/filename\s*=\s*"([^"]+)"|filename\s*=\s*([^;]+)/i);
    if (simple) return (simple[1] || simple[2]).trim().replace(/^"(.*)"$/, '$1');
    return null;
  }

  async function exportScenario(letter) {
    if (!latestScenarios || !latestScenarios[letter]) return;

    const fmt = document.getElementById('exportFormat').value || 'csv';
    const projectName = document.getElementById('projectName').value || '';
    const addTs = document.getElementById('exportTimestamp').checked;

    const res = await fetch('/api/export', {
      method: 'POST',
      headers: {'Content-Type': 'application/json'},
      body: JSON.stringify({
        scenario: latestScenarios[letter],
        project_name: projectName,
        file_format: fmt,
        scenario_label: `Scenario ${letter}`,
        add_timestamp: addTs
      })
    });

    const blob = await res.blob();
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;

    // Prefer server-suggested filename; otherwise build from UI
    const cd = res.headers.get('content-disposition');
    a.download = parseFilenameFromContentDisposition(cd) || fallbackName(letter, fmt);

    a.click();
    URL.revokeObjectURL(url);
  }

  document.getElementById('btnExportA').onclick = () => exportScenario('A');
  document.getElementById('btnExportB').onclick = () => exportScenario('B');
</script>
