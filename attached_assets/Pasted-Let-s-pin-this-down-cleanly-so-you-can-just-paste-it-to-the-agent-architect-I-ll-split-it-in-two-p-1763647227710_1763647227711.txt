Helper to get “parent” WBS
def parent_wbs(wbs: str) -> str | None:
    """
    If wbs == '1.2.3.4.5.6' -> '1.2.3.4.5'
    If wbs == '1'          -> None
    """
    if not wbs:
        return None
    parts = wbs.split(".")
    if len(parts) <= 1:
        return None
    return ".".join(parts[:-1])

Core: normalize predecessor to L5 and emit edges
def build_wbs_dependencies(rows: list[dict]) -> list[tuple[int, int, str, int]]:
    """
    Returns list of (pred_uid, succ_uid, dep_type, lag)
    dep_type: 'FS' for Finish-to-Start (what you use today)
    lag: 0 for now
    """
    edges: list[tuple[int, int, str, int]] = []

    # --- build indexes as above ---
    wbs_to_uid, uid_to_wbs, uid_to_outline = build_indexes(rows)

    for row in rows:
        succ_uid = row["UID"]
        succ_outline = row["OutlineLevel"]

        # Only put predecessors on L5+ summary rows
        if succ_outline > 5:
            continue

        deps_str = (row.get("Dependencies") or "").strip()
        if not deps_str:
            continue

        for raw_dep in deps_str.split(","):
            dep_wbs = raw_dep.strip()
            if not dep_wbs:
                continue

            # 1) direct lookup WBS -> UID
            pred_uid = wbs_to_uid.get(dep_wbs)
            if pred_uid is None:
                log_warn(f"[WBS-DEPS] Orphan dependency {dep_wbs} on UID {succ_uid}")
                continue

            # 2) If that UID is deeper than level 5, walk up until <=5
            outline = uid_to_outline.get(pred_uid, 99)
            while outline > 5:
                pw = parent_wbs(uid_to_wbs[pred_uid])
                if not pw:
                    break
                parent_uid = wbs_to_uid.get(pw)
                if parent_uid is None:
                    break
                pred_uid = parent_uid
                outline = uid_to_outline.get(pred_uid, 99)

            # If we still don't have a ≤5 task, skip
            if outline > 5:
                log_warn(
                    f"[WBS-DEPS] Could not find ≤L5 parent for {dep_wbs} "
                    f"(UID {pred_uid}) used by UID {succ_uid}"
                )
                continue

            # Avoid self-deps after normalization
            if pred_uid == succ_uid:
                continue

            edges.append((pred_uid, succ_uid, "FS", 0))

    return edges

Pseudocode: summarize roles per L5 component
from collections import defaultdict

@dataclass
class RoleSummary:
    total_hours: float = 0.0   # sum of L6 role hours under the component
    # You can also store rate, cost, etc., if needed

def build_component_role_summary(rows: list[dict]) -> dict[tuple[int, str], RoleSummary]:
    """
    Returns a map:
      (l5_uid, role_name_or_id) -> RoleSummary
    Used as the data source for future L5 multi-assignments.
    """
    wbs_to_uid, uid_to_wbs, uid_to_outline = build_indexes(rows)

    summary: dict[tuple[int, str], RoleSummary] = defaultdict(RoleSummary)

    for row in rows:
        uid = row["UID"]
        outline = row["OutlineLevel"]

        # We only care about true leaf / role rows here (L6)
        if outline <= 5:
            continue

        role = row.get("RoleName") or row.get("RoleID")
        hours = float(row.get("PlanHrs", 0) or 0)

        # Find this task's L5 component ancestor
        wbs = uid_to_wbs[uid]
        comp_uid = uid  # fallback, but we want L5
        comp_outline = outline

        while comp_outline > 5:
            pw = parent_wbs(wbs)
            if not pw:
                break
            parent_uid = wbs_to_uid.get(pw)
            if parent_uid is None:
                break
            comp_uid = parent_uid
            wbs = pw
            comp_outline = uid_to_outline.get(comp_uid, comp_outline)

        # We *only* summarize under real L5 components
        if comp_outline != 5:
            continue

        key = (comp_uid, role)
        summary[key].total_hours += hours

    return summary
