What a Sheets mirror is (plain English)

Your Scenario = the single project “universe” a user edits across Steps 2–4: Deliverables → Components → L2 Tasks → Pricing → Timeline → Export.

A Sheets mirror = one Google Spreadsheet per scenario with predictable tabs:

Meta | Deliverables | Components | Tasks_L2 | Pricing | Scenario_Pricing | Timeline | Lookups


The app is the cockpit; Sheets is the ledger. Every meaningful edit writes to the sheet; the app can also re‑hydrate from the sheet. Benefits:

Durable: if the preview dies or the SSE disconnects, the scenario persists.

Auditable: humans can read the tabs; easy to diff and debug.

QA‑able: the test harness below validates the contract without UI clicks.

In your screenshots, Step 2’s AI‑Suggested list with departments/confidence/risks and the three selectors is the gold standard we restore. The counters like 31 / 50 / 208 and the selection panels are shown in your PDFs, which this plan targets exactly.

from __future__ import annotations
from abc import ABC, abstractmethod
from typing import Dict, Any

class StorageDriver(ABC):
    """Abstract interface for scenario persistence."""
    @abstractmethod
    def create(self, name: str, meta: Dict[str, Any]) -> Dict[str, Any]: ...
    @abstractmethod
    def save(self, scenario: Dict[str, Any]) -> None: ...
    @abstractmethod
    def load(self, scenario_id: str) -> Dict[str, Any]: ...

Concrete Python — Google Sheets driver (works with Google API or Replit connector)
# code/sheets_driver.py
from __future__ import annotations
from typing import Dict, Any, List
import uuid

TABS = ["Meta","Deliverables","Components","Tasks_L2","Pricing",
        "Scenario_Pricing","Timeline","Lookups"]

HEADERS = {
    "Meta": ["scenario_id","name","created_at","project_start","pricing_mode","rate_band",
             "blended_rate","complexity","volume_tier","sheet_id"],
    "Deliverables": ["deliverable_id","dept","name","selected","confidence","risk_notes"],
    "Components": ["component_id","deliverable_id","name","selected"],
    "Tasks_L2": ["task_id","component_id","name","selected","hours_default"],
    "Pricing": ["rate_band","role","rate"],
    "Scenario_Pricing": ["deliverable_id","component_id","task_id","months","hours","rate","cost"],
    "Timeline": ["task_id","wbs","start","end","duration","deps"],
    "Lookups": ["key","value"],
}

class SheetsAdapter:
    def create_spreadsheet(self, title: str) -> str: ...
    def add_sheet_with_headers(self, spreadsheet_id: str, sheet_title: str, headers: List[str]) -> None: ...
    def batch_values_update(self, spreadsheet_id: str, data: List[Dict[str, Any]], value_input_option: str = "USER_ENTERED") -> None: ...
    def get_values(self, spreadsheet_id: str, range_a1: str) -> List[List[Any]]: ...

# --- Google API adapter (service accounts) ---
class GoogleApiAdapter(SheetsAdapter):
    def __init__(self, sheets_service, drive_service):
        self.sheets = sheets_service
        self.drive = drive_service
    def create_spreadsheet(self, title: str) -> str:
        body = {"mimeType":"application/vnd.google-apps.spreadsheet","name":title}
        file = self.drive.files().create(body=body, fields="id").execute()
        return file["id"]
    def add_sheet_with_headers(self, spreadsheet_id: str, sheet_title: str, headers: List[str]) -> None:
        self.sheets.spreadsheets().batchUpdate(
            spreadsheetId=spreadsheet_id,
            body={"requests":[{"addSheet":{"properties":{"title":sheet_title}}}]}
        ).execute()
        self.batch_values_update(spreadsheet_id, [{
            "range": f"{sheet_title}!A1:{chr(64+len(headers))}1", "values":[headers]
        }])
    def batch_values_update(self, spreadsheet_id: str, data: List[Dict[str, Any]], value_input_option: str="USER_ENTERED") -> None:
        self.sheets.spreadsheets().values().batchUpdate(
            spreadsheetId=spreadsheet_id,
            body={"valueInputOption":value_input_option,"data":data}
        ).execute()
    def get_values(self, spreadsheet_id: str, range_a1: str) -> List[List[Any]]:
        return self.sheets.spreadsheets().values().get(
            spreadsheetId=spreadsheet_id, range=range_a1
        ).execute().get("values", [])

# --- Replit first‑party connector adapter (if you go that route) ---
class ReplitConnectorAdapter(SheetsAdapter):
    def __init__(self, connector): self.conn = connector
    def create_spreadsheet(self, title: str) -> str: return self.conn.create_spreadsheet(title)
    def add_sheet_with_headers(self, spreadsheet_id: str, sheet_title: str, headers: List[str]) -> None:
        self.conn.add_sheet(spreadsheet_id, sheet_title)
        self.batch_values_update(spreadsheet_id, [{
            "range": f"{sheet_title}!A1:{chr(64+len(headers))}1","values":[headers]
        }])
    def batch_values_update(self, spreadsheet_id: str, data: List[Dict[str, Any]], value_input_option: str="USER_ENTERED") -> None:
        self.conn.batch_values_update(spreadsheet_id, data, value_input_option=value_input_option)
    def get_values(self, spreadsheet_id: str, range_a1: str) -> List[List[Any]]:
        return self.conn.get_values(spreadsheet_id, range_a1)

def _as_num(v):
    try: return float(v) if v not in (None, "") else None
    except: return None

class SheetsDriver:
    def __init__(self, adapter: SheetsAdapter):
        self.adapter = adapter

    def create(self, name: str, meta: Dict[str, Any]) -> Dict[str, Any]:
        spreadsheet_title = f"Project Builder – {name}"
        sheet_id = self.adapter.create_spreadsheet(spreadsheet_title)
        for tab in TABS:
            self.adapter.add_sheet_with_headers(sheet_id, tab, HEADERS[tab])
        scenario_id = str(uuid.uuid4())
        meta_row = [scenario_id, name, meta.get("created_at"), meta.get("project_start"),
                    meta.get("pricing_mode"), meta.get("rate_band"), meta.get("blended_rate"),
                    meta.get("complexity"), meta.get("volume_tier"), sheet_id]
        self.adapter.batch_values_update(sheet_id, [{"range":"Meta!A2:J2","values":[meta_row]}])
        return {"scenario_id": scenario_id, "sheet_id": sheet_id}

    def save(self, scenario: Dict[str, Any]) -> None:
        sheet_id = scenario["meta"].get("sheet_id") or scenario.get("sheet_id")
        if not sheet_id:
            raise ValueError("Scenario missing 'sheet_id' in meta")
        meta = scenario.get("meta", {})
        sel = scenario.get("selection", {})
        pricing = scenario.get("pricing", {})
        timeline = scenario.get("timeline", {})

        deliv_rows = [[did, d.get("dept"), d.get("name"), bool(d.get("selected")),
                       d.get("confidence"), d.get("risk")]
                      for did, d in (sel.get("deliverables") or {}).items()]

        comp_rows = []
        for cid, flag in (sel.get("components") or {}).items():
            parts = cid.split("::", 1)
            deliverable_id = parts[0] if len(parts)>1 else ""
            comp_rows.append([cid, deliverable_id, parts[1] if len(parts)>1 else cid, bool(flag)])

        task_rows = []
        for tid, flag in (sel.get("tasks_l2") or {}).items():
            parts = tid.split("::")
            component_id = "::".join(parts[0:2]) if len(parts)>=2 else ""
            task_rows.append([tid, component_id, parts[-1], bool(flag), None])

        pricing_rows = [[pricing.get("rate_band"), role, rate]
                        for role, rate in (pricing.get("roles") or {}).items()]

        scen_rows = [[it.get("deliverable_id"), it.get("component_id"), it.get("task_id"),
                      it.get("months"), it.get("hours"), it.get("rate"), it.get("cost")]
                     for it in (pricing.get("line_items") or [])]

        t_rows = [[it.get("id"), it.get("wbs"), it.get("start"), it.get("end"),
                   it.get("duration"), ",".join(it.get("deps", []))]
                  for it in (timeline.get("items") or [])]

        def vr(tab, n): return f"{tab}!A2:{chr(64+n)}"

        data=[]
        if deliv_rows:  data.append({"range": vr("Deliverables", 6), "values": deliv_rows})
        if comp_rows:   data.append({"range": vr("Components",   4), "values": comp_rows})
        if task_rows:   data.append({"range": vr("Tasks_L2",     5), "values": task_rows})
        if pricing_rows:data.append({"range": vr("Pricing",      3), "values": pricing_rows})
        if scen_rows:   data.append({"range": vr("Scenario_Pricing", 7), "values": scen_rows})
        if t_rows:      data.append({"range": vr("Timeline",     6), "values": t_rows})

        meta_row = [scenario.get("scenario_id"), meta.get("name"), meta.get("created_at"),
                    timeline.get("project_start"), pricing.get("pricing_mode"),
                    pricing.get("rate_band"), pricing.get("blended_rate"),
                    pricing.get("complexity"), pricing.get("volume_tier"), sheet_id]
        data.append({"range":"Meta!A2:J2","values":[meta_row]})

        self.adapter.batch_values_update(sheet_id, data)

    def load(self, scenario_id: str, sheet_id: str) -> Dict[str, Any]:
        def get(range_a1): return self.adapter.get_values(sheet_id, range_a1)
        meta_vals = get("Meta!A1:J1000")
        hdr = meta_vals[0] if meta_vals else []
        row = next((r for r in meta_vals[1:] if r and r[0]==scenario_id), None)
        if not row:
            raise ValueError(f"Scenario {scenario_id} not found")
        meta = dict(zip(hdr, row))

        def table(tab):
            vals = get(f"{tab}!A1:Z100000")
            if not vals: return []
            h = vals[0]
            return [dict(zip(h, r + [None]*(len(h)-len(r)))) for r in vals[1:]]

        deliverables, components, tasks_l2 = table("Deliverables"), table("Components"), table("Tasks_L2")
        pricing_rows, scen_pricing, timeline_rows = table("Pricing"), table("Scenario_Pricing"), table("Timeline")

        sel_deliv = {
            d.get("deliverable_id"): {
                "selected": str(d.get("selected")).lower()=="true",
                "confidence": float(d.get("confidence") or 0),
                "dept": d.get("dept"),
                "name": d.get("name")
            } for d in deliverables if d.get("deliverable_id")
        }
        sel_comp  = {c.get("component_id"): str(c.get("selected")).lower()=="true" for c in components if c.get("component_id")}
        sel_tasks = {t.get("task_id"): str(t.get("selected")).lower()=="true" for t in tasks_l2 if t.get("task_id")}

        roles = {}
        for pr in pricing_rows:
            if pr.get("role"):
                roles[pr["role"]] = float(pr.get("rate") or 0)

        def as_num(v):
            try: return float(v) if v not in (None, "") else None
            except: return None

        line_items=[{
            "deliverable_id": it.get("deliverable_id"),
            "component_id":   it.get("component_id"),
            "task_id":        it.get("task_id"),
            "months":         as_num(it.get("months")),
            "hours":          as_num(it.get("hours")),
            "rate":           as_num(it.get("rate")),
            "cost":           as_num(it.get("cost"))
        } for it in scen_pricing]

        timeline_items=[{
            "id": r.get("task_id"), "wbs": r.get("wbs"),
            "start": r.get("start"), "end": r.get("end"),
            "duration": r.get("duration"),
            "deps": (r.get("deps") or "").split(",") if r.get("deps") else []
        } for r in timeline_rows]

        return {
            "scenario_id": scenario_id,
            "meta": {"name": meta.get("name"), "created_at": meta.get("created_at"), "sheet_id": meta.get("sheet_id")},
            "selection": {"deliverables": sel_deliv, "components": sel_comp, "tasks_l2": sel_tasks},
            "pricing": {"pricing_mode": meta.get("pricing_mode"), "rate_band": meta.get("rate_band"),
                        "blended_rate": as_num(meta.get("blended_rate")),
                        "complexity": meta.get("complexity"), "volume_tier": meta.get("volume_tier"),
                        "roles": roles, "line_items": line_items},
            "timeline": {"project_start": meta.get("project_start"), "items": timeline_items},
        }

# code/test_harness/test_sheets_driver.py
import os, json
from sheets_driver import SheetsDriver, GoogleApiAdapter
from google.oauth2.service_account import Credentials
from googleapiclient.discovery import build

def build_adapter_from_env():
    creds_json = os.environ.get("GOOGLE_CREDENTIALS_JSON")
    assert creds_json, "Set GOOGLE_CREDENTIALS_JSON to a service account JSON string"
    creds = Credentials.from_service_account_info(json.loads(creds_json), scopes=[
        "https://www.googleapis.com/auth/drive",
        "https://www.googleapis.com/auth/spreadsheets"
    ])
    sheets = build("sheets","v4",credentials=creds)
    drive  = build("drive","v3",credentials=creds)
    return GoogleApiAdapter(sheets, drive)

def main():
    adapter = build_adapter_from_env()
    driver = SheetsDriver(adapter)
    record = driver.create("QA Harness Test", {

# api/routes_scenarios.py
from fastapi import APIRouter
from pydantic import BaseModel
from typing import Dict, Any
from sheets_driver import SheetsDriver, GoogleApiAdapter
# ...init adapter as in the harness...

router = APIRouter(prefix="/api/scenario", tags=["scenario"])

class Scenario(BaseModel):
    scenario_id: str
    meta: Dict[str, Any]
    selection: Dict[str, Any]
    pricing: Dict[str, Any]
    timeline: Dict[str, Any]

@router.post("/create")
def create_scenario(payload: Dict[str, Any]):
    # name + meta inputs expected
    record = driver.create(payload["name"], payload.get("meta", {}))
    return record

@router.post("/save")
def save_scenario(scenario: Scenario):
    driver.save(scenario.dict())
    return {"ok": True}

@router.get("/load/{sheet_id}/{scenario_id}")
def load_scenario(sheet_id: str, scenario_id: str):
    data = driver.load(scenario_id, sheet_id=sheet_id)
    return data

from future import annotations
from typing import Dict, Any, List
import uuid

TABS = ["Meta","Deliverables","Components","Tasks_L2","Pricing","Scenario_Pricing","Timeline","Lookups"]

HEADERS = {
"Meta": ["scenario_id","name","created_at","project_start","pricing_mode","rate_band","blended_rate","complexity","volume_tier","sheet_id"],
"Deliverables": ["deliverable_id","dept","name","selected","confidence","risk_notes"],
"Components": ["component_id","deliverable_id","name","selected"],
"Tasks_L2": ["task_id","component_id","name","selected","hours_default"],
"Pricing": ["rate_band","role","rate"],
"Scenario_Pricing": ["deliverable_id","component_id","task_id","months","hours","rate","cost"],
"Timeline": ["task_id","wbs","start","end","duration","deps"],
"Lookups": ["key","value"],
}

class SheetsAdapter:
def create_spreadsheet(self, title: str) -> str: ...
def add_sheet_with_headers(self, spreadsheet_id: str, sheet_title: str, headers: List[str]) -> None: ...
def batch_values_update(self, spreadsheet_id: str, data: List[Dict[str, Any]], value_input_option: str = "USER_ENTERED") -> None: ...
def get_values(self, spreadsheet_id: str, range_a1: str) -> List[List[Any]]: ...

class GoogleApiAdapter(SheetsAdapter):
def init(self, sheets_service, drive_service):
self.sheets = sheets_service
self.drive = drive_service

def create_spreadsheet(self, title: str) -> str:
    body = {"mimeType":"application/vnd.google-apps.spreadsheet","name":title}
    file = self.drive.files().create(body=body, fields="id").execute()
    return file["id"]

def add_sheet_with_headers(self, spreadsheet_id: str, sheet_title: str, headers: List[str]) -> None:
    self.sheets.spreadsheets().batchUpdate(
        spreadsheetId=spreadsheet_id,
        body={"requests":[{"addSheet":{"properties":{"title":sheet_title}}}]}
    ).execute()
    self.batch_values_update(spreadsheet_id, [{
        "range": f"{sheet_title}!A1:{chr(64+len(headers))}1",
        "values":[headers]
    }])

def batch_values_update(self, spreadsheet_id: str, data: List[Dict[str, Any]], value_input_option: str="USER_ENTERED") -> None:
    self.sheets.spreadsheets().values().batchUpdate(
        spreadsheetId=spreadsheet_id,
        body={"valueInputOption":value_input_option,"data":data}
    ).execute()

def get_values(self, spreadsheet_id: str, range_a1: str) -> List[List[Any]]:
    return self.sheets.spreadsheets().values().get(
        spreadsheetId=spreadsheet_id, range=range_a1
    ).execute().get("values", [])


class ReplitConnectorAdapter(SheetsAdapter):
def init(self, connector): self.conn = connector
def create_spreadsheet(self, title: str) -> str: return self.conn.create_spreadsheet(title)
def add_sheet_with_headers(self, spreadsheet_id: str, sheet_title: str, headers: List[str]) -> None:

try: return float(v) if v not in (None, "") else None
except: return None

class SheetsDriver:
def init(self, adapter: SheetsAdapter):
self.adapter = adapter

def create(self, name: str, meta: Dict[str, Any]) -> Dict[str, Any]:
    spreadsheet_title = f"Project Builder – {name}"
    sheet_id = self.adapter.create_spreadsheet(spreadsheet_title)
    for tab in TABS:
        self.adapter.add_sheet_with_headers(sheet_id, tab, HEADERS[tab])
    scenario_id = str(uuid.uuid4())
    meta_row = [scenario_id, name, meta.get("created_at"), meta.get("project_start"),
                meta.get("pricing_mode"), meta.get("rate_band"), meta.get("blended_rate"),
                meta.get("complexity"), meta.get("volume_tier"), sheet_id]
    self.adapter.batch_values_update(sheet_id, [{"range":"Meta!A2:J2","values":[meta_row]}])
    return {"scenario_id": scenario_id, "sheet_id": sheet_id}

def save(self, scenario: Dict[str, Any]) -> None:
    sheet_id = scenario["meta"].get("sheet_id") or scenario.get("sheet_id")
    if not sheet_id:
        raise ValueError("Scenario missing 'sheet_id' in meta")
    meta = scenario.get("meta", {})
    sel = scenario.get("selection", {})
    pricing = scenario.get("pricing", {})
    timeline = scenario.get("timeline", {})

    deliv_rows = [[did, d.get("dept"), d.get("name"), bool(d.get("selected")),
                   d.get("confidence"), d.get("risk")]
                  for did, d in (sel.get("deliverables") or {}).items()]

    comp_rows = []
    for cid, flag in (sel.get("components") or {}).items():
        parts = cid.split("::", 1)
        deliverable_id = parts[0] if len(parts)>1 else ""
        comp_rows.append([cid, deliverable_id, parts[1] if len(parts)>1 else cid, bool(flag)])

    task_rows = []
    for tid, flag in (sel.get("tasks_l2") or {}).items():
        parts = tid.split("::")
        component_id = "::".join(parts[0:2]) if len(parts)>=2 else ""
        task_rows.append([tid, component_id, parts[-1], bool(flag), None])

    pricing_rows = [[pricing.get("rate_band"), role, rate]
                    for role, rate in (pricing.get("roles") or {}).items()]

    scen_rows = [[it.get("deliverable_id"), it.get("component_id"), it.get("task_id"),
                  it.get("months"), it.get("hours"), it.get("rate"), it.get("cost")]
                 for it in (pricing.get("line_items") or [])]

    t_rows = [[it.get("id"), it.get("wbs"), it.get("start"), it.get("end"),
               it.get("duration"), ",".join(it.get("deps", []))]
              for it in (timeline.get("items") or [])]

    def vr(tab, n): return f"{tab}!A2:{chr(64+n)}"

    data=[]
    if deliv_rows:  data.append({"range": vr("Deliverables", 6), "values": deliv_rows})
    if comp_rows:   data.append({"range": vr("Components",   4), "values": comp_rows})
    if task_rows:   data.append({"range": vr("Tasks_L2",     5), "values": task_rows})
    if pricing_rows:data.append({"range": vr("Pricing",      3), "values": pricing_rows})
    if scen_rows:   data.append({"range": vr("Scenario_Pricing", 7), "values": scen_rows})
    if t_rows:      data.append({"range": vr("Timeline",     6), "values": t_rows})

    meta_row = [scenario.get("scenario_id"), meta.get("name"), meta.get("created_at"),
                timeline.get("project_start"), pricing.get("pricing_mode"),
                pricing.get("rate_band"), pricing.get("blended_rate"),
                pricing.get("complexity"), pricing.get("volume_tier"), sheet_id]
    data.append({"range":"Meta!A2:J2","values":[meta_row]})

    self.adapter.batch_values_update(sheet_id, data)

def load(self, scenario_id: str, sheet_id: str) -> Dict[str, Any]:
    def get(range_a1): return self.adapter.get_values(sheet_id, range_a1)
    meta_vals = get("Meta!A1:J1000")
    hdr = meta_vals[0] if meta_vals else []
    row = next((r for r in meta_vals[1:] if r and r[0]==scenario_id), None)
    if not row:
        raise ValueError(f"Scenario {scenario_id} not found")
    meta = dict(zip(hdr, row))

    def table(tab):
        vals = get(f"{tab}!A1:Z100000")
        if not vals: return []
        h = vals[0]
        return [dict(zip(h, r + [None]*(len(h)-len(r)))) for r in vals[1:]]

    deliverables, components, tasks_l2 = table("Deliverables"), table("Components"), table("Tasks_L2")
    pricing_rows, scen_pricing, timeline_rows = table("Pricing"), table("Scenario_Pricing"), table("Timeline")

    sel_deliv = {
        d.get("deliverable_id"): {
            "selected": str(d.get("selected")).lower()=="true",
            "confidence": float(d.get("confidence") or 0),
            "dept": d.get("dept"),
            "name": d.get("name")
        } for d in deliverables if d.get("deliverable_id")
    }
    sel_comp  = {c.get("component_id"): str(c.get("selected")).lower()=="true" for c in components if c.get("component_id")}
    sel_tasks = {t.get("task_id"): str(t.get("selected")).lower()=="true" for t in tasks_l2 if t.get("task_id")}

    roles = {}
    for pr in pricing_rows:
        if pr.get("role"):
            roles[pr["role"]] = float(pr.get("rate") or 0)

    def as_num(v):
        try: return float(v) if v not in (None, "") else None
        except: return None

    line_items=[{
        "deliverable_id": it.get("deliverable_id"),
        "component_id":   it.get("component_id"),
        "task_id":        it.get("task_id"),
        "months":         as_num(it.get("months")),
        "hours":          as_num(it.get("hours")),
        "rate":           as_num(it.get("rate")),
        "cost":           as_num(it.get("cost"))
    } for it in scen_pricing]

    timeline_items=[{
        "id": r.get("task_id"), "wbs": r.get("wbs"),
        "start": r.get("start"), "end": r.get("end"),
        "duration": r.get("duration"),
        "deps": (r.get("deps") or "").split(",") if r.get("deps") else []
    } for r in timeline_rows]

    return {
        "scenario_id": scenario_id,
        "meta": {"name": meta.get("name"), "created_at": meta.get("created_at"), "sheet_id": meta.get("sheet_id")},
        "selection": {"deliverables": sel_deliv, "components": sel_comp, "tasks_l2": sel_tasks},
        "pricing": {"pricing_mode": meta.get("pricing_mode"), "rate_band": meta.get("rate_band"),
                    "blended_rate": as_num(meta.get("blended_rate")),
                    "complexity": meta.get("complexity"), "volume_tier": meta.get("volume_tier"),
                    "roles": roles, "line_items": line_items},
        "timeline": {"project_start": meta.get("project_start"), "items": timeline_items},
    }

from fastapi import FastAPI
from pydantic import BaseModel
from typing import Dict, Any
import os, json
from google.oauth2.service_account import Credentials
from googleapiclient.discovery import build
from sheets_driver import SheetsDriver, GoogleApiAdapter

app = FastAPI(title="Project Builder Scenario API")

def _build_adapter():
creds_json = os.environ.get("GOOGLE_CREDENTIALS_JSON")
if not creds_json:
raise RuntimeError("Set GOOGLE_CREDENTIALS_JSON to a service account JSON string")
creds = Credentials.from_service_account_info(json.loads(creds_json), scopes=[
"https://www.googleapis.com/auth/drive
",
"https://www.googleapis.com/auth/spreadsheets
"
])
sheets = build("sheets","v4",credentials=creds)
drive = build("drive","v3",credentials=creds)
return GoogleApiAdapter(sheets, drive)

adapter = None
driver = None

@app.on_event("startup")
def _init():
global adapter, driver
adapter = _build_adapter()
driver = SheetsDriver(adapter)

class Scenario(BaseModel):
scenario_id: str
meta: Dict[str, Any]
selection: Dict[str, Any] = {}
pricing: Dict[str, Any] = {}
timeline: Dict[str, Any] = {}

@app.post("/api/scenario/create")
def create_scenario(payload: Dict[str, Any]):
name = payload.get("name") or "Untitled"
meta = payload.get("meta") or {}
record = driver.create(name, meta)
return record

@app.post("/api/scenario/save")
def save_scenario(scenario: Scenario):
driver.save(scenario.dict())
return {"ok": True}

@app.get("/api/scenario/load/{sheet_id}/{scenario_id}")
def load_scenario(sheet_id: str, scenario_id: str):
data = driver.load(scenario_id, sheet_id=sheet_id)
return data
