1) Add a helper to find v3 by env var or common paths
# Add near AgencyDB class (top-level helper)
def _find_v3_path() -> str | None:
    candidates = [
        os.getenv("APB_V3_DB"),                                     # preferred: Replit Secret
        "Replit_App_DB_READABLE_FullRows_v3.xlsx",                  # project root
        "data/Replit_App_DB_READABLE_FullRows_v3.xlsx",             # ./data
        "static/data/Replit_App_DB_READABLE_FullRows_v3.xlsx",      # ./static/data
    ]
    for p in candidates:
        if p and os.path.exists(p):
            return p
    return None

2) In AgencyDB.load(), replace the hard‑coded v3 lookup with:
# after v4 sheets are loaded
v3_path = _find_v3_path()
if v3_path:
    try:
        self.drivers_v3 = pd.read_excel(v3_path, sheet_name="Drivers")
    except Exception:
        self.drivers_v3 = None
    try:
        self.v3_all_rows = pd.read_excel(v3_path, sheet_name="All_Task_Rows")
    except Exception:
        self.v3_all_rows = None
    print(f"[DB] v3 loaded from: {v3_path}")
else:
    self.drivers_v3 = None
    self.v3_all_rows = None
    print("[DB] v3 not found (Drivers/All_Task_Rows unavailable)")

3) Add quick diagnostics endpoints
@app.get("/api/db/status")
def db_status():
    if not DB.loaded: DB.load()
    def _ok(df): return (df is not None) and (not getattr(df, "empty", True))
    return {
        "loaded": DB.loaded,
        "v4_src": DB.src,
        "has_v3": _ok(DB.v3_all_rows),
        "v3_sheets": {
            "drivers": _ok(DB.drivers_v3),
            "all_rows": _ok(DB.v3_all_rows),
        }
    }

@app.post("/api/db/reload")
def db_reload():
    DB.loaded = False
    DB.load()
    return {"ok": True, "status": db_status()}


Now you can upload/rename the file and POST /api/db/reload without restarting the repl.
