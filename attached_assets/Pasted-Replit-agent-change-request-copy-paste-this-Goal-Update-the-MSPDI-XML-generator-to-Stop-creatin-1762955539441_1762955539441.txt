def clean_for_workfront(tasks):
    """
    tasks: iterable of dicts with at least:
      deliverable_code, component_name, name, is_summary (bool),
      start, finish, duration_minutes, work_hours, revenue,
      predecessors=[(pred_uid, type)], children=[...]
    Returns: cleaned, de-duped tree with correct summaries and links.
    """
    # 1) strip wrappers/suffixes
    def sanitize(n):
        n = n.strip()
        n = re.sub(r'\s*-\s*complete$', '', n, flags=re.I)
        return n

    for t in tasks:
        t["name"] = sanitize(t["name"])

    # 2) drop phase/client-approval pseudo-rows unless they are true leaf tasks
    def is_wrapper(n):
        return bool(re.match(r'^(phase\s*\d+|client\s*approval)$', n, re.I))
    tasks = [t for t in tasks if not (t["is_summary"] and is_wrapper(t["name"]))]

    # 3) de-dupe by (deliverable_code, component_name, task_name) on LEAF tasks
    key = lambda t: (t["deliverable_code"], t["component_name"], t["name"])
    merged = {}
    for t in tasks:
        if t["is_summary"]:
            merged.setdefault(("__summary__", id(t)), t)
            continue
        k = key(t)
        if k not in merged:
            merged[k] = t
        else:
            m = merged[k]
            # merge timings
            m["start"]  = min(m["start"],  t["start"])
            m["finish"] = max(m["finish"], t["finish"])
            m["duration_minutes"] = max(m["duration_minutes"], t["duration_minutes"])
            # merge quantitative fields
            m["work_hours"] += t.get("work_hours", 0) or 0
            m["revenue"]    += t.get("revenue", 0) or 0
            # merge predecessor links (set by (uid,type))
            seen = {(p[0], p[1]) for p in m.get("predecessors", [])}
            for p in t.get("predecessors", []):
                if (p[0], p[1]) not in seen:
                    m.setdefault("predecessors", []).append(p)
    tasks = list(merged.values())

    # 4) recompute summaries: no manual dates, duration=0; roll up dates
    def rollup(node):
        if not node["children"]:
            node["is_summary"] = False
            return node["start"], node["finish"]
        starts, finishes = [], []
        for c in node["children"]:
            s, f = rollup(c)
            starts.append(s); finishes.append(f)
        node["is_summary"] = True
        node["duration_minutes"] = 0
        node["start"], node["finish"] = min(starts), max(finishes)
        node["manual_start"] = node["manual_finish"] = None
        node["constraint_type"] = 0
        return node["start"], node["finish"]

    ...
    return tasks
