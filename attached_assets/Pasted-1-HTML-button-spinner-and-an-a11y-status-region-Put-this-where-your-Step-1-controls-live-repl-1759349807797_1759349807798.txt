3) JS – wire it to your existing analyze flow

Add this to static/app.js (or wherever you handle Step‑1 click). The runAnalyze() function wraps your current fetch to /api/suggest_by_text (or /api/analyze_rfp) and toggles the throbber state. If your function already exists, keep its guts and just adopt the “busy” toggles shown here.

const analyzeBtn       = document.getElementById('analyzeBtn');
const analyzeCancelBtn = document.getElementById('analyzeCancelBtn');
const analyzeSpinner   = document.getElementById('analyzeSpinner');
const analyzeLabel     = document.getElementById('analyzeLabel');
const aiStatus         = document.getElementById('aiStatus');

let aiAbort = null; // set when a request is in-flight

function setAnalyzing(isBusy, labelBusy = 'Analyzing…') {
  analyzeBtn.setAttribute('aria-busy', isBusy ? 'true' : 'false');
  analyzeBtn.disabled = !!isBusy;
  analyzeLabel.textContent = isBusy ? labelBusy : 'Analyze with AI';
  aiStatus.textContent = isBusy ? 'Analyzing your RFP…' : 'Analysis finished.';
  analyzeCancelBtn.style.display = isBusy ? 'inline-block' : 'none';
}

async function runAnalyze() {
  // guard: avoid double clicks
  if (analyzeBtn.getAttribute('aria-busy') === 'true') return;

  // collect the RFP content you already gather (textarea or file text)
  const rfpText = getRfpTextFromUI();    // ← replace with your existing accessor
  const fileRef = getRfpFileFromUI();    // ← optional, if you support file uploads

  if (!rfpText && !fileRef) {
    alert('Paste RFP text or upload a file first.');
    return;
  }

  // prepare request body and AbortController
  aiAbort = new AbortController();
  setAnalyzing(true);

  try {
    let res;
    if (rfpText) {
      // text-based analysis
      res = await fetch('/api/suggest_by_text', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ rfp_text: rfpText }),
        signal: aiAbort.signal
      });
    } else {
      // file-based analysis (example — keep your existing code here)
      const fd = new FormData();
      fd.append('file', fileRef);
      res = await fetch('/api/suggest_by_file', { method: 'POST', body: fd, signal: aiAbort.signal });
    }

    if (!res.ok) throw new Error(`Analyze failed: ${res.status}`);
    const data = await res.json();

    // hand results to your Step‑2 initializer (suggestions + AI summary)
    // e.g. renderAISummary(data.summary); renderAISuggestions(data.suggested);
    onAnalyzeComplete(data); // ← call your existing handler

  } catch (err) {
    if (err.name !== 'AbortError') {
      console.error(err);
      alert('Sorry, the AI analysis didn’t complete. Please try again.');
    }
  } finally {
    setAnalyzing(false);
    aiAbort = null;
  }
}

analyzeBtn?.addEventListener('click', runAnalyze);

analyzeCancelBtn?.addEventListener('click', () => {
  if (aiAbort) {
    aiAbort.abort();
    aiStatus.textContent = 'Analysis canceled.';
    setAnalyzing(false);
    aiAbort = null;
  }
});

How it behaves

Clicking Analyze with AI changes the label to “Analyzing…”, shows the spinner, disables the button, and updates an aria‑live status for assistive tech.

When your analyze request resolves (or errors), the button returns to normal.

If you include the Cancel button, users can abort long analyses immediately.
