1) “Search deliverables …” box disappeared — put it back (and make it fast)

What happened
When we moved the Step‑2 layout, the search <input> was dropped and the previous filter handler wasn’t re‑bound after the panel moved. Result: no search and a large list.

Drop‑in UI patch (vanilla JS; no framework assumptions)

HTML – inside the Step‑2 left panel header, add the search box:

<div class="s2-panel-head">
  <input id="s2-deliv-search" type="search" placeholder="Search deliverables…" autocomplete="off">
  <div class="btn-row">
    <button id="s2-select-all" class="btn">Select all</button>
    <button id="s2-clear" class="btn">Clear</button>
  </div>
</div>


JS – wire the filter (runs after options are loaded and the list is rendered):

// One-time: call this after renderDeliverableList()
function bindDeliverableSearch() {
  const box = document.querySelector('#s2-deliv-search');
  const list = document.querySelector('#s2-deliv-list'); // <ul> or <div> that holds rows
  if (!box || !list) return;

  const norm = s => (s || '').toLowerCase().replace(/\s+/g, ' ').trim();

  box.addEventListener('input', () => {
    const q = norm(box.value);
    for (const row of list.querySelectorAll('[data-deliv-row]')) {
      const text = row.dataset.search || '';  // precomputed on render
      row.style.display = (!q || text.includes(q)) ? '' : 'none';
    }
  });
}


While rendering each row, stash a lowercase search key so filtering is O(1):

// inside renderDeliverableList()
row.dataset.delivRow = '1';
row.dataset.search = `${d.Deliverable} ${d.Category || ''} ${d['Service Department'] || ''}`.toLowerCase();


Result: instant search over name, category, and service department.

def emit_task(tasks, *, uid, name, outline, summary, start, finish, work_min=0, dur_min=480, preds=None):
    t = SubElement(tasks, 'Task')
    SubElement(t, 'UID').text = str(uid)
    SubElement(t, 'ID').text = str(uid)
    SubElement(t, 'Name').text = name
    SubElement(t, 'OutlineLevel').text = str(outline)
    SubElement(t, 'Summary').text = '1' if summary else '0'
    SubElement(t, 'Start').text = start
    SubElement(t, 'Finish').text = finish
    SubElement(t, 'Work').text = f'PT{int(work_min)}M'
    SubElement(t, 'Duration').text = f'PT{int(dur_min)}M'
    # Constraint/Calendar same as your project nodes; add predecessors if any...
    if preds:
        for p in preds:
            pl = SubElement(t, 'PredecessorLink')
            SubElement(pl, 'PredecessorUID').text = str(p)
            SubElement(pl, 'Type').text = '1'
            SubElement(pl, 'LinkLag').text = '0'
            SubElement(pl, 'LagFormat').text = '7'
    return uid
