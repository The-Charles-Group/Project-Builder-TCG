RFP_SNIPPET = "media management across 5 markets with paid social, reporting, optimizations, and research..."

class APBUser(HttpUser):

    @task(3)

        self.client.post("/api/step2/ai/weights", json={"rfp_text": RFP_SNIPPET})

    @task(1)
    def options(self):

6.2 (Optional) SSE status stream pattern

If you later want streaming progress (instead of polling), add in main.py:
from fastapi import Request
from starlette.responses import StreamingResponse

@app.get("/api/ai/stream/{job_id}")
async def stream(job_id: str, request: Request):
    async def gen():
        last = None
        while True:
            if await request.is_disconnected(): break
            s = AI_JOB_STORE.get(job_id)
            if s and s != last:
                yield f"data: {json.dumps({'stage': s.current_stage, 'progress': s.progress})}\n\n"
                last = s
            await asyncio.sleep(0.5)
    return StreamingResponse(gen(), media_type="text/event-stream")
