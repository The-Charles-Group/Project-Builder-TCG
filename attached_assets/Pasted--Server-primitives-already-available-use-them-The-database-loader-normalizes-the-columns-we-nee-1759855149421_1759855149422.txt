Components for selected deliverables

@app.post("/api/step2/components")
async def step2_components(payload: dict):
    """
    payload = {"deliverables": ["deck_strategy", "production", ...]}
    returns = {
      "deck_strategy": ["brief", "art_direction", "internal_review", ...],
      "production": ["prepro_coord", "shoot", "post", ...],
    }
    """
    codes = [str(c).strip() for c in payload.get("deliverables", [])]
    if not getattr(app.state, "db", None):
        app.state.db = AgencyDB(); app.state.db.load()
    df = app.state.db.all_rows.copy()

    out = {}
    for code in codes:
        subset = df[df["Deliverable_Code"] == code]
        comps = (
            subset["Component"]
            .fillna("").astype(str).str.strip()
            .replace({"nan": ""})
            .tolist()
        )
        comps = sorted({c for c in comps if c})
        out[code] = comps or ["general"]
    return JSONResponse(out)


L3 subtasks for a deliverable + component

@app.post("/api/step2/l3")
async def step2_l3(payload: dict):
    """
    payload = {"deliverable": "deck_strategy", "component": "brief"}
    returns = ["deck_build", "internal_review", "client_review", ...]
    """
    dcode = str(payload.get("deliverable", "")).strip()
    comp  = str(payload.get("component", "")).strip()
    if not getattr(app.state, "db", None):
        app.state.db = AgencyDB(); app.state.db.load()
    df = app.state.db.all_rows.copy()

    rows = df[(df["Deliverable_Code"] == dcode) & (df["Component"] == comp)]
    # Prefer Task_Label (normalized), fallback to task_group if empty
    label_col = "Task_Label"
    if label_col not in rows.columns or rows[label_col].eq("").all():
        label_col = "task_group" if "task_group" in rows.columns else None

    tasks = []
    if label_col:
        tasks = rows[label_col].fillna("").astype(str).str.strip().tolist()
    tasks = sorted({t for t in tasks if t})

    return JSONResponse(tasks)

async function addDeliverable(code: string) {
  if (step2.selectedDeliverables.has(code)) return;

  step2.selectedDeliverables.add(code);
  // Fetch components for ALL currently selected deliverables
  const body = { deliverables: Array.from(step2.selectedDeliverables) };
  const res = await fetch("/api/step2/components", {
    method: "POST", headers: {"Content-Type":"application/json"},
    body: JSON.stringify(body)
  });
  const data = await res.json();
  // Update components cache and default selections (none pre‑selected)
  Object.entries(data).forEach(([dcode, comps]) => {
    step2.componentsByDeliverable.set(dcode, comps as string[]);
    if (!step2.selectedComponents.has(dcode)) {
      step2.selectedComponents.set(dcode, new Set());
    }
  });

  renderDeliverablesList();     // pins selected to top + checkmarks
  renderComponentsPanel();      // dropdown defaults to the first selected deliverable
  renderAISuggestedBox();       // flips “Add” ➜ “Added” + “Remove”
  renderSummary();              // counts in the blue Summary box
}

function removeDeliverable(code: string) {
  step2.selectedDeliverables.delete(code);
  step2.componentsByDeliverable.delete(code);
  step2.selectedComponents.delete(code);

  // purge any cached L3 for this deliverable
  [...step2.l3ByKey.keys()].forEach(k => { if (k.startsWith(code + "|")) step2.l3ByKey.delete(k); });
  [...step2.selectedL3.keys()].forEach(k => { if (k.startsWith(code + "|")) step2.selectedL3.delete(k); });

  renderDeliverablesList();
  renderComponentsPanel();  // resets if nothing selected
  renderL3Panel(true);      // clear
  renderAISuggestedBox();   // “Add” button reappears
  renderSummary();
}

async function ensureL3Loaded(code: string, component: string) {
  const key = `${code}|${component}`;
  if (step2.l3ByKey.has(key)) return;

  const res = await fetch("/api/step2/l3", {
    method: "POST", headers: {"Content-Type":"application/json"},
    body: JSON.stringify({ deliverable: code, component })
  });
  const tasks = await res.json();
  step2.l3ByKey.set(key, tasks);
  if (!step2.selectedL3.has(key)) step2.selectedL3.set(key, new Set());
  renderL3Panel();
  renderSummary();
}

function sortDeliverablesForRender(items: {code:string; label:string;}[]) {
  return items.sort((a,b) => {
    const aSel = step2.selectedDeliverables.has(a.code) ? 0 : 1;
    const bSel = step2.selectedDeliverables.has(b.code) ? 0 : 1;
    return aSel - bSel || a.label.localeCompare(b.label);
  });
}
