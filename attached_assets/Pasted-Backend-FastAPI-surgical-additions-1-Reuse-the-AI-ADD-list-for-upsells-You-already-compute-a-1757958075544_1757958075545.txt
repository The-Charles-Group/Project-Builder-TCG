2) New endpoint: /api/build_scenario_c

Builds C by taking a base (“A” or “B”) and unioning with a user‑chosen subset of add[] codes, then pricing and scheduling with the same builder used for A/B (including scenario_letter="C" so task‑ordering can differ later if you want it to). Under the hood this calls the same schedule + pricing helpers you already have: hours_by_role_for_deliverable, price_for_hours_by_role, build_schedule, etc.

class BuildScenarioCPayload(BaseModel):
    base: str  # "A" or "B"
    add_on_codes: list[str] = []
    pricing_mode: str = "Flat_Blended"   # or "Per_Resource"
    blended_rate: float = 195.0
    rate_band: str = "Standard_US"
    complexity: str | None = None        # default to base scenario’s
    tier: str | None = None
    use_slack: bool | None = None
    slack_after_internal: int | None = None
    slack_after_client: int | None = None
    slack_global_pct: float | None = None
    project_start: str | None = None     # "YYYY-MM-DD"

@app.post("/api/build_scenario_c")
def api_build_scenario_c(p: BuildScenarioCPayload):
    if not DB.loaded: DB.load()

    # 1) fetch the base scenario (A or B) from wherever you’re storing it (session/state)
    base_letter = p.base.upper()
    base = _current_scenarios().get(base_letter)
    if not base:
        raise HTTPException(400, f"Base scenario {base_letter} not found.")

    # 2) union of deliverable codes
    base_codes = [i["deliverable_code"] for i in (base.get("items") or [])]
    sel_codes = list(dict.fromkeys(base_codes + list(p.add_on_codes)))

    # 3) inherit params from base unless explicitly provided
    complexity = p.complexity or base.get("complexity", "Advanced")
    tier       = p.tier or base.get("tier", "T2_MediumVolume")
    use_slack  = base.get("use_slack", True) if p.use_slack is None else p.use_slack
    s_i        = base.get("slack_after_internal", 1) if p.slack_after_internal is None else p.slack_after_internal
    s_c        = base.get("slack_after_client", 2)   if p.slack_after_client is None else p.slack_after_client
    s_pct      = base.get("slack_global_pct", 0.05)  if p.slack_global_pct is None else p.slack_global_pct
    start      = p.project_start or base.get("project_start")
    scen_col   = DB.scenario_hours_col(complexity, tier)

    items = []
    for code in sel_codes:
        # included task groups for this deliverable (same logic as A/B)
        included = DB.sorted_task_groups(DB.included_task_groups_for_code(code) if hasattr(DB, "included_task_groups_for_code") else
                                         list({str(r["task_group"]) for _, r in DB.all_rows[DB.all_rows["Deliverable_Code"]==code].iterrows()}))
        # schedule uses scenario_letter="C"
        sched = DB.build_schedule(code, included, complexity, tier, use_slack, s_i, s_c, s_pct,
                                  project_start=start, scenario_letter="C")
        # price (Flat_Blended or Per_Resource)
        hrs_by_role = DB.hours_by_role_for_deliverable(code, included, scen_col)
        if p.pricing_mode == "Flat_Blended":
            price_total = DB.blended_price(hrs_by_role["Hours"].sum() if not hrs_by_role.empty else 0.0, p.blended_rate)
        else:
            price_total = DB.per_resource_price(hrs_by_role, p.rate_band)

        items.append({
            "deliverable_code": code,
            "deliverable": str(DB.deliverables.loc[DB.deliverables["Deliverable_Code"]==code, "Deliverable"].iloc[0]) if not DB.deliverables.empty else code,
            "included_task_groups": included,
            "schedule": sched,
            "price_total": round(price_total, 0)
        })

    scenario_c = {
        "label": "Scenario C (Upsell)",
        "complexity": complexity, "tier": tier, "use_slack": use_slack,
        "slack_after_internal": s_i, "slack_after_client": s_c, "slack_global_pct": s_pct,
        "project_start": start,
        "items": items,
        "totals": {"price_total": int(sum(i["price_total"] or 0 for i in items))}
    }
    # store/update in memory next to A/B for this session
    _current_scenarios()["C"] = scenario_c
    return {"C": scenario_c}

function buildScenarioC() {
  const base = document.querySelector('#c-base')?.value || 'A';
  const addCodes = [...document.querySelectorAll('#c-adds input[type=checkbox]:checked')]
                    .map(cb => cb.dataset.code);
  const prefs = getPricingPrefs(); // pricing_mode, rate_band, blended_rate, slack, etc.

  return fetch('/api/build_scenario_c', {
    method: 'POST', headers: {'Content-Type':'application/json'},
    body: JSON.stringify({ base, add_on_codes: addCodes, ...prefs })
  }).then(r => r.json()).then(res => {
    appState.scenarios = { ...(appState.scenarios||{}), ...res };
    renderScenario('C', appState.scenarios.C);      // reuse your existing renderer
    // Show delta vs base
    const baseTotal = appState.scenarios[base]?.totals?.price_total || 0;
    const cTotal = appState.scenarios.C?.totals?.price_total || 0;
    setText('c-delta', `Δ vs ${base}: $${(cTotal - baseTotal).toLocaleString()}`);
  });
}

function renderUpsellList() {
  const base = document.querySelector('#c-base')?.value || 'A';
  const baseCodes = new Set((appState.scenarios?.[base]?.items || []).map(i => i.deliverable_code));
  const adds = (appState.reconcile?.add || []).filter(s => !baseCodes.has(s.code));
  const html = adds.length ? adds.map(s => 
      `<label><input type="checkbox" data-code="${s.code}" checked> ${s.label}</label>`
    ).join('<br>') : '<em>No AI add‑ons available</em>';
  setHTML('c-adds', html);
}
document.querySelector('#c-base')?.addEventListener('change', renderUpsellList);

function scopeOnlyFilter(labels) {
  const drop = [/microsite/i, /editorial/i, /live\s*stream/i, /experiential/i];
  return labels.filter(s => !drop.some(rx => rx.test(s.label)));
}
