2) Frontend — show Budget Analysis for A, B and C

Where you currently render the green “Budget Analysis” pill in the Scenario A card, generalize it to a helper and call it for all scenarios.

Add helper:

function renderBudgetPill(el, scenarioTotalsPrice, clientBudget) {
  const budget = Number(clientBudget || 0);
  el.innerHTML = ""; el.className = "budget-pill";
  if (!budget || !scenarioTotalsPrice) return;

  const delta = budget - scenarioTotalsPrice;
  const pct   = scenarioTotalsPrice / budget;
  const span  = document.createElement("span");

  if (delta >= 0) {
    span.textContent = `Under budget by $${delta.toLocaleString()} (${(100*(1-pct)).toFixed(1)}%)`;
    el.classList.add("under");
  } else {
    span.textContent = `Over budget by $${Math.abs(delta).toLocaleString()} (${(100*(pct-1)).toFixed(1)}%)`;
    el.classList.add("over");
  }
  el.appendChild(span);
}

async function onBuildScenarioC() {
  const base = document.querySelector('#scenarioCBase').value;       // "A" or "B"
  const addOnCodes = getSelectedUpsellCodes();                       // [] for now if no UI
  const pricingMode = state.pricingMode;
  const blendedRate = state.blendedRate;
  const rateBand    = state.rateBand;

  // Optional: pass retainers months if user set any in Step 3 for these deliverables
  const retainers = state.retainers || [];  // [{deliverable_code, months}, ...]

  const payload = {
    base,
    add_on_codes: addOnCodes,
    pricing_mode: pricingMode,
    blended_rate: blendedRate,
    rate_band: rateBand,
    project_start: state.projectStartISO,   // "YYYY-MM-DD"
    retainers
  };

  const res = await fetch("/api/build_scenario_c", {
    method:"POST", headers:{ "Content-Type":"application/json" },
    body: JSON.stringify(payload)
  });
  if (!res.ok) { toast("Build C failed"); return; }
  const data = await res.json();
  state.scenarioC = data.C;
  renderScenarioCard("C", state.scenarioC);               // mirror A/B card rendering
  renderBudgetPill(document.getElementById("budgetPillC"),
                   state.scenarioC?.totals?.price,
                   document.getElementById("clientBudgetInput").value);
}
