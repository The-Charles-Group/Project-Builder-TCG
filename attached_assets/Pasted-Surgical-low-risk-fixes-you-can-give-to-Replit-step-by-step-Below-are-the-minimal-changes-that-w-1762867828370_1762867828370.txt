4) Add a 1‑pass “lint” just before sending the file (belt & suspenders)

This is a 20‑line sanitizer that catches any stragglers if a different branch ever reintroduces formatted strings:

import re

def lint_mspdi_in_place(xml_path: str):
    with open(xml_path, "r", encoding="utf-8") as f:
        s = f.read()
    # 1) Rates: strip currency and “/h”, keep digits+dot
    s = re.sub(r"<StandardRate>[^<]*</StandardRate>",
               lambda m: f"<StandardRate>{re.sub(r'[^0-9.]', '', m.group(0))}</StandardRate>", s)
    s = re.sub(r"<OvertimeRate>[^<]*</OvertimeRate>",
               lambda m: f"<OvertimeRate>{re.sub(r'[^0-9.]', '', m.group(0))}</OvertimeRate>", s)
    # 2) MaxUnits/PeakUnits: clamp to 1 if more
    s = re.sub(r"<MaxUnits>(\d+)</MaxUnits>", lambda m: "<MaxUnits>1</MaxUnits>" if int(m.group(1))>1 else m.group(0), s)
    s = re.sub(r"<PeakUnits>(\d+)</PeakUnits>", lambda m: "<PeakUnits>1</PeakUnits>" if int(m.group(1))>1 else m.group(0), s)
    # 3) Assignment Units: convert 100 -> 1.0
    s = re.sub(r"<Units>(\d+)</Units>", lambda m: "<Units>1.0000</Units>" if int(m.group(1))>1 else "<Units>1.0000</Units>", s)
    with open(xml_path, "w", encoding="utf-8") as f:
        f.write(s)


Call lint_mspdi_in_place(output_xml) right after conversion and before returning the file.
