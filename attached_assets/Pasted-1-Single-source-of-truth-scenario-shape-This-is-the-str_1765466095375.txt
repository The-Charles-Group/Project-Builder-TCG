2.2. Tracking component edits in the scenario

We need a single function that updates:

item.component_hours[componentLabel]

Parent hours fields (Planned_Hours, etc.)

The UI parent row

Ask them to wire every Hours input in the Deliverable Pricing Details panel to this:

let pricingDetailsDirty = false;

function onComponentHoursChange(delivCode, componentLabel, newValue) {
  const scenario = window.SCENARIO_A;
  if (!scenario || !scenario.items) return;

  const item = scenario.items.find(
    it => String(it.deliverable_code) === String(delivCode)
  );
  if (!item) return;

  const hours = Number(newValue) || 0;

  // 1) Ensure component_hours exists
  if (!item.component_hours) {
    item.component_hours = {};
  }

  // 2) Update that component
  item.component_hours[componentLabel] = hours;

  // 3) Re‑sum parent hours from ALL components (fixes the "+3 hours" bug)
  const totalFromComponents = Object.values(item.component_hours)
    .map(v => Number(v) || 0)
    .reduce((a, b) => a + b, 0);

  item.Planned_Hours = totalFromComponents;
  item.planned_hours = totalFromComponents;
  item.hours = totalFromComponents;

  // 4) Reflect in the Project Scenario table parent row
  updateProjectScenarioRow(delivCode, {
    hours: totalFromComponents
  });

  pricingDetailsDirty = true;
}


Then, in the pricing details render code, each hours <input> should have:

<input
  value={hours}
  onChange={e => onComponentHoursChange(delivCode, componentLabel, e.target.value)}
/>

2.3. Update Pricing button (in the Pricing Details panel)

This button must:

Send the updated scenario (with component_hours) to the backend.

Ask the backend to rebuild the authoritative totals.

Update the UI from the backend response.

Clear the dirty flag.

async function syncScenarioToBackend(scenario, sessionId) {
  await fetch('/api/scenario/sync', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ session_id: sessionId, scenario })
  });
}

async function rebuildPricingFromBackend(scenario, sessionId) {
  const res = await fetch('/api/pricing/rebuild_breakdown', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ session_id: sessionId, scenario })
  });

  const data = await res.json();
  return data.scenario;  // authoritative scenario from backend
}

async function handleUpdatePricing(sessionId) {
  if (!pricingDetailsDirty) return;

  const scenario = window.SCENARIO_A;
  if (!scenario) return;

  // 1) Persist edits (including component_hours) to backend
  await syncScenarioToBackend(scenario, sessionId);

  // 2) Get authoritative scenario + totals
  const rebuiltScenario = await rebuildPricingFromBackend(scenario, sessionId);

  // 3) Replace local scenario + re-render both panels
  window.SCENARIO_A = rebuiltScenario;
  renderProjectScenario(rebuiltScenario);
  renderPricingDetails(rebuiltScenario);

  pricingDetailsDirty = false;
}


Wire the Update Pricing CTA in the Deliverable Pricing Details panel to handleUpdatePricing(currentSessionId).

2.4. Build Scenario button (top panel)

Important: On Step 3, the Build Scenario button should not throw away your edits and rebuild from scratch.

Instead, have it share the same path as Update Pricing:

async function handleBuildScenario(sessionId) {
  const scenario = window.SCENARIO_A;

  // If no scenario yet (first time from Step 2), fall back to the original build call
  if (!scenario || !scenario.items || !scenario.items.length) {
    const data = await buildScenariosAB(sessionId); // existing endpoint
    window.SCENARIO_A = data.scenario;
    renderProjectScenario(window.SCENARIO_A);
    renderPricingDetails(window.SCENARIO_A);
    return;
  }

  // Otherwise, treat it like a save + recompute
  await handleUpdatePricing(sessionId);
}


This guarantees one source of truth for Step 3: all edits flow into window.SCENARIO_A → backend → exports.
