Step 2 — Vanilla JS module (replace the earlier Step2 block)

Important: deliverable rows in your left column should include both data-code (canonical code) and data-name (friendly display). The handlers below rely on them.

// ===== Step 2 State & Rendering (Vanilla JS) — v2 =====

const Step2 = (() => {
  // ---------- State ----------
  const LS_KEY = "apb.step2.selection.v2";

  const state = {
    selectedDeliverables: new Set(),                       // Set<deliverable_code>
    selectedComponents: {},                                // { [dCode]: Set<component> }
    selectedL3: {},                                        // { [dCode]: { [component]: Set<taskLabel> } }
    caches: {                                              // response caches
      components: {},                                      // { [dCode]: string[] }
      l3: {}                                               // { [dCode]: { [component]: string[] } }
    },
    activeDeliverable: null,                                // dCode for the components pane
    includeAnchors: false                                   // XML toggle
  };

  // ---------- Persistence ----------
  const persist = () => {
    const obj = {
      selectedDeliverables: [...state.selectedDeliverables],
      selectedComponents: Object.fromEntries(
        Object.entries(state.selectedComponents).map(([d, set]) => [d, [...set]])
      ),
      selectedL3: Object.fromEntries(
        Object.entries(state.selectedL3).map(([d, compMap]) => [
          d, Object.fromEntries(Object.entries(compMap).map(([c, set]) => [c, [...set]]))
        ])
      ),
      activeDeliverable: state.activeDeliverable,
      includeAnchors: state.includeAnchors,
    };
    localStorage.setItem(LS_KEY, JSON.stringify(obj));
  };

  const rehydrate = () => {
    try {
      const raw = localStorage.getItem(LS_KEY);
      if (!raw) return;
      const obj = JSON.parse(raw);

      state.selectedDeliverables = new Set(obj.selectedDeliverables || []);
      state.selectedComponents = {};
      for (const [d, arr] of Object.entries(obj.selectedComponents || {})) {
        state.selectedComponents[d] = new Set(arr || []);
      }
      state.selectedL3 = {};
      for (const [d, compMap] of Object.entries(obj.selectedL3 || {})) {
        state.selectedL3[d] = {};
        for (const [c, arr] of Object.entries(compMap || {})) {
          state.selectedL3[d][c] = new Set(arr || []);
        }
      }
      state.activeDeliverable = obj.activeDeliverable || null;
      state.includeAnchors = !!obj.includeAnchors;

      const chk = document.getElementById("toggle-anchors");
      if (chk) chk.checked = state.includeAnchors;
    } catch {}
  };

  // ---------- API ----------
  async function fetchComponents(dCode) {
    if (state.caches.components[dCode]) return state.caches.components[dCode];
    const res = await fetch("/api/step2/components", {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify({ deliverables: [dCode] })
    });
    const data = await res.json(); // { [dCode]: [components...] }
    const list = (data && data[dCode]) || [];
    state.caches.components[dCode] = list;
    return list;
  }

  async function fetchL3(dCode, components) {
    if (!components || !components.length) return {};
    const missing = components.filter(c => !((state.caches.l3[dCode]||{})[c]));
    if (missing.length) {
      const res = await fetch("/api/step2/l3/bulk", {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({ deliverable: dCode, components: missing })
      });
      const data = await res.json(); // { [component]: [taskLabel...] }
      state.caches.l3[dCode] = state.caches.l3[dCode] || {};
      for (const [c, arr] of Object.entries(data || {})) {
        state.caches.l3[dCode][c] = arr || [];
      }
    }
    const out = {};
    for (const c of components) {
      out[c] = (state.caches.l3[dCode] && state.caches.l3[dCode][c]) || [];
    }
    return out;
  }

  // ---------- Helpers ----------
  const ensureMaps = (d) => {
    state.selectedComponents[d] ||= new Set();
    state.selectedL3[d] ||= {};
  };

  const setActiveDeliverable = async (dCode) => {
    state.activeDeliverable = dCode;
    persist();
    // Always re-render components for the new active deliverable
    const comps = await fetchComponents(dCode);
    renderComponentsPane(dCode, comps);
  };

  // Preload all L3 for currently selected components of a deliverable (for instant Summary render)
  const ensureAllL3LoadedForDeliverable = async (dCode) => {
    const comps = [...(state.selectedComponents[dCode] || new Set())];
    if (!comps.length) return;
    // show "loading" placeholders in Summary while fetching
    renderSummary({ showLoadingFor: { [dCode]: new Set(comps) } });
    await fetchL3(dCode, comps);
    renderSummary();
  };

  // ---------- Mutations ----------
  async function toggleDeliverable(dCode, checked) {
    ensureMaps(dCode);
    if (checked) {
      state.selectedDeliverables.add(dCode);
      await setActiveDeliverable(dCode);         // make it active so its components match the selection
    } else {
      state.selectedDeliverables.delete(dCode);
      delete state.selectedComponents[dCode];
      delete state.selectedL3[dCode];
      if (state.activeDeliverable === dCode) state.activeDeliverable = null;
    }
    persist(); recount(); renderSummary();
  }

  async function toggleComponent(dCode, comp, checked) {
    ensureMaps(dCode);
    if (checked) {
      state.selectedComponents[dCode].add(comp);
      // default: select ALL L3 for this component
      await fetchL3(dCode, [comp]);
      const all = (state.caches.l3[dCode] && state.caches.l3[dCode][comp]) || [];
      state.selectedL3[dCode][comp] = new Set(all);
    } else {
      state.selectedComponents[dCode].delete(comp);
      delete state.selectedL3[dCode][comp];
    }
    persist(); recount(); renderSummary();
  }

  function toggleL3(dCode, comp, label, checked) {
    ensureMaps(dCode);
    state.selectedL3[dCode][comp] ||= new Set();
    if (checked) state.selectedL3[dCode][comp].add(label);
    else state.selectedL3[dCode][comp].delete(label);
    persist(); recount(); // Summary already visible
  }

  // ---------- Counts ----------
  const recount = () => {
    const d = state.selectedDeliverables.size;
    let c = 0, t = 0;
    for (const dCode of Object.keys(state.selectedComponents)) {
      c += (state.selectedComponents[dCode] || new Set()).size;
    }
    for (const [dCode, map] of Object.entries(state.selectedL3)) {
      for (const set of Object.values(map)) t += set.size;
    }
    (document.getElementById("count-deliverables")||{}).textContent = d;
    (document.getElementById("count-components")||{}).textContent = c;
    (document.getElementById("count-l3")||{}).textContent = t;
  };

  // ---------- Rendering ----------
  function renderComponentsPane(dCode, components) {
    const pane = document.querySelector("#components-pane") || document.querySelector('[data-pane="components"]');
    if (!pane) return;
    const sel = state.selectedComponents[dCode] || new Set();

    pane.innerHTML = `
      <div class="panel-head row-between">
        <h3>Components</h3>
        <div class="row-between" style="gap:8px">
          <div class="muted" title="Deliverable Code">${dCode}</div>
          <button type="button" class="btn-ghost" id="btn-suggest-components" data-d="${dCode}">Suggest</button>
        </div>
      </div>
      <div class="list">
        ${components.map(c => `
          <label class="row">
            <input data-kind="component" data-d="${dCode}" data-c="${c}" type="checkbox" ${sel.has(c) ? "checked": ""}>
            <span>${c}</span>
          </label>
        `).join("")}
      </div>
    `;
  }

  function renderSummary(options = {}) {
    const root = document.getElementById("summary-list");
    if (!root) return;

    const loading = options.showLoadingFor || {};
    const blocks = [];

    for (const dCode of state.selectedDeliverables) {
      const compList = [...(state.selectedComponents[dCode] || new Set())].sort();
      if (!compList.length) {
        blocks.push(`<div class="summary-group" data-d="${dCode}"><h4>${dCode}</h4><div class="muted">No components selected.</div></div>`);
        continue;
      }

      blocks.push(`
        <div class="summary-group" data-d="${dCode}">
          <h4>${dCode}</h4>
          ${compList.map(c => {
            const all = (state.caches.l3[dCode] && state.caches.l3[dCode][c]) || [];
            const sel = (state.selectedL3[dCode] && state.selectedL3[dCode][c]) ? [...state.selectedL3[dCode][c]] : [];
            const isLoading = loading[dCode] && loading[dCode].has && loading[dCode].has(c);

            return `
              <div class="summary-comp" data-d="${dCode}" data-c="${c}">
                <div class="bar">
                  <h5>${c}</h5>
                  <div class="summary-actions">
                    <button type="button" data-act="all">All</button>
                    <button type="button" data-act="none">None</button>
                    <button type="button" data-act="suggest-l3">Suggest</button>
                  </div>
                </div>
                <div class="summary-l3">
                  ${isLoading ? `<div class="muted">Loading tasks…</div>` :
                    (!all.length ? `<div class="muted">(no tasks defined)</div>` :
                      all.map(t => `
                        <label title="${t}">
                          <input data-kind="l3" data-d="${dCode}" data-c="${c}" data-t="${encodeURIComponent(t)}" type="checkbox" ${sel.includes(t) ? "checked": ""}>
                          <span>${t}</span>
                        </label>
                      `).join("")
                    )
                  }
                </div>
              </div>
            `;
          }).join("")}
        </div>
      `);
    }

    root.innerHTML = blocks.join("");
  }

  // ---------- Events ----------
  function bindEvents() {
    // DELIVERABLES pane (left)
    const delivPane = document.querySelector("#deliverables-pane") || document.querySelector('[data-pane="deliverables"]');

    if (delivPane) {
      // Row click = PREVIEW (does not mutate selections)
      delivPane.addEventListener("click", async (e) => {
        const row = e.target.closest("[data-deliverable-row]");
        if (!row) return;
        if (e.target.closest('input[type="checkbox"]')) return; // checkbox handled below
        const dCode = row.getAttribute("data-code");            // <-- always use code
        if (dCode) await setActiveDeliverable(dCode);
      });

      // Checkbox = SELECT / UNSELECT
      delivPane.addEventListener("change", async (e) => {
        const cb = e.target.closest('input[type="checkbox"][data-deliverable]');
        if (!cb) return;
        const dCode = cb.getAttribute("data-code");             // <-- always use code
        await toggleDeliverable(dCode, cb.checked);
        if (cb.checked) {
          // UX: show its components immediately
          await ensureAllL3LoadedForDeliverable(dCode);
        }
      });
    }

    // COMPONENTS pane (middle)
    const compPane = document.querySelector("#components-pane") || document.querySelector('[data-pane="components"]');
    if (compPane) {
      compPane.addEventListener("change", async (e) => {
        const cb = e.target.closest('input[type="checkbox"][data-kind="component"]');
        if (!cb) return;
        const d = cb.getAttribute("data-d");
        const c = cb.getAttribute("data-c");
        await toggleComponent(d, c, cb.checked);
      });

      compPane.addEventListener("click", async (e) => {
        const btn = e.target.closest("#btn-suggest-components");
        if (!btn) return;
        const dCode = btn.getAttribute("data-d");
        await suggestComponentsForDeliverable(dCode);
      });
    }

    // SUMMARY pane (right)
    const summary = document.getElementById("summary-list");
    if (summary) {
      summary.addEventListener("click", async (e) => {
        const btn = e.target.closest('button[data-act]');
        if (!btn) return;
        const block = btn.closest(".summary-comp");
        if (!block) return;
        const d = block.getAttribute("data-d");
        const c = block.getAttribute("data-c");
        if (btn.dataset.act === "all") {
          const all = (state.caches.l3[d] && state.caches.l3[d][c]) || [];
          state.selectedL3[d][c] = new Set(all);
          persist(); recount(); renderSummary();
        } else if (btn.dataset.act === "none") {
          state.selectedL3[d][c] = new Set();
          persist(); recount(); renderSummary();
        } else if (btn.dataset.act === "suggest-l3") {
          await suggestL3ForComponent(d, c);
        }
      });

      summary.addEventListener("change", (e) => {
        const cb = e.target.closest('input[type="checkbox"][data-kind="l3"]');
        if (!cb) return;
        const d = cb.getAttribute("data-d");
        const c = cb.getAttribute("data-c");
        const t = decodeURIComponent(cb.getAttribute("data-t") || "");
        toggleL3(d, c, t, cb.checked);
      });
    }

    // Anchors toggle
    const anchorsChk = document.getElementById("toggle-anchors");
    if (anchorsChk) {
      anchorsChk.addEventListener("change", () => {
        state.includeAnchors = anchorsChk.checked;
        persist();
      });
    }
  }

  // ---------- AI Suggest (client-side call helpers) ----------
  async function suggestComponentsForDeliverable(dCode) {
    if (!dCode) return;
    const res = await fetch("/api/step2/suggest/components", {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify({ deliverable_code: dCode })
    });
    const data = await res.json(); // ["comp A","comp B",...]
    ensureMaps(dCode);
    for (const c of data || []) {
      state.selectedComponents[dCode].add(c);
    }
    // Preselect all L3 for newly added comps
    await fetchL3(dCode, data || []);
    for (const c of data || []) {
      const all = (state.caches.l3[dCode] && state.caches.l3[dCode][c]) || [];
      state.selectedL3[dCode][c] = new Set(all);
    }
    persist(); recount(); renderSummary();
    // refresh components pane checkboxes
    const comps = await fetchComponents(dCode);
    renderComponentsPane(dCode, comps);
  }

  async function suggestL3ForComponent(dCode, comp) {
    const exclude = new Set();
    // exclude L3 already selected across ALL deliverables/components to dedupe globally
    for (const [dx, cmap] of Object.entries(state.selectedL3)) {
      for (const set of Object.values(cmap)) for (const t of set) exclude.add(t.toLowerCase());
    }
    const res = await fetch("/api/step2/suggest/l3", {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify({ deliverable_code: dCode, components: [comp], exclude_labels: [...exclude] })
    });
    const payload = await res.json(); // { [comp]: ["task a","task b"] }
    const sugg = payload[comp] || [];

    ensureMaps(dCode);
    state.selectedL3[dCode][comp] ||= new Set();
    for (const t of sugg) state.selectedL3[dCode][comp].add(t);

    // Make sure cache has full list so checkboxes can render
    await fetchL3(dCode, [comp]);
    persist(); recount(); renderSummary();
  }

  // ---------- Public ----------
  function init() {
    rehydrate(); recount(); renderSummary(); bindEvents();
    // If we have an active deliverable from a previous session, show its components
    if (state.activeDeliverable) {
      fetchComponents(state.activeDeliverable).then(list => renderComponentsPane(state.activeDeliverable, list));
    }
  }

  return { init };
})();

document.addEventListener("DOMContentLoaded", () => Step2.init());

Two small backend endpoints — “Suggest” buttons

These are pure FastAPI and use the data you already load into memory in Stage 2, including the cached RFP text (RFP_TEXT_CACHE). If OpenAI isn’t configured, they fall back to deterministic heuristics. (The global RFP_TEXT_CACHE is already in your app for Step‑1→2 handoff. )

Add near your other Step‑2 routes:

# --- Suggest Components for a deliverable ---
class SuggestComponentsReq(BaseModel):
    deliverable_code: str
    limit: int | None = 6

@app.post("/api/step2/suggest/components")
def suggest_components(req: SuggestComponentsReq):
    db: AgencyDB = app.state.db  # however you store it; or keep a module-global
    d = str(req.deliverable_code)

    sub = db.all_rows[db.all_rows["Deliverable_Code"].astype(str) == d]
    if sub.empty:
      return JSONResponse([])

    # Simple weighting: frequency of tasks per Component, boosted by RFP keyword overlap
    txt = (globals().get("RFP_TEXT_CACHE") or "").lower()  # RFP from Step 1
    tokens = {t for t in re.findall(r"[a-z0-9]{3,}", txt)}

    # score component by (#rows) + bonus if Task_Label matches RFP
    scores = {}
    for _, r in sub.iterrows():
        comp = str(r.get("Component","") or "").strip() or "General"
        scores.setdefault(comp, 0)
        scores[comp] += 1
        lab = str(r.get("Task_Label","") or "").lower()
        if tokens and any(tok in lab for tok in tokens):
            scores[comp] += 2  # small boost for RFP relevance

    ranked = sorted(scores.items(), key=lambda x: (-x[1], x[0]))
    out = [c for c,_ in ranked[: (req.limit or 999)]]
    return JSONResponse(out)


# --- Suggest L3 tasks for one or more components (dedupe-aware) ---
class SuggestL3Req(BaseModel):
    deliverable_code: str
    components: list[str]
    exclude_labels: list[str] | None = None
    limit_per_component: int | None = 20

@app.post("/api/step2/suggest/l3")
def suggest_l3(req: SuggestL3Req):
    db: AgencyDB = app.state.db
    d = str(req.deliverable_code)
    excl = { (x or "").strip().lower() for x in (req.exclude_labels or []) }
    txt = (globals().get("RFP_TEXT_CACHE") or "").lower()
    tokens = {t for t in re.findall(r"[a-z0-9]{3,}", txt)}

    resp: dict[str, list[str]] = {}
    sub = db.all_rows[db.all_rows["Deliverable_Code"].astype(str)==d]
    if sub.empty:
        return JSONResponse(resp)

    for c in req.components or []:
        cand = sub[(sub["Component"].astype(str)==str(c)) | ((sub["Component"]=="") & (c=="General"))]
        if cand.empty:
            resp[c] = []
            continue

        # score tasks: frequency + RFP keyword overlap
        t_scores: dict[str,int] = {}
        for _, r in cand.iterrows():
            lab = str(r.get("Task_Label","") or "").strip()
            if not lab:
                continue
            key = lab.lower()
            t_scores.setdefault(lab, 0)
            t_scores[lab] += 1
            if tokens and any(tok in key for tok in tokens):
                t_scores[lab] += 2

        # drop duplicates across the whole project if requested
        if excl:
            t_scores = {lab:sc for lab,sc in t_scores.items() if lab.lower() not in excl}

        ranked = sorted(t_scores.items(), key=lambda x: (-x[1], x[0]))
        resp[c] = [lab for lab,_ in ranked[: (req.limit_per_component or 999)]]

    return JSONResponse(resp)

UI wiring for the “Suggest” buttons

Components panel (middle) now shows a Suggest button for the active deliverable. Clicking it calls /api/step2/suggest/components, selects those components, fetches their L3, and updates Summary.

Summary panel (right) shows a Suggest button per component group. Clicking it calls /api/step2/suggest/l3 and merges the recommendations into your current L3 selection.
Dedupe is global: we send exclude_labels with all already‑selected L3 (across all deliverables/components) so you don’t get duplicative tasks between groups — per Bradley’s note.

How to test quickly

Select a deliverable by the checkbox. The Components panel header shows the deliverable code it is currently bound to; the list should instantly match.

Check a few components → you should immediately see their full L3 appear in the Summary.

Click Suggest in the Components panel → it should pick relevant components (RFP‑aware if you pasted text in Step 1).

Click Suggest in any Summary component → it should add a curated L3 subset (no duplicates already selected elsewhere).

Toggle “Include Start/End anchors in XML” before export to include/omit the Start — DEL‑####/End — DEL‑#### rows in the XML (the converter flag exists already; we’re just exposing it).
