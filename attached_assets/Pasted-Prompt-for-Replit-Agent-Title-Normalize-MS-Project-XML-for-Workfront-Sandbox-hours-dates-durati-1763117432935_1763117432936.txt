import re, sys, datetime as dt
import xml.etree.ElementTree as ET

MSP_NS = "http://schemas.microsoft.com/project"
NS = {'msp': MSP_NS}
ET.register_namespace('', MSP_NS)

DUR_RE = re.compile(r'^PT(?:(\d+)H)?(?:(\d+)M)?(?:(\d+)S)?$')
WORK_WINDOWS = [(dt.time(9,0), dt.time(13,0)), (dt.time(14,0), dt.time(18,0))]
WORKING_WEEKDAYS = {0,1,2,3,4}  # Mon–Fri
MINUTES_PER_DAY = 480

def parse_pt_to_minutes(s):
    if not s: return 0
    m = DUR_RE.match(s.strip())
    if not m: return 0
    h = int(m.group(1) or 0); mns = int(m.group(2) or 0); sec = int(m.group(3) or 0)
    return h*60 + mns + (sec // 60)

def minutes_to_pt(m): return f"PT{int(round(m))}M"

def parse_dt(s): return dt.datetime.fromisoformat(s)

def clamp_to_work_window(ts):
    # Move ts forward into the next working minute according to WORK_WINDOWS / WORKING_WEEKDAYS
    d = ts
    while d.weekday() not in WORKING_WEEKDAYS:
        d = (d + dt.timedelta(days=1)).replace(hour=9, minute=0, second=0, microsecond=0)
    for start,end in WORK_WINDOWS:
        if d.time() < start: return d.replace(hour=start.hour, minute=start.minute, second=0, microsecond=0)
        if start <= d.time() <= end: return d
    # past last window → next day at 09:00
    d = (d + dt.timedelta(days=1)).replace(hour=9, minute=0, second=0, microsecond=0)
    while d.weekday() not in WORKING_WEEKDAYS:
        d = (d + dt.timedelta(days=1)).replace(hour=9, minute=0, second=0, microsecond=0)
    return d

def move_back_into_window(ts):
    # Move ts backward into a working minute
    d = ts
    while d.weekday() not in WORKING_WEEKDAYS:
        d = (d - dt.timedelta(days=1)).replace(hour=18, minute=0, second=0, microsecond=0)
    for start,end in reversed(WORK_WINDOWS):
        if d.time() > end: return d.replace(hour=end.hour, minute=end.minute, second=0, microsecond=0)
        if start <= d.time() <= end: return d
    # before first window → previous day 18:00
    d = (d - dt.timedelta(days=1)).replace(hour=18, minute=0, second=0, microsecond=0)
    while d.weekday() not in WORKING_WEEKDAYS:
        d = (d - dt.timedelta(days=1)).replace(hour=18, minute=0, second=0, microsecond=0)
    return d

def working_minutes_between(a, b):
    if b <= a: return 0
    total = 0
    cur = a
    # iterate day by day, summing overlaps with WORK_WINDOWS on working weekdays
    while cur.date() <= b.date():
        if cur.weekday() in WORKING_WEEKDAYS:
            for start,end in WORK_WINDOWS:
                sw = dt.datetime.combine(cur.date(), start)
                ew = dt.datetime.combine(cur.date(), end)
                s = max(sw, a)
                e = min(ew, b)
                if e > s:
                    total += int((e - s).total_seconds() // 60)
        cur += dt.timedelta(days=1)
    return total

def text(elem, tag, default=''):
    x = elem.find(f'msp:{tag}', NS)
    return x.text if x is not None and x.text is not None else default

def set_text(elem, tag, value):
    x = elem.find(f'msp:{tag}', NS)
    if x is None: x = ET.SubElement(elem, f'{{{MSP_NS}}}{tag}')
    x.text = value

def remove(elem, tag):
    x = elem.find(f'msp:{tag}', NS)
    if x is not None: elem.remove(x)

def main(in_path, out_path):
    tree = ET.parse(in_path)
    root = tree.getroot()

    # enforce project constants
    for k, v in (('MinutesPerDay','480'), ('MinutesPerWeek','2400'),
                 ('DaysPerMonth','20'), ('WorkFormat','2')):
        f = root.find(f'msp:{k}', NS)
        if f is not None: f.text = v

    tasks = root.findall('.//msp:Task', NS)
    rows = []
    for t in tasks:
        summ = text(t,'Summary','0') == '1'
        onum = text(t,'OutlineNumber')
        name = text(t,'Name')
        start = text(t,'Start')
        finish = text(t,'Finish')
        work_m = parse_pt_to_minutes(text(t,'Work','PT0M'))
        rows.append({'elem':t,'summary':summ,'onum':onum,'name':name,
                     'start':start,'finish':finish,'work_m':work_m})

    # Normalize leaves: snap to working time and set Duration to working minutes
    for r in rows:
        if r['summary']: continue
        if r['start'] and r['finish']:
            s = clamp_to_work_window(parse_dt(r['start']))
            f = move_back_into_window(parse_dt(r['finish']))
            if f <= s:
                f = clamp_to_work_window(s + dt.timedelta(minutes=max(r['work_m'], MINUTES_PER_DAY//2)))
            set_text(r['elem'],'Start', s.isoformat())
            set_text(r['elem'],'Finish', f.isoformat())
            dur = working_minutes_between(s,f)
            set_text(r['elem'],'Duration', minutes_to_pt(dur))
            # mirror RemainingDuration if present
            if r['elem'].find('msp:RemainingDuration', NS) is not None:
                set_text(r['elem'],'RemainingDuration', minutes_to_pt(dur))

    # Build outline → descendants index
    leaves_by_parent = {}
    for p in rows:
        prefix = p['onum'] + '.'
        leaves_by_parent[p['onum']] = [c for c in rows if not c['summary'] and c['onum'].startswith(prefix)]

    # Update summaries: Work, Start/Finish, Duration; remove cost fields later
    changes = []
    for r in rows:
        if not r['summary'] or r['onum'] in ('','0'): continue
        leaves = leaves_by_parent.get(r['onum'], [])
        if not leaves:  # no children; leave as-is
            continue
        # compute rollups from leaves
        total_work = sum(c['work_m'] for c in leaves)
        starts = [parse_dt(text(c['elem'],'Start')) for c in leaves if text(c['elem'],'Start')]
        finishes = [parse_dt(text(c['elem'],'Finish')) for c in leaves if text(c['elem'],'Finish')]
        if starts and finishes:
            s = min(starts); f = max(finishes)
            dur = working_minutes_between(clamp_to_work_window(s), move_back_into_window(f))
            old_h = r['work_m']/60.0
            set_text(r['elem'],'Work', minutes_to_pt(total_work))
            set_text(r['elem'],'Start', clamp_to_work_window(s).isoformat())
            set_text(r['elem'],'Finish', move_back_into_window(f).isoformat())
            set_text(r['elem'],'Duration', minutes_to_pt(dur))
            for tag in ('RegularWork','RemainingWork'):
                if r['elem'].find(f'msp:{tag}', NS) is not None:
                    set_text(r['elem'],tag, minutes_to_pt(total_work))
            changes.append((r['name'], old_h, total_work/60.0))
        # strip cost fields
        remove(r['elem'],'Cost'); remove(r['elem'],'FixedCost')

    for r in rows:
        remove(r['elem'],'Cost'); remove(r['elem'],'FixedCost')

    tree.write(out_path, encoding='utf-8', xml_declaration=True)
    print(f"Updated {len(changes)} summary tasks:")
    for name, old_h, new_h in changes:
        print(f"  - {name}: {old_h:.2f}h → {new_h:.2f}h")
    print("Saved:", out_path)

if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Usage: python fix_workfront_xml.py input.xml [output.xml]")
        sys.exit(1)
    in_xml = sys.argv[1]
    out_xml = sys.argv[2] if len(sys.argv) > 2 else in_xml.replace('.xml', '_wf_fixed.xml')
    main(in_xml, out_xml)
