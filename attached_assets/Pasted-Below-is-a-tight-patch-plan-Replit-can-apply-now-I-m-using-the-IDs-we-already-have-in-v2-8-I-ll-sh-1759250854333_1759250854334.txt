We call /api/auto_build and store payload.suggested.

Right: AI Suggestions – a checklist of the suggested deliverables from Step 1 (pre‑checked).

Proceed to Pricing builds A/B with exactly what’s in Your Selection.

3) Render the AI Suggestions checklist (JS)

Add this renderer after your other Step 2 functions:

function mapCodeToName(code) {
  // options.deliverables is loaded from /api/options at init
  const d = (APB.options?.deliverables || []).find(x => String(x.Deliverable_Code) === String(code));
  return d ? d.Deliverable : String(code);
}

function renderAISuggestions() {
  const box = document.getElementById("s2-ai-suggest"); // right column container
  if (!box) return;

  // If user hasn’t run Step 1 yet
  if (!APB.suggested || APB.suggested.length === 0) {
    box.innerHTML = `<div class="muted">Run <b>Analyze with AI</b> in Step 1 to populate suggestions.</div>`;
    return;
  }

  // Build list with checkboxes
  const rows = APB.suggested.map(s => {
    const code = String(s.deliverable_code);
    const name = s.label || mapCodeToName(code);
    const checked = APB.suggestedChecked.has(code) ? "checked" : "";
    return `
      <label class="row chk">
        <input type="checkbox" data-ai-suggest-code="${code}" ${checked}>
        <span>${name}</span>
      </label>`;
  }).join("");

  box.innerHTML = `
    <div class="toolbar">
      <button id="s2-ai-selall" class="btn btn-xs">Select all</button>
      <button id="s2-ai-clear"  class="btn btn-xs">Clear</button>
    </div>
    <div class="list scroll">${rows}</div>
  `;

  // Wire events
  box.querySelectorAll('input[type="checkbox"][data-ai-suggest-code]').forEach(inp => {
    inp.addEventListener("change", (e) => {
      const code = e.target.getAttribute("data-ai-suggest-code");
      if (e.target.checked) APB.suggestedChecked.add(code);
      else APB.suggestedChecked.delete(code);
    });
  });
  box.querySelector("#s2-ai-selall")?.addEventListener("click", () => {
    APB.suggested.forEach(s => APB.suggestedChecked.add(String(s.deliverable_code)));
    renderAISuggestions();
  });
  box.querySelector("#s2-ai-clear")?.addEventListener("click", () => {
    APB.suggestedChecked.clear();
    renderAISuggestions();
  });
}

4) Merge Left + AI when Apply selection is clicked (JS)

Make sure your existing “Apply selection” handler adds the checked AI suggestions before painting Your Selection:

function getLeftPanelCheckedCodes() {
  const list = document.getElementById("s2-deliv-list");
  if (!list) return [];
  const out = [];
  list.querySelectorAll('input[type="checkbox"][data-deliv-code]:checked').forEach(inp => {
    out.push(String(inp.getAttribute("data-deliv-code")));
  });
  return out;
}

document.getElementById("s2-apply")?.addEventListener("click", () => {
  const left = getLeftPanelCheckedCodes();
  const ai = Array.from(APB.suggestedChecked || []);
  const merged = Array.from(new Set([ ...left, ...ai ]));

  if (merged.length === 0) {
    alert("Please select at least one deliverable.");
    return;
  }

  APB.selectedCodes = new Set(merged);

  // Repaint "Your Selection" (middle column)
  renderYourSelection();

  // Optional UX: scroll to the middle column, flash success, etc.
});
